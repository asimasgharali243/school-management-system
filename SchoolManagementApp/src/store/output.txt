import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { BASE_URL, API_TIMEOUT } from '../config/constants';
import { logout } from './slices/authSlice';

// Base query with authentication
const baseQuery = fetchBaseQuery({
    baseUrl: BASE_URL,
    prepareHeaders: async (headers) => {
        const token = await AsyncStorage.getItem('token');
        if (token) {
            headers.set('Authorization', `Bearer ${token}`);
        }
        headers.set('Content-Type', 'application/json');
        headers.set('Accept', 'application/json');
        return headers;
    },
    timeout: API_TIMEOUT,
});

// Custom base query with error handling
const baseQueryWithReauth = async (args, api, extraOptions) => {
    let result = await baseQuery(args, api, extraOptions);

    if (result.error) {
        console.log('API Error:', result.error);

        // Handle 401 Unauthorized
        if (result.error.status === 401) {
            // Clear stored tokens
            await AsyncStorage.removeItem('token');
            await AsyncStorage.removeItem('user');

            // Dispatch logout action
            api.dispatch(logout());

            // You can redirect to login screen here if needed
        }

        // Handle network errors
        if (result.error.status === 'FETCH_ERROR') {
            return {
                error: {
                    status: 'NETWORK_ERROR',
                    data: { message: 'Network connection failed. Please check your internet connection.' }
                }
            };
        }
    }

    return result;
};

export const apiSlice = createApi({
    reducerPath: 'api',
    baseQuery: baseQueryWithReauth,
    tagTypes: [
        'User',
        'Student',
        'Teacher',
        'Class',
        'Section',
        'Attendance',
        'Fee',
        'Notification',
        'Campus'
    ],
    endpoints: () => ({}),
});

            query: ({ childId, filters = {} }) => {

                // Add filters if they exist

                if (filters.batchId) params.append('batchId', filters.batchId);
                if (filters.yearId) params.append('yearId', filters.yearId);

                if (filters.month) params.append('month', filters.month);
                if (filters.year) params.append('year', filters.year);

                // Handle both wrapped and unwrapped responses

                    return response.data || [];

                return response || [];

                if (response?.success !== undefined) {
                    return response.data || [];
                }
                return response || [];
            },

        getGradeTrends: builder.query({
            query: ({ childId, subjectId }) =>
                `/api/grades/child/${childId}/trends${subjectId ? `?subjectId=${subjectId}` : ''}`,
            transformResponse: (response) => {
                if (response?.success !== undefined) {
                    return response.data || [];
                }
                return response || [];
            },
            providesTags: ['Grades'],
        }),

        // Download grade report
        downloadGradeReport: builder.mutation({
            query: ({ childId, filters = {} }) => ({
                url: `/api/grades/child/${childId}/report`,
                method: 'GET',
                params: filters,
                responseHandler: async (response) => {
                    const contentType = response.headers.get('content-type');

                    if (contentType?.includes('application/json')) {
                        const data = await response.json();
                        return data;
                    }

                    // Handle PDF download
                    const blob = await response.blob();
                    const url = window.URL.createObjectURL(blob);
                    const link = document.createElement('a');
                    link.href = url;
                    link.setAttribute('download', `grade-report-${childId}.pdf`);
                    document.body.appendChild(link);
                    link.click();
                    link.remove();
                    window.URL.revokeObjectURL(url);

                    return { success: true, message: 'Report downloaded successfully' };
                },
            }),
        }),
    }),
});

export const loginUser = createAsyncThunk(
    'auth/login',
    async ({ username, password }, { rejectWithValue, dispatch }) => {
        try {
            console.log('🔐 Login attempt for:', username);

            const response = await fetch(`${BASE_URL}/api/Auth/login`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                    'Accept': 'application/json',
                },
                body: JSON.stringify({
                    username: username.trim(),
                    password: password,
                }),
            });

            console.log('📤 Response status:', response.status);

            const data = await response.json();

            });

            if (!response.ok) {
                throw new Error(data.message || 'Login failed');
            }

            // Store tokens and user data
            console.log('💾 Storing tokens and user data...');
            await AsyncStorage.multiSet([
                ['token', data.token],
                ['user', JSON.stringify(data.user)],
                ['children', JSON.stringify(data.children || [])],
                ['user_id', String(data.user?.id || '')],
            ]);

            // Dispatch the credentials to update Redux state immediately
            dispatch(setCredentials({
                user: data.user,
                token: data.token,
                children: data.children || []
            }));

            // Initialize notifications for the user with safety check
            if (data.user) {
                console.log('🔔 Attempting to register user for notifications...');
                setTimeout(async () => {
                    try {
                        if (firebaseNotifications && typeof firebaseNotifications.registerUser === 'function') {
                            const success = await firebaseNotifications.registerUser(data.user);
                            if (success) {
                                console.log('✅ User registered for notifications');
                            } else {
                                console.log('⚠️ User registration returned false');
                            }
                        } else {
                            console.log('⚠️ Notification service not available yet');
                            await AsyncStorage.setItem('pending_notification_user', JSON.stringify(data.user));
                        }
                    } catch (notifError) {
                        console.error('❌ Failed to register for notifications:', notifError);
                    }
                }, 1000);
            }

            return data;
        } catch (error) {
            console.error('❌ Login error:', error);
            return rejectWithValue(error.message);
        }
    }
);

export const validateToken = createAsyncThunk(
    'auth/validate',
    async (_, { rejectWithValue, getState }) => {
        try {
            console.log('🔐 Starting token validation...');

            // Get token from Redux state first
            let token = getState().auth.token;
            if (!token) {
                console.log('🔐 Token not in Redux state, checking AsyncStorage...');
                token = await AsyncStorage.getItem('token');
            }

            if (!token) {
                console.error('❌ No token found for validation');
                throw new Error('No token found');
            }

            console.log('🔐 Token found, length:', token.length);

            const response = await fetch(`${BASE_URL}/api/Auth/validate`, {
                method: 'GET',
                headers: {
                    'Authorization': `Bearer ${token}`,
                    'Accept': 'application/json',
                    'Content-Type': 'application/json',
                },
            });

            const data = await response.json();
            console.log('🔐 Validation response:', {
                status: response.status,
                hasUser: !!data.user,
                hasChildren: !!(data.children && data.children.length > 0),
                responseData: data
            });

            // FIX: Check response.ok only, not data.success
            if (!response.ok) {
                console.error('❌ Token validation failed with status:', response.status);
                throw new Error(data.message || 'Token validation failed');
            }

            // If we get here with user data, it's successful
            if (!data.user) {
                throw new Error('Invalid response: No user data');
            }

            // Update stored user data
            await AsyncStorage.multiSet([
                ['user', JSON.stringify(data.user)],
                ['children', JSON.stringify(data.children || [])],
            ]);

            // Return the data with the EXISTING token (not from response)
            return {
                user: data.user,
                children: data.children || [],
                token: token  // Keep using the existing token
            };
        } catch (error) {
            console.error('❌ Validate token error:', error);
            return rejectWithValue(error.message);
        }
    }
);

export const initializeAuth = createAsyncThunk(
    'auth/initialize',
    async (_, { dispatch }) => {
        try {
            console.log('🔄 Initializing auth state from AsyncStorage...');

            // Load ALL auth data from AsyncStorage
            const [
                token,
                userStr,
                childrenStr,
            ] = await Promise.all([
                AsyncStorage.getItem('token'),
                AsyncStorage.getItem('user'),
                AsyncStorage.getItem('children'),
            ]);

            console.log('📥 Loaded from storage:', {
                hasToken: !!token,
                tokenLength: token?.length,
                tokenPreview: token ? `${token.substring(0, 30)}...` : 'No token',
                hasUser: !!userStr,
                hasChildren: !!childrenStr
            });

            let user = null;
            let children = [];

            try {
                user = userStr ? JSON.parse(userStr) : null;
                children = childrenStr ? JSON.parse(childrenStr) : [];
            } catch (parseError) {
                console.error('❌ Error parsing stored data:', parseError);
            }

            // Set initial state with loaded data
            if (token && user) {
                // CRITICAL: Set the token in Redux state
                dispatch(setInitialState({
                    token,
                    user,
                    children,
                    isAuthenticated: true,
                    isInitialized: true,
                    isLoading: false,
                }));

                console.log('✅ Token loaded into Redux state:', {
                    stateHasToken: !!token,
                    userLoaded: !!user,
                    childrenCount: children.length
                });

                // Then validate the token
                try {
                    const result = await dispatch(validateToken()).unwrap();
                    console.log('✅ Token validated successfully');

                    // Update with validated data
                    dispatch(setCredentials({
                        user: result.user,
                        token: token, // Keep the same token
                        children: result.children || []
                    }));

                    return {
                        isAuthenticated: true,
                        token: token
                    };
                } catch (validationError) {
                    console.error('❌ Token validation failed:', validationError);
                    // Token is invalid, clear it
                    await AsyncStorage.multiRemove(['token']);
                    dispatch(logout());
                    return { isAuthenticated: false };
                }
            } else {
                console.log('ℹ️ No valid token or user found in storage');
                dispatch(setInitialState({
                    token: null,
                    user: null,
                    children: [],
                    isAuthenticated: false,
                    isInitialized: true,
                    isLoading: false,
                }));
                return { isAuthenticated: false };
            }
        } catch (error) {
            console.error('❌ Auth initialization error:', error);
            dispatch(setInitialState({
                token: null,
                user: null,
                children: [],
                isAuthenticated: false,
                isInitialized: true,
                isLoading: false,
            }));
            return { isAuthenticated: false };
        }
    }
);

const initialState = {

    unreadNotifications: 0,
    settings: {
        notifications: true,
        biometricLogin: false,
        autoSync: true,
        language: 'en',
    },
};

const userSlice = createSlice({
    name: 'user',
    initialState,
    reducers: {
        setProfile: (state, action) => {
            state.profile = action.payload;
        },
        updateProfile: (state, action) => {
            if (state.profile) {
                state.profile = { ...state.profile, ...action.payload };
            }
        },
        addNotification: (state, action) => {
            state.notifications.unshift({
                ...action.payload,
                id: Date.now().toString(),
                timestamp: new Date().toISOString(),
                read: false,
            });
            state.unreadNotifications += 1;
        },
        markNotificationAsRead: (state, action) => {
            const notification = state.notifications.find(n => n.id === action.payload);
            if (notification && !notification.read) {
                notification.read = true;
                state.unreadNotifications = Math.max(0, state.unreadNotifications - 1);
            }
        },
        markAllNotificationsAsRead: (state) => {
            state.notifications.forEach(notification => {
                notification.read = true;
            });
            state.unreadNotifications = 0;
        },
        clearNotifications: (state) => {
            state.notifications = [];
            state.unreadNotifications = 0;
        },
        updateSettings: (state, action) => {
            state.settings = { ...state.settings, ...action.payload };
        },
    },
});

export const {
    setProfile,
    updateProfile,
    addNotification,
    markNotificationAsRead,
    markAllNotificationsAsRead,
    clearNotifications,
    updateSettings,
} = userSlice.actions;
