// src/screens/Teacher/AttendanceReportScreen.js
import React, { useState, useEffect } from 'react';
import {
    View,
    Text,
    StyleSheet,
    ScrollView,
    TouchableOpacity,
    ActivityIndicator,
    Dimensions,
} from 'react-native';
import { useSelector, useDispatch } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialCommunityIcons';
import { selectThemeData } from '../../store/slices/themeSlice';
import { fetchAttendanceHistory } from '../../store/slices/teacherSlice';

const { width } = Dimensions.get('window');

const AttendanceReportScreen = ({ navigation, route }) => {
    const theme = useSelector(selectThemeData);
    const dispatch = useDispatch();
    const { classId, className, sectionId, sectionName } = route.params;

    const [selectedMonth, setSelectedMonth] = useState(new Date().getMonth() + 1);
    const [selectedYear, setSelectedYear] = useState(new Date().getFullYear());
    const [loading, setLoading] = useState(false);
    const [history, setHistory] = useState([]);

    const months = [
        'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
        'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'
    ];

    const years = [2024, 2025, 2026];

    useEffect(() => {
        loadAttendanceHistory();
    }, [selectedMonth, selectedYear]);

    const loadAttendanceHistory = async () => {
        setLoading(true);
        const result = await dispatch(fetchAttendanceHistory({
            classId,
            sectionId,
            month: selectedMonth,
            year: selectedYear
        }));

        if (result.payload?.success) {
            setHistory(result.payload.history || []);
        }
        setLoading(false);
    };

    const getMonthData = () => {
        const monthData = Array.from({ length: 31 }, (_, i) => {
            const day = i + 1;
            const dateStr = `${selectedYear}-${selectedMonth.toString().padStart(2, '0')}-${day.toString().padStart(2, '0')}`;
            const dayData = history.find(h => h.date === dateStr);

            return {
                day,
                date: dateStr,
                present: dayData?.presentCount || 0,
                total: dayData?.totalStudents || 0,
                attendance: dayData ? true : false
            };
        });

        return monthData;
    };

    const calculateMonthlyStats = () => {
        const attendedDays = history.filter(h => h.totalStudents > 0).length;
        const totalPresent = history.reduce((sum, day) => sum + (day.presentCount || 0), 0);
        const totalAbsent = history.reduce((sum, day) => sum + (day.absentCount || 0), 0);
        const totalLeave = history.reduce((sum, day) => sum + (day.leaveCount || 0), 0);
        const totalStudents = history.length > 0 ? history[0].totalStudents : 0;

        return {
            attendedDays,
            totalPresent,
            totalAbsent,
            totalLeave,
            attendancePercentage: attendedDays > 0 ? Math.round((totalPresent / (attendedDays * totalStudents)) * 100) : 0
        };
    };

    const stats = calculateMonthlyStats();

    if (loading) {
        return (
            <View style={[styles.container, { backgroundColor: theme.background }]}>
                <ActivityIndicator size="large" color={theme.primary} />
            </View>
        );
    }

    return (
        <ScrollView style={[styles.container, { backgroundColor: theme.background }]}>
            {/* Header */}
            <View style={[styles.header, { backgroundColor: theme.card }]}>
                <TouchableOpacity onPress={() => navigation.goBack()}>
                    <Icon name="arrow-left" size={24} color={theme.text} />
                </TouchableOpacity>
                <View style={styles.headerInfo}>
                    <Text style={[styles.headerTitle, { color: theme.text }]}>
                        {className} - {sectionName}
                    </Text>
                    <Text style={[styles.headerSubtitle, { color: theme.textSecondary }]}>
                        Attendance Report
                    </Text>
                </View>
            </View>

            {/* Month/Year Selector */}
            <View style={[styles.selectorContainer, { backgroundColor: theme.card }]}>
                <View style={styles.selectorRow}>
                    <Text style={[styles.selectorLabel, { color: theme.text }]}>Month:</Text>
                    <ScrollView horizontal showsHorizontalScrollIndicator={false}>
                        {months.map((month, index) => (
                            <TouchableOpacity
                                key={index}
                                style={[
                                    styles.selectorButton,
                                    selectedMonth === index + 1 && { backgroundColor: theme.primary }
                                ]}
                                onPress={() => setSelectedMonth(index + 1)}
                            >
                                <Text style={[
                                    styles.selectorButtonText,
                                    selectedMonth === index + 1 && { color: '#FFFFFF' },
                                    { color: theme.text }
                                ]}>
                                    {month}
                                </Text>
                            </TouchableOpacity>
                        ))}
                    </ScrollView>
                </View>

                <View style={styles.selectorRow}>
                    <Text style={[styles.selectorLabel, { color: theme.text }]}>Year:</Text>
                    <ScrollView horizontal showsHorizontalScrollIndicator={false}>
                        {years.map((year) => (
                            <TouchableOpacity
                                key={year}
                                style={[
                                    styles.selectorButton,
                                    selectedYear === year && { backgroundColor: theme.primary }
                                ]}
                                onPress={() => setSelectedYear(year)}
                            >
                                <Text style={[
                                    styles.selectorButtonText,
                                    selectedYear === year && { color: '#FFFFFF' },
                                    { color: theme.text }
                                ]}>
                                    {year}
                                </Text>
                            </TouchableOpacity>
                        ))}
                    </ScrollView>
                </View>
            </View>

            {/* Monthly Stats */}
            <View style={styles.statsGrid}>
                <View style={[styles.statCard, { backgroundColor: theme.card }]}>
                    <Icon name="calendar-check" size={24} color={theme.primary} />
                    <Text style={[styles.statValue, { color: theme.text }]}>
                        {stats.attendedDays}
                    </Text>
                    <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                        Days Attended
                    </Text>
                </View>

                <View style={[styles.statCard, { backgroundColor: theme.card }]}>
                    <Icon name="chart-line" size={24} color={theme.success} />
                    <Text style={[styles.statValue, { color: theme.text }]}>
                        {stats.attendancePercentage}%
                    </Text>
                    <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                        Attendance Rate
                    </Text>
                </View>
            </View>

            <View style={styles.statsGrid}>
                <View style={[styles.statCard, { backgroundColor: theme.card }]}>
                    <Icon name="account-check" size={24} color={theme.success} />
                    <Text style={[styles.statValue, { color: theme.text }]}>
                        {stats.totalPresent}
                    </Text>
                    <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                        Total Present
                    </Text>
                </View>

                <View style={[styles.statCard, { backgroundColor: theme.card }]}>
                    <Icon name="account-cancel" size={24} color={theme.error} />
                    <Text style={[styles.statValue, { color: theme.text }]}>
                        {stats.totalAbsent}
                    </Text>
                    <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                        Total Absent
                    </Text>
                </View>

                <View style={[styles.statCard, { backgroundColor: theme.card }]}>
                    <Icon name="calendar-clock" size={24} color={theme.warning} />
                    <Text style={[styles.statValue, { color: theme.text }]}>
                        {stats.totalLeave}
                    </Text>
                    <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                        Total Leave
                    </Text>
                </View>
            </View>

            {/* Calendar View */}
            <View style={[styles.calendarContainer, { backgroundColor: theme.card }]}>
                <Text style={[styles.calendarTitle, { color: theme.text }]}>
                    Daily Attendance - {months[selectedMonth - 1]} {selectedYear}
                </Text>

                <View style={styles.weekHeader}>
                    {['S', 'M', 'T', 'W', 'T', 'F', 'S'].map((day, index) => (
                        <Text key={index} style={[styles.weekDay, { color: theme.textSecondary }]}>
                            {day}
                        </Text>
                    ))}
                </View>

                <View style={styles.calendarGrid}>
                    {getMonthData().map((dayData, index) => {
                        const percentage = dayData.total > 0
                            ? Math.round((dayData.present / dayData.total) * 100)
                            : 0;

                        let bgColor = theme.background;
                        if (dayData.attendance) {
                            if (percentage >= 90) bgColor = theme.success + '40';
                            else if (percentage >= 75) bgColor = theme.warning + '40';
                            else bgColor = theme.error + '40';
                        }

                        return (
                            <View key={index} style={[styles.dayCell, { backgroundColor: bgColor }]}>
                                <Text style={[styles.dayNumber, { color: theme.text }]}>
                                    {dayData.day}
                                </Text>
                                {dayData.attendance && (
                                    <>
                                        <Text style={[styles.dayPresent, { color: theme.text }]}>
                                            {dayData.present}
                                        </Text>
                                        <Text style={[styles.dayTotal, { color: theme.textSecondary }]}>
                                            /{dayData.total}
                                        </Text>
                                        <Text style={[styles.dayPercentage, { color: theme.textSecondary }]}>
                                            {percentage}%
                                        </Text>
                                    </>
                                )}
                            </View>
                        );
                    })}
                </View>

                <View style={styles.legendContainer}>
                    <View style={styles.legendItem}>
                        <View style={[styles.legendColor, { backgroundColor: theme.success + '40' }]} />
                        <Text style={[styles.legendText, { color: theme.textSecondary }]}>
                            ≥90% Attendance
                        </Text>
                    </View>
                    <View style={styles.legendItem}>
                        <View style={[styles.legendColor, { backgroundColor: theme.warning + '40' }]} />
                        <Text style={[styles.legendText, { color: theme.textSecondary }]}>
                            75-89% Attendance
                        </Text>
                    </View>
                    <View style={styles.legendItem}>
                        <View style={[styles.legendColor, { backgroundColor: theme.error + '40' }]} />
                        <Text style={[styles.legendText, { color: theme.textSecondary }]}>
                            &lt;75% Attendance
                        </Text>
                    </View>
                </View>
            </View>

            {/* Daily Details */}
            <View style={[styles.detailsContainer, { backgroundColor: theme.card }]}>
                <Text style={[styles.detailsTitle, { color: theme.text }]}>
                    Daily Attendance Details
                </Text>

                {history.length > 0 ? (
                    history.map((day, index) => (
                        <View key={index} style={styles.detailItem}>
                            <View style={styles.detailDate}>
                                <Text style={[styles.detailDateText, { color: theme.text }]}>
                                    {new Date(day.date).toLocaleDateString('en-US', {
                                        weekday: 'short',
                                        month: 'short',
                                        day: 'numeric'
                                    })}
                                </Text>
                            </View>

                            <View style={styles.detailStats}>
                                <View style={[styles.detailStat, { backgroundColor: theme.success + '20' }]}>
                                    <Icon name="check" size={14} color={theme.success} />
                                    <Text style={[styles.detailStatText, { color: theme.text }]}>
                                        {day.presentCount}
                                    </Text>
                                </View>

                                <View style={[styles.detailStat, { backgroundColor: theme.error + '20' }]}>
                                    <Icon name="close" size={14} color={theme.error} />
                                    <Text style={[styles.detailStatText, { color: theme.text }]}>
                                        {day.absentCount}
                                    </Text>
                                </View>

                                <View style={[styles.detailStat, { backgroundColor: theme.warning + '20' }]}>
                                    <Icon name="clock" size={14} color={theme.warning} />
                                    <Text style={[styles.detailStatText, { color: theme.text }]}>
                                        {day.leaveCount}
                                    </Text>
                                </View>

                                {day.lateCount > 0 && (
                                    <View style={[styles.detailStat, { backgroundColor: theme.secondary + '20' }]}>
                                        <Icon name="clock-alert" size={14} color={theme.secondary} />
                                        <Text style={[styles.detailStatText, { color: theme.text }]}>
                                            {day.lateCount}
                                        </Text>
                                    </View>
                                )}
                            </View>
                        </View>
                    ))
                ) : (
                    <View style={styles.noDataContainer}>
                        <Icon name="calendar-remove" size={50} color={theme.textSecondary} />
                        <Text style={[styles.noDataText, { color: theme.textSecondary }]}>
                            No attendance data for {months[selectedMonth - 1]} {selectedYear}
                        </Text>
                    </View>
                )}
            </View>
        </ScrollView>
    );
};

// src/screens/Teacher/AttendanceScreen.js
import React, { useState, useEffect } from 'react';
import {
    View,
    Text,
    StyleSheet,
    ScrollView,
    TouchableOpacity,
    FlatList,
    ActivityIndicator,
    RefreshControl,
    Modal,
    Alert,
} from 'react-native';
import { useSelector, useDispatch } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialCommunityIcons';
import { selectThemeData } from '../../store/slices/themeSlice';
import {
    fetchTeacherClasses,
    selectTeacherClasses,
    selectTeacherLoading,
    clearError,
    fetchTodayAttendance,
    markAttendance,
    updateAttendanceStatus,
    clearAttendanceData
} from '../../store/slices/teacherSlice';

const AttendanceScreen = ({ navigation, route }) => {
    const theme = useSelector(selectThemeData);
    const dispatch = useDispatch();
    const classes = useSelector(selectTeacherClasses);
    const loading = useSelector((state) => state.teacher.loading);
    const attendanceLoading = useSelector((state) => state.teacher.attendanceLoading);

    const [selectedDate, setSelectedDate] = useState(new Date().toISOString().split('T')[0]);
    const [selectedClass, setSelectedClass] = useState(null);
    const [showClassModal, setShowClassModal] = useState(false);
    const [refreshing, setRefreshing] = useState(false);

    // Check if class is passed from navigation
    useEffect(() => {
        if (route.params?.classId) {
            const classItem = classes.find(
                c => c.classId === route.params.classId &&
                    c.sectionId === route.params.sectionId
            );
            if (classItem) {
                setSelectedClass(classItem);
                loadAttendance(classItem);
            }
        }
    }, [route.params, classes]);

    useEffect(() => {
        loadClasses();
    }, []);

    const loadClasses = async () => {
        await dispatch(fetchTeacherClasses());
    };

    const loadAttendance = async (classItem) => {
        await dispatch(fetchTodayAttendance({
            classId: classItem.classId,
            sectionId: classItem.sectionId
        }));
    };

    const onRefresh = async () => {
        setRefreshing(true);
        if (selectedClass) {
            await loadAttendance(selectedClass);
        }
        await loadClasses();
        setRefreshing(false);
    };

    const handleClassSelect = (classItem) => {
        setSelectedClass(classItem);
        setShowClassModal(false);
        loadAttendance(classItem);
    };

    const handleTakeAttendance = () => {
        if (!selectedClass) {
            Alert.alert('Select Class', 'Please select a class first');
            return;
        }
        navigation.navigate('MarkAttendance', {
            classId: selectedClass.classId,
            className: selectedClass.className,
            sectionId: selectedClass.sectionId,
            sectionName: selectedClass.sectionName
        });
    };

    const handleViewAttendance = () => {
        if (!selectedClass) {
            Alert.alert('Select Class', 'Please select a class first');
            return;
        }
        navigation.navigate('AttendanceReport', {
            classId: selectedClass.classId,
            className: selectedClass.className,
            sectionId: selectedClass.sectionId,
            sectionName: selectedClass.sectionName
        });
    };

    const renderClassItem = ({ item }) => (
        <TouchableOpacity
            style={[styles.classItem, { backgroundColor: theme.card }]}
            onPress={() => handleClassSelect(item)}
        >
            <View style={styles.classItemContent}>
                <View style={[styles.classIcon, { backgroundColor: theme.primary + '20' }]}>
                    <Icon name="school" size={20} color={theme.primary} />
                </View>
                <View style={styles.classItemInfo}>
                    <Text style={[styles.classItemName, { color: theme.text }]}>
                        {item.className} - {item.sectionName}
                    </Text>
                    <Text style={[styles.classItemSubject, { color: theme.textSecondary }]}>
                        {item.subjectName || 'Class Incharge'}
                    </Text>
                </View>
            </View>
        </TouchableOpacity>
    );

    const renderClassModal = () => (
        <Modal
            visible={showClassModal}
            animationType="slide"
            transparent={true}
            onRequestClose={() => setShowClassModal(false)}
        >
            <View style={styles.modalOverlay}>
                <View style={[styles.modalContent, { backgroundColor: theme.card }]}>
                    <View style={styles.modalHeader}>
                        <Text style={[styles.modalTitle, { color: theme.text }]}>
                            Select Class
                        </Text>
                        <TouchableOpacity onPress={() => setShowClassModal(false)}>
                            <Icon name="close" size={24} color={theme.text} />
                        </TouchableOpacity>
                    </View>

                    <FlatList
                        data={classes}
                        renderItem={renderClassItem}
                        keyExtractor={(item) => `${item.classId}-${item.sectionId}`}
                        style={styles.classList}
                    />
                </View>
            </View>
        </Modal>
    );

    if (loading) {
        return (
            <View style={[styles.container, { backgroundColor: theme.background }]}>
                <ActivityIndicator size="large" color={theme.primary} />
            </View>
        );
    }

    return (
        <ScrollView
            style={[styles.container, { backgroundColor: theme.background }]}
            refreshControl={
                <RefreshControl
                    refreshing={refreshing}
                    onRefresh={onRefresh}
                    colors={[theme.primary]}
                />
            }
        >
            {/* Header */}
            <View style={[styles.header, { backgroundColor: theme.card }]}>
                <View>
                    <Text style={[styles.title, { color: theme.text }]}>
                        Attendance Management
                    </Text>
                    <Text style={[styles.subtitle, { color: theme.textSecondary }]}>
                        Track and manage student attendance
                    </Text>
                </View>
                <Icon name="calendar-check" size={40} color={theme.primary} />
            </View>

            {/* Class Selection */}
            <View style={[styles.section, { backgroundColor: theme.card }]}>
                <Text style={[styles.sectionTitle, { color: theme.text }]}>
                    Select Class
                </Text>

                <TouchableOpacity
                    style={[styles.classSelector, { backgroundColor: theme.background }]}
                    onPress={() => setShowClassModal(true)}
                >
                    {selectedClass ? (
                        <View style={styles.selectedClass}>
                            <Text style={[styles.selectedClassName, { color: theme.text }]}>
                                {selectedClass.className} - {selectedClass.sectionName}
                            </Text>
                            <Text style={[styles.selectedClassSubject, { color: theme.textSecondary }]}>
                                {selectedClass.subjectName}
                            </Text>
                        </View>
                    ) : (
                        <Text style={[styles.placeholder, { color: theme.textSecondary }]}>
                            Tap to select class
                        </Text>
                    )}
                    <Icon name="chevron-down" size={24} color={theme.textSecondary} />
                </TouchableOpacity>
            </View>

            {/* Action Buttons */}
            {selectedClass && (
                <View style={styles.actionsContainer}>
                    <TouchableOpacity
                        style={[styles.actionButton, { backgroundColor: theme.primary }]}
                        onPress={handleTakeAttendance}
                        disabled={attendanceLoading}
                    >
                        {attendanceLoading ? (
                            <ActivityIndicator color="#FFFFFF" />
                        ) : (
                            <>
                                <Icon name="clipboard-check" size={20} color="#FFFFFF" />
                                <Text style={styles.actionButtonText}>Take Attendance</Text>
                            </>
                        )}
                    </TouchableOpacity>

                    <TouchableOpacity
                        style={[styles.actionButton, { backgroundColor: theme.secondary }]}
                        onPress={handleViewAttendance}
                    >
                        <Icon name="chart-bar" size={20} color="#FFFFFF" />
                        <Text style={styles.actionButtonText}>View Reports</Text>
                    </TouchableOpacity>
                </View>
            )}

            {/* Today's Summary */}
            <View style={[styles.section, { backgroundColor: theme.card }]}>
                <Text style={[styles.sectionTitle, { color: theme.text }]}>
                    Today's Summary ({selectedDate})
                </Text>

const calculateAverageGrade = (grades) => {
    const gradeValues = {
        'A+': 4.0, 'A': 4.0, 'A-': 3.7,
        'B+': 3.3, 'B': 3.0, 'B-': 2.7,
        'C+': 2.3, 'C': 2.0, 'D': 1.0, 'F': 0.0,
    };

    const total = grades.reduce((sum, grade) => {
        return sum + (gradeValues[grade.grade] || 0);
    }, 0);

    return (total / grades.length).toFixed(1) || '0.0';
};

import React, { useState, useEffect } from 'react';
import {
    View,
    Text,
    StyleSheet,
    ScrollView,
    TouchableOpacity,
    TextInput,
    Alert,
    ActivityIndicator,
    Modal,
    FlatList,
} from 'react-native';
import { useSelector, useDispatch } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialCommunityIcons';
import { selectThemeData } from '../../store/slices/themeSlice';
import {
    fetchClassStudents,
    fetchTodayAttendance,
    markAttendance,
    updateAttendanceStatus,
    selectTeacherStudents,
    selectTeacherAttendance,
    selectTeacherTodayAttendance,
    clearAttendanceData
} from '../../store/slices/teacherSlice';

const MarkAttendanceScreen = ({ navigation, route }) => {
    const theme = useSelector(selectThemeData);
    const dispatch = useDispatch();
    const { classId, className, sectionId, sectionName } = route.params;

    const students = useSelector(selectTeacherStudents);
    const attendance = useSelector((state) => state.teacher.attendance);
    const todayAttendance = useSelector(selectTeacherTodayAttendance);
    const loading = useSelector((state) => state.teacher.loading);
    const attendanceLoading = useSelector((state) => state.teacher.attendanceLoading);

    const [remarks, setRemarks] = useState('');
    const [showRemarksModal, setShowRemarksModal] = useState(false);
    const [selectedStudent, setSelectedStudent] = useState(null);
    const [lateModal, setLateModal] = useState(false);
    const [leaveModal, setLeaveModal] = useState(false);

    useEffect(() => {
        loadData();
    }, []);

    const loadData = async () => {
        await Promise.all([
            dispatch(fetchClassStudents({ classId, sectionId })),
            dispatch(fetchTodayAttendance({ classId, sectionId }))
        ]);
    };

    const getStudentAttendance = (studentId) => {
        const attendanceRecord = attendance.find(a => a.studentId === studentId);
        if (attendanceRecord) {
            return {
                status: attendanceRecord.status,
                lateStatus: attendanceRecord.lateStatus,
                remarks: attendanceRecord.remarks
            };
        }

        // If today's attendance already exists, use that
        if (todayAttendance?.attendanceTaken && todayAttendance?.attendanceRecords) {
            const todayRecord = todayAttendance.attendanceRecords.find(r => r.studentId === studentId);
            if (todayRecord) {
                return {
                    status: todayRecord.status,
                    lateStatus: todayRecord.lateStatus,
                    remarks: todayRecord.remarks
                };
            }
        }

        // Default to present
        return {
            status: 'P',
            lateStatus: 'OnTime',
            remarks: ''
        };
    };

    const handleStatusChange = (studentId, newStatus) => {
        const currentAttendance = getStudentAttendance(studentId);

        switch (newStatus) {
            case 'P':
                setSelectedStudent(studentId);
                setLateModal(true);
                break;
            case 'A':
                dispatch(updateAttendanceStatus({
                    studentId,
                    status: 'A',
                    lateStatus: '',
                    remarks: 'Absent'
                }));
                break;
            case 'L':
                setSelectedStudent(studentId);
                setLeaveModal(true);
                break;
        }
    };

    const handleLateSelect = (isLate) => {
        if (selectedStudent) {
            dispatch(updateAttendanceStatus({
                studentId: selectedStudent,
                status: 'P',
                lateStatus: isLate ? 'Late' : 'OnTime',
                remarks: isLate ? 'Late arrival' : ''
            }));
            setLateModal(false);
        }
    };

    const handleLeaveConfirm = (leaveRemarks) => {
        if (selectedStudent) {
            dispatch(updateAttendanceStatus({
                studentId: selectedStudent,
                status: 'L',
                lateStatus: '',
                remarks: leaveRemarks || 'Leave'
            }));
            setLeaveModal(false);
            setSelectedStudent(null);
        }
    };

    const handleSaveAttendance = async () => {
        if (students.length === 0) {
            Alert.alert('No Students', 'No students found in this class');
            return;
        }

        // Prepare attendance records
        const attendanceRecords = students.map(student => {
            const studentAttendance = getStudentAttendance(student.studentId);
            return {
                studentId: student.studentId,
                status: studentAttendance.status,
                lateStatus: studentAttendance.lateStatus,
                remarks: studentAttendance.remarks
            };
        });

        const attendanceData = {
            classId,
            sectionId,
            attendanceRecords,
            remarks
        };

        const result = await dispatch(markAttendance(attendanceData));

        if (result.payload?.success) {
            Alert.alert(
                'Success',
                `Attendance saved successfully for ${students.length} students`,
                [
                    {
                        text: 'OK',
                        onPress: () => navigation.goBack()
                    }
                ]
            );
        } else {
            Alert.alert('Error', result.payload?.message || 'Failed to save attendance');
        }
    };

    const getStatusColor = (status) => {
        switch (status) {
            case 'P': return theme.success;
            case 'A': return theme.error;
            case 'L': return theme.warning;
            default: return theme.textSecondary;
        }
    };

    const getStatusIcon = (status) => {
        switch (status) {
            case 'P': return 'check-circle';
            case 'A': return 'close-circle';
            case 'L': return 'calendar-clock';
            default: return 'circle-outline';
        }
    };

    const renderStudentItem = ({ item }) => {
        const attendance = getStudentAttendance(item.studentId);

        return (
            <View style={[styles.studentCard, { backgroundColor: theme.card }]}>
                <View style={styles.studentInfo}>
                    <View style={[styles.avatar, { backgroundColor: theme.primary + '20' }]}>
                        <Text style={[styles.avatarText, { color: theme.primary }]}>
                            {item.name.charAt(0).toUpperCase()}
                        </Text>
                    </View>
                    <View style={styles.studentDetails}>
                        <Text style={[styles.studentName, { color: theme.text }]}>
                            {item.name}
                        </Text>
                        <Text style={[styles.studentRollNo, { color: theme.textSecondary }]}>
                            Roll No: {item.rollNo}
                        </Text>
                        {attendance.remarks && (
                            <Text style={[styles.remarks, { color: theme.textSecondary }]}>
                                {attendance.remarks}
                            </Text>
                        )}
                    </View>
                </View>

                <View style={styles.statusContainer}>
                    {/* Present Button */}
                    <TouchableOpacity
                        style={[
                            styles.statusButton,
                            attendance.status === 'P' && { backgroundColor: theme.success + '20' },
                            { borderColor: attendance.status === 'P' ? theme.success : theme.border }
                        ]}
                        onPress={() => handleStatusChange(item.studentId, 'P')}
                    >
                        <Icon
                            name={attendance.status === 'P' ? 'check-circle' : 'circle-outline'}
                            size={20}
                            color={attendance.status === 'P' ? theme.success : theme.textSecondary}
                        />
                        <Text style={[
                            styles.statusText,
                            { color: attendance.status === 'P' ? theme.success : theme.textSecondary }
                        ]}>
                            P
                        </Text>
                    </TouchableOpacity>

                    {/* Absent Button */}
                    <TouchableOpacity
                        style={[
                            styles.statusButton,
                            attendance.status === 'A' && { backgroundColor: theme.error + '20' },
                            { borderColor: attendance.status === 'A' ? theme.error : theme.border }
                        ]}
                        onPress={() => handleStatusChange(item.studentId, 'A')}
                    >
                        <Icon
                            name={attendance.status === 'A' ? 'close-circle' : 'circle-outline'}
                            size={20}
                            color={attendance.status === 'A' ? theme.error : theme.textSecondary}
                        />
                        <Text style={[
                            styles.statusText,
                            { color: attendance.status === 'A' ? theme.error : theme.textSecondary }
                        ]}>
                            A
                        </Text>
                    </TouchableOpacity>

                    {/* Leave Button */}
                    <TouchableOpacity
                        style={[
                            styles.statusButton,
                            attendance.status === 'L' && { backgroundColor: theme.warning + '20' },
                            { borderColor: attendance.status === 'L' ? theme.warning : theme.border }
                        ]}
                        onPress={() => handleStatusChange(item.studentId, 'L')}
                    >
                        <Icon
                            name={attendance.status === 'L' ? 'calendar-clock' : 'calendar'}
                            size={20}
                            color={attendance.status === 'L' ? theme.warning : theme.textSecondary}
                        />
                        <Text style={[
                            styles.statusText,
                            { color: attendance.status === 'L' ? theme.warning : theme.textSecondary }
                        ]}>
                            L
                        </Text>
                    </TouchableOpacity>

                    {/* Remarks Button */}
                    <TouchableOpacity
                        style={[styles.remarksButton, { borderColor: theme.border }]}
                        onPress={() => {
                            setSelectedStudent(item.studentId);
                            setShowRemarksModal(true);
                        }}
                    >

                    </TouchableOpacity>
                </View>
            </View>
        );
    };

    if (loading) {

            </View>
        );
    }

    return (
        <View style={[styles.container, { backgroundColor: theme.background }]}>
            {/* Header */}

// TeacherDiaryListScreen.js - OPTIMIZED with Today's Diaries by Default
import React, { useState, useCallback, useMemo, useEffect } from 'react';
import {
    View,
    Text,
    StyleSheet,
    TouchableOpacity,
    FlatList,
    ActivityIndicator,
    RefreshControl,
    Platform,
} from 'react-native';
import { useSelector } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialIcons';
import { selectThemeData } from '../../store/slices/themeSlice';
import { useGetTeacherDiariesQuery, useGetTeacherClassesQuery } from '../../store/diaryApi';
import DiaryFilterBar from './components/DiaryFilterBar';
import DiaryCard from './components/DiaryCard';
import { format, startOfToday, endOfToday } from 'date-fns';

const TeacherDiaryListScreen = ({ navigation }) => {
    const theme = useSelector(selectThemeData);
    const [refreshing, setRefreshing] = useState(false);
    
    // Get today's date range
    const todayStart = useMemo(() => format(startOfToday(), 'yyyy-MM-dd'), []);
    const todayEnd = useMemo(() => format(endOfToday(), 'yyyy-MM-dd'), []);
    
    // Initialize filters with today's date range
    const [filters, setFilters] = useState({
        classId: null,
        sectionIds: [],
        subjectId: null,
        dateFrom: todayStart,
        dateTo: todayEnd,
    });
    const [appliedFilters, setAppliedFilters] = useState(filters);

    const { data: classesData, isLoading: classesLoading } = useGetTeacherClassesQuery();
    const {
        data: diariesData,
        isLoading,
        isFetching,
        error,
        refetch,
    } = useGetTeacherDiariesQuery(appliedFilters);

    // Reset to today's date when component mounts
    useEffect(() => {
        const todayFilters = {
            classId: null,
            sectionIds: [],
            subjectId: null,
            dateFrom: todayStart,
            dateTo: todayEnd,
        };
        setFilters(todayFilters);
        setAppliedFilters(todayFilters);
    }, [todayStart, todayEnd]);

    const onRefresh = useCallback(async () => {
        setRefreshing(true);
        await refetch();
        setRefreshing(false);
    }, [refetch]);

    const handleApplyFilters = useCallback((newFilters) => {
        setAppliedFilters(newFilters);
    }, []);

    const handleResetFilters = useCallback(() => {
        const resetFilters = {
            classId: null,
            sectionIds: [],
            subjectId: null,
            dateFrom: todayStart,
            dateTo: todayEnd,
        };
        setFilters(resetFilters);
        setAppliedFilters(resetFilters);
    }, [todayStart, todayEnd]);

    const handleCreateDiary = useCallback(() => {
        navigation.navigate('CreateDiary');
    }, [navigation]);

    const handleEditDiary = useCallback((entry) => {
        const editData = {
            id: entry.id,
            classId: entry.classId,
            className: entry.className,
            sectionIds: Array.isArray(entry.sectionIds) ? entry.sectionIds : [],
            sectionNames: Array.isArray(entry.sectionNames) ? entry.sectionNames : [],
            subjectId: entry.subjectId,
            subjectName: entry.subjectName,
            content: entry.content || '',
            isPublished: entry.isPublished,
            date: entry.date
        };
        
        navigation.navigate('CreateDiary', { 
            editEntry: editData
        });
    }, [navigation]);

    // Memoized strip HTML function
    const stripHtml = useCallback((html) => {
        if (!html) return '';
        return html.replace(/<[^>]*>/g, '').substring(0, 100);
    }, []);

    // Memoize transformed entries
    const transformedEntries = useMemo(() => {
        const entries = diariesData?.data?.entries || [];
        return entries.map(entry => ({
            id: entry.id,
            date: entry.date || '',
            classId: entry.classId,
            className: entry.className || '',
            sectionIds: entry.sectionIds || [],
            sectionNames: entry.sectionNames || [],
            subjectId: entry.subjectId,
            subjectName: entry.subjectName || '',
            content: entry.content || '',
            isPublished: entry.isPublished || false,
            previewContent: stripHtml(entry.content),
        }));
    }, [diariesData?.data?.entries, stripHtml]);

    // Memoize empty state component
    const renderEmptyState = useCallback(() => (
        <View style={[styles.emptyContainer, { backgroundColor: theme.card }]}>
            <Icon name="article" size={60} color={theme.textSecondary} />
            <Text style={[styles.emptyTitle, { color: theme.text }]}>
                No Diary Entries
            </Text>
            <Text style={[styles.emptySubtitle, { color: theme.textSecondary }]}>
                No diaries found for today. Create your first diary entry by tapping the + button
            </Text>
            <TouchableOpacity
                style={[styles.emptyButton, { backgroundColor: theme.primary }]}
                onPress={handleCreateDiary}
            >
                <Icon name="add" size={20} color="#FFF" />
                <Text style={styles.emptyButtonText}>Create Diary</Text>
            </TouchableOpacity>
        </View>
    ), [theme, handleCreateDiary]);

    // Memoize render item
    const renderItem = useCallback(({ item }) => (
        <DiaryCard 
            entry={item} 
            theme={theme} 
            onPress={() => handleEditDiary(item)} 
        />
    ), [theme, handleEditDiary]);

    // Memoize key extractor
    const keyExtractor = useCallback((item) => item.id?.toString() || Math.random().toString(), []);

    const totalCount = diariesData?.data?.count || 0;
    const showClearButton = appliedFilters.classId || 
                           appliedFilters.sectionIds?.length > 0 || 
                           appliedFilters.subjectId || 
                           (appliedFilters.dateFrom !== todayStart || appliedFilters.dateTo !== todayEnd);

    if (isLoading && !refreshing) {
        return (
            <View style={[styles.loadingContainer, { backgroundColor: theme.background }]}>
                <ActivityIndicator size="large" color={theme.primary} />
                <Text style={[styles.loadingText, { color: theme.textSecondary }]}>
                    Loading today's diaries...
                </Text>
            </View>
        );
    }

    return (
        <View style={[styles.container, { backgroundColor: theme.background }]}>
            {/* Header */}
            <View style={[styles.header, { backgroundColor: theme.card }]}>
                <TouchableOpacity onPress={() => navigation.goBack()} hitSlop={{ top: 10, bottom: 10, left: 10, right: 10 }}>
                    <Icon name="arrow-back" size={24} color={theme.text} />
                </TouchableOpacity>
                <Text style={[styles.headerTitle, { color: theme.text }]}>Today's Diaries</Text>
                <View style={{ width: 40 }} />
            </View>

            {/* Filter Bar */}
            <DiaryFilterBar
                filters={filters}
                setFilters={setFilters}
                onApply={handleApplyFilters}
                onReset={handleResetFilters}
                classesData={classesData}
            />

            {/* Results Count */}
            {!isLoading && (
                <View style={[styles.resultsBar, { backgroundColor: theme.card }]}>
                    <Text style={[styles.resultsText, { color: theme.textSecondary }]}>
                        {totalCount} {totalCount === 1 ? 'entry' : 'entries'} for today
                    </Text>
                    {showClearButton && (
                        <TouchableOpacity onPress={handleResetFilters} hitSlop={{ top: 10, bottom: 10, left: 10, right: 10 }}>
                            <Text style={[styles.clearText, { color: theme.primary }]}>Show Today</Text>
                        </TouchableOpacity>
                    )}
                </View>
            )}

            {/* Diary List */}
            <FlatList
                data={transformedEntries}
                renderItem={renderItem}
                keyExtractor={keyExtractor}
                contentContainerStyle={styles.listContent}
                refreshControl={
                    <RefreshControl 
                        refreshing={refreshing} 
                        onRefresh={onRefresh} 
                        colors={[theme.primary]} 
                        tintColor={theme.primary}
                    />
                }
                ListEmptyComponent={renderEmptyState}
                showsVerticalScrollIndicator={false}
                initialNumToRender={10}
                maxToRenderPerBatch={10}
                windowSize={10}
                removeClippedSubviews={Platform.OS === 'android'}
                updateCellsBatchingPeriod={50}
            />

            {/* FAB Button */}
            <TouchableOpacity
                style={[styles.fab, { backgroundColor: theme.primary }]}
                onPress={handleCreateDiary}
                activeOpacity={0.8}
                hitSlop={{ top: 10, bottom: 10, left: 10, right: 10 }}
            >
                <Icon name="add" size={28} color="#FFF" />
            </TouchableOpacity>
        </View>
    );
};
