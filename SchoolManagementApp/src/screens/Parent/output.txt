import React, { useState, useEffect } from 'react';
import {
    View,
    Text,
    StyleSheet,
    ScrollView,
    TouchableOpacity,
    ActivityIndicator,
    RefreshControl,
    Dimensions,
    Modal,
    FlatList,
    Platform,
    Alert,
} from 'react-native';
import { useSelector } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialCommunityIcons';
import { selectThemeData } from '../../store/slices/themeSlice';
import { useGetChildAttendanceHistoryQuery } from '../../store/attendanceApi';
import DateTimePicker from '@react-native-community/datetimepicker';

const { width } = Dimensions.get('window');

const ChildAttendanceHistoryScreen = ({ route, navigation }) => {
    const theme = useSelector(selectThemeData);
    const { childId, childName } = route.params;

    const [refreshing, setRefreshing] = useState(false);
    const [attendanceHistory, setAttendanceHistory] = useState([]);
    const [filteredHistory, setFilteredHistory] = useState([]);
    const [overallStats, setOverallStats] = useState({
        totalDays: 0,
        presentDays: 0,
        absentDays: 0,
        lateDays: 0,
        leaveDays: 0,
        presentPercentage: 0,
        currentStreak: 0,
        longestStreak: 0,
        bestMonth: 'N/A',
        worstMonth: 'N/A'
    });
    const [showFilterModal, setShowFilterModal] = useState(false);
    const [selectedMonth, setSelectedMonth] = useState(new Date().getMonth());
    const [selectedYear, setSelectedYear] = useState(new Date().getFullYear());
    const [selectedStatus, setSelectedStatus] = useState('all');
    const [dateFrom, setDateFrom] = useState(null);
    const [dateTo, setDateTo] = useState(null);
    const [showDateFromPicker, setShowDateFromPicker] = useState(false);
    const [showDateToPicker, setShowDateToPicker] = useState(false);

    const months = [
        'January', 'February', 'March', 'April', 'May', 'June',
        'July', 'August', 'September', 'October', 'November', 'December'
    ];

    const years = Array.from({ length: 5 }, (_, i) => new Date().getFullYear() - i);
    const statuses = [
        { id: 'all', label: 'All Status' },
        { id: 'present', label: 'Present' },
        { id: 'absent', label: 'Absent' },
        { id: 'late', label: 'Late' },
        { id: 'leave', label: 'Leave' },
    ];

    // API Query
    const {
        data: attendanceResponse,
        isLoading,
        isFetching,
        error,
        refetch
    } = useGetChildAttendanceHistoryQuery({
        childId,
        filters: {
            month: selectedMonth + 1,
            year: selectedYear
        }
    });

    // Debug logging
    useEffect(() => {
        console.log('🔍 API State:', {
            isLoading,
            isFetching,
            error: error ? 'Error exists' : 'No error',
            hasResponse: !!attendanceResponse,
            responseSuccess: attendanceResponse?.success
        });
    }, [isLoading, isFetching, error, attendanceResponse]);

    // Process API response
    useEffect(() => {
        console.log('📊 Processing API response...');

        if (attendanceResponse) {
            console.log('📊 API Response:', attendanceResponse);

            if (attendanceResponse.success === true) {
                const responseData = attendanceResponse.data;

                // Extract history from different possible structures
                let history = [];
                if (responseData && responseData.history && Array.isArray(responseData.history)) {
                    history = responseData.history;
                } else if (Array.isArray(responseData)) {
                    history = responseData;
                } else if (responseData && responseData.data && Array.isArray(responseData.data)) {
                    history = responseData.data;
                }

                console.log('📊 Extracted history:', history.length, 'records');
                processAttendanceData(history);
            } else if (attendanceResponse.success === false) {
                console.error('❌ API returned error:', attendanceResponse.error);
                Alert.alert('Error', attendanceResponse.error || 'Failed to load attendance data');
                resetData();
            } else {
                // If response doesn't have success property, try to process it directly
                console.log('⚠️ Response format different, trying to process directly');
                processAttendanceData(attendanceResponse);
            }
        } else if (error) {
            console.error('❌ RTK Query error:', error);
            Alert.alert('Error', 'Network error occurred. Please check your connection.');
            resetData();
        }
    }, [attendanceResponse, error]);

    // Apply filters when data or filters change
    useEffect(() => {
        applyFilters();
    }, [attendanceHistory, selectedStatus, dateFrom, dateTo]);

    const resetData = () => {
        setAttendanceHistory([]);
        setFilteredHistory([]);
        setOverallStats({
            totalDays: 0,
            presentDays: 0,
            absentDays: 0,
            lateDays: 0,
            leaveDays: 0,
            presentPercentage: 0,
            currentStreak: 0,
            longestStreak: 0,
            bestMonth: 'N/A',
            worstMonth: 'N/A'
        });
    };

    const processAttendanceData = (rawData) => {
        try {
            console.log('📈 Processing raw data:', rawData);

            // Handle different data structures
            let history = [];
            if (Array.isArray(rawData)) {
                history = rawData;
            } else if (rawData && typeof rawData === 'object') {
                // Try common data structures
                if (Array.isArray(rawData.history)) {
                    history = rawData.history;
                } else if (Array.isArray(rawData.data)) {
                    history = rawData.data;
                } else if (Array.isArray(rawData.records)) {
                    history = rawData.records;
                } else if (rawData.History && Array.isArray(rawData.History)) {
                    history = rawData.History;
                }
            }

            console.log('📈 Processed history:', history.length, 'records');

            // Validate and format data
            const validatedHistory = history.map(record => ({
                ...record,
                date: record.date || '',
                status: record.status || 'unknown',
                lateStatus: record.lateStatus || '',
                remarks: record.remarks || ''
            }));

            setAttendanceHistory(validatedHistory);
            setFilteredHistory(validatedHistory);

            // Calculate statistics
            if (validatedHistory.length > 0) {
                const totalDays = validatedHistory.length;
                const presentDays = validatedHistory.filter(day => {
                    const status = (day.status || '').toLowerCase();
                    return status === 'present' || status === 'p';
                }).length;

                const absentDays = validatedHistory.filter(day => {
                    const status = (day.status || '').toLowerCase();
                    return status === 'absent' || status === 'a';
                }).length;

                const lateDays = validatedHistory.filter(day => {
                    const status = (day.status || '').toLowerCase();
                    const lateStatus = (day.lateStatus || '').toLowerCase();
                    return status === 'late' || status === 'l' || lateStatus === 'late';
                }).length;

                const leaveDays = validatedHistory.filter(day => {
                    const status = (day.status || '').toLowerCase();
                    return status === 'leave';
                }).length;

                const presentPercentage = totalDays > 0 ? Math.round((presentDays / totalDays) * 100) : 0;

                setOverallStats({
                    totalDays,
                    presentDays,
                    absentDays,
                    lateDays,
                    leaveDays,
                    presentPercentage,
                    currentStreak: calculateCurrentStreak(validatedHistory),
                    longestStreak: calculateLongestStreak(validatedHistory),
                    bestMonth: getBestMonth(validatedHistory),
                    worstMonth: getWorstMonth(validatedHistory)
                });

                console.log('📊 Calculated stats:', {
                    totalDays,
                    presentDays,
                    absentDays,
                    presentPercentage
                });
            } else {
                console.log('📊 No attendance records found');
                resetData();
            }
        } catch (error) {
            console.error('❌ Error processing data:', error);
            Alert.alert('Error', 'Failed to process attendance data. Please try again.');
            resetData();
        }
    };

    const applyFilters = () => {
        let filtered = [...attendanceHistory];

        // Filter by status
        if (selectedStatus !== 'all') {
            filtered = filtered.filter(day => {
                const status = (day.status || '').toLowerCase();
                return status === selectedStatus;
            });
        }

        // Filter by date range
        if (dateFrom) {
            const fromDate = new Date(dateFrom);
            fromDate.setHours(0, 0, 0, 0);
            filtered = filtered.filter(day => {
                if (!day.date) return false;
                try {
                    const dayDate = new Date(day.date);
                    if (isNaN(dayDate.getTime())) return false;
                    dayDate.setHours(0, 0, 0, 0);
                    return dayDate >= fromDate;
                } catch {
                    return false;
                }
            });
        }

        if (dateTo) {
            const toDate = new Date(dateTo);
            toDate.setHours(23, 59, 59, 999);
            filtered = filtered.filter(day => {
                if (!day.date) return false;
                try {
                    const dayDate = new Date(day.date);
                    if (isNaN(dayDate.getTime())) return false;
                    dayDate.setHours(0, 0, 0, 0);
                    return dayDate <= toDate;
                } catch {
                    return false;
                }
            });
        }

        console.log('🔍 Filters applied:', {
            original: attendanceHistory.length,
            filtered: filtered.length,
            selectedStatus
        });

        setFilteredHistory(filtered);
    };

    const getBestMonth = (history) => {
        try {
            const monthStats = {};
            history.forEach(day => {
                if (day.date) {
                    try {
                        const date = new Date(day.date);
                        if (!isNaN(date.getTime())) {
                            const monthYear = `${date.getMonth()}-${date.getFullYear()}`;
                            if (!monthStats[monthYear]) {
                                monthStats[monthYear] = { present: 0, total: 0 };
                            }
                            monthStats[monthYear].total++;

                            const status = (day.status || '').toLowerCase();
                            if (status === 'present' || status === 'p') {
                                monthStats[monthYear].present++;
                            }
                        }
                    } catch (error) {
                        console.warn('Invalid date:', day.date);
                    }
                }
            });

            let bestMonth = 'N/A';
            let bestPercentage = 0;

            Object.entries(monthStats).forEach(([monthYear, stats]) => {
                const percentage = stats.total > 0 ? (stats.present / stats.total) * 100 : 0;
                if (percentage > bestPercentage) {
                    bestPercentage = percentage;
                    const [month, year] = monthYear.split('-');
                    bestMonth = `${months[parseInt(month)]} ${year}`;
                }
            });

            return bestMonth;
        } catch (error) {
            console.error('Error calculating best month:', error);
            return 'N/A';
        }
    };

    const getWorstMonth = (history) => {
        try {
            const monthStats = {};
            history.forEach(day => {
                if (day.date) {
                    try {
                        const date = new Date(day.date);
                        if (!isNaN(date.getTime())) {
                            const monthYear = `${date.getMonth()}-${date.getFullYear()}`;
                            if (!monthStats[monthYear]) {
                                monthStats[monthYear] = { present: 0, total: 0 };
                            }
                            monthStats[monthYear].total++;

                            const status = (day.status || '').toLowerCase();
                            if (status === 'present' || status === 'p') {
                                monthStats[monthYear].present++;
                            }
                        }
                    } catch (error) {
                        console.warn('Invalid date:', day.date);
                    }
                }
            });

            let worstMonth = 'N/A';
            let worstPercentage = 100;

            Object.entries(monthStats).forEach(([monthYear, stats]) => {
                const percentage = stats.total > 0 ? (stats.present / stats.total) * 100 : 0;
                if (percentage < worstPercentage) {
                    worstPercentage = percentage;
                    const [month, year] = monthYear.split('-');
                    worstMonth = `${months[parseInt(month)]} ${year}`;
                }
            });

            return worstMonth;
        } catch (error) {
            console.error('Error calculating worst month:', error);
            return 'N/A';
        }
    };

    const calculateCurrentStreak = (history) => {
        try {
            let streak = 0;
            const today = new Date();
            today.setHours(23, 59, 59, 999);

            // Get valid dates sorted in descending order
            const sortedHistory = [...history]
                .filter(day => day.date)
                .map(day => {
                    try {
                        const parsedDate = new Date(day.date);
                        return isNaN(parsedDate.getTime()) ? null : { ...day, parsedDate };
                    } catch {
                        return null;
                    }
                })
                .filter(day => day && day.parsedDate <= today)
                .sort((a, b) => b.parsedDate - a.parsedDate);

            for (const day of sortedHistory) {
                const status = (day.status || '').toLowerCase();
                if (status === 'present' || status === 'p') {
                    streak++;
                } else {
                    break;
                }
            }

            return streak;
        } catch (error) {
            console.error('Error calculating current streak:', error);
            return 0;
        }
    };

    const calculateLongestStreak = (history) => {
        try {
            let longest = 0;
            let current = 0;

            // Get valid dates sorted in ascending order
            const sortedHistory = [...history]
                .filter(day => day.date)
                .map(day => {
                    try {
                        const parsedDate = new Date(day.date);
                        return isNaN(parsedDate.getTime()) ? null : { ...day, parsedDate };
                    } catch {
                        return null;
                    }
                })
                .filter(day => day !== null)
                .sort((a, b) => a.parsedDate - b.parsedDate);

            for (const day of sortedHistory) {
                const status = (day.status || '').toLowerCase();
                if (status === 'present' || status === 'p') {
                    current++;
                    longest = Math.max(longest, current);
                } else {
                    current = 0;
                }
            }

            return longest;
        } catch (error) {
            console.error('Error calculating longest streak:', error);
            return 0;
        }
    };

    const onRefresh = async () => {
        setRefreshing(true);
        try {
            console.log('🔄 Refreshing data...');
            await refetch();
        } catch (error) {
            console.error('Refresh error:', error);
        }
        setRefreshing(false);
    };

    const getStatusColor = (status) => {
        const statusLower = (status || '').toLowerCase();
        switch (statusLower) {
            case 'present':
            case 'p': return theme.success || '#4CAF50';
            case 'absent':
            case 'a': return theme.error || '#F44336';
            case 'late':
            case 'l': return theme.warning || '#FF9800';
            case 'leave': return theme.info || '#2196F3';
            default: return theme.textSecondary || '#9E9E9E';
        }
    };

    const getStatusIcon = (status) => {
        const statusLower = (status || '').toLowerCase();
        switch (statusLower) {
            case 'present':
            case 'p': return 'check-circle';
            case 'absent':
            case 'a': return 'close-circle';
            case 'late':
            case 'l': return 'clock-alert';
            case 'leave': return 'account-arrow-right';
            default: return 'help-circle';
        }
    };

    const getStatusText = (status) => {
        const statusLower = (status || '').toLowerCase();
        switch (statusLower) {
            case 'present':
            case 'p': return 'Present';
            case 'absent':
            case 'a': return 'Absent';
            case 'late':
            case 'l': return 'Late';
            case 'leave': return 'Leave';
            default: return 'Unknown';
        }
    };

    const handleDateFromChange = (event, selectedDate) => {
        setShowDateFromPicker(Platform.OS === 'ios');
        if (selectedDate) {
            setDateFrom(selectedDate);
        }
    };

    const handleDateToChange = (event, selectedDate) => {
        setShowDateToPicker(Platform.OS === 'ios');
        if (selectedDate) {
            setDateTo(selectedDate);
        }
    };

    const handleResetFilters = () => {
        setSelectedStatus('all');
        setDateFrom(null);
        setDateTo(null);
        setSelectedMonth(new Date().getMonth());
        setSelectedYear(new Date().getFullYear());
    };

    const handleApplyFilters = () => {
        applyFilters();
        setShowFilterModal(false);
    };

    const formatDate = (dateString) => {
        try {
            if (!dateString) return 'No date';
            const date = new Date(dateString);
            if (isNaN(date.getTime())) return 'Invalid Date';

            return date.toLocaleDateString('en-US', {
                weekday: 'short',
                month: 'short',
                day: 'numeric',
                year: 'numeric'
            });
        } catch (error) {
            return 'Invalid Date';
        }
    };

    const getDayName = (dateString) => {
        try {
            if (!dateString) return '';
            const date = new Date(dateString);
            if (isNaN(date.getTime())) return '';

            return date.toLocaleDateString('en-US', { weekday: 'long' });
        } catch (error) {
            return '';
        }
    };

    const renderFilterModal = () => (
        <Modal
            visible={showFilterModal}
            animationType="slide"
            transparent={true}
            onRequestClose={() => setShowFilterModal(false)}
        >
            <View style={styles.modalOverlay}>
                <View style={[styles.modalContent, { backgroundColor: theme.background || '#FFFFFF' }]}>
                    <View style={styles.modalHeader}>
                        <Text style={[styles.modalTitle, { color: theme.text || '#000000' }]}>
                            Filter Attendance
                        </Text>
                        <TouchableOpacity onPress={() => setShowFilterModal(false)}>
                            <Icon name="close" size={24} color={theme.text || '#000000'} />
                        </TouchableOpacity>
                    </View>

                    <ScrollView style={styles.filterContent} showsVerticalScrollIndicator={false}>
                        {/* Month Selection */}
                        <View style={[styles.filterSection, { backgroundColor: theme.card || '#F5F5F5' }]}>
                            <Text style={[styles.filterSectionTitle, { color: theme.text || '#000000' }]}>
                                Select Month
                            </Text>
                            <ScrollView horizontal showsHorizontalScrollIndicator={false}>
                                {months.map((month, index) => (
                                    <TouchableOpacity
                                        key={index}
                                        style={[
                                            styles.monthButton,
                                            {
                                                backgroundColor: selectedMonth === index
                                                    ? (theme.primary || '#3F51B5')
                                                    : (theme.card || '#F5F5F5'),
                                                borderColor: selectedMonth === index
                                                    ? (theme.primary || '#3F51B5')
                                                    : (theme.border || '#E0E0E0')
                                            }
                                        ]}
                                        onPress={() => setSelectedMonth(index)}
                                    >
                                        <Text style={[
                                            styles.monthButtonText,
                                            {
                                                color: selectedMonth === index
                                                    ? '#FFFFFF'
                                                    : (theme.text || '#000000')
                                            }
                                        ]}>
                                            {month.substring(0, 3)}
                                        </Text>
                                    </TouchableOpacity>
                                ))}
                            </ScrollView>
                        </View>

                        {/* Year Selection */}
                        <View style={[styles.filterSection, { backgroundColor: theme.card || '#F5F5F5' }]}>
                            <Text style={[styles.filterSectionTitle, { color: theme.text || '#000000' }]}>
                                Select Year
                            </Text>
                            <ScrollView horizontal showsHorizontalScrollIndicator={false}>
                                {years.map((year) => (
                                    <TouchableOpacity
                                        key={year}
                                        style={[
                                            styles.yearButton,
                                            {
                                                backgroundColor: selectedYear === year
                                                    ? (theme.primary || '#3F51B5')
                                                    : (theme.card || '#F5F5F5'),
                                                borderColor: selectedYear === year
                                                    ? (theme.primary || '#3F51B5')
                                                    : (theme.border || '#E0E0E0')
                                            }
                                        ]}
                                        onPress={() => setSelectedYear(year)}
                                    >
                                        <Text style={[
                                            styles.yearButtonText,
                                            {
                                                color: selectedYear === year
                                                    ? '#FFFFFF'
                                                    : (theme.text || '#000000')
                                            }
                                        ]}>
                                            {year}
                                        </Text>
                                    </TouchableOpacity>
                                ))}
                            </ScrollView>
                        </View>

                        {/* Status Filter */}
                        <View style={[styles.filterSection, { backgroundColor: theme.card || '#F5F5F5' }]}>
                            <Text style={[styles.filterSectionTitle, { color: theme.text || '#000000' }]}>
                                Status

                                {statuses.map((status) => (

                                        style={[
                                            styles.statusButton,
                                            {
                                                backgroundColor: selectedStatus === status.id
                                                    ? (theme.primary || '#3F51B5') + '20'
                                                    : (theme.background || '#FFFFFF'),
                                                borderColor: selectedStatus === status.id
                                                    ? (theme.primary || '#3F51B5')
                                                    : (theme.border || '#E0E0E0')
                                            }
                                        ]}
                                        onPress={() => setSelectedStatus(status.id)}
                                    >
                                        <Icon
                                            name={getStatusIcon(status.id)}
                                            size={18}
                                            color={getStatusColor(status.id === 'all' ? 'present' : status.id)}
                                        />
                                        <Text style={[
                                            styles.statusButtonText,
                                            {
                                                color: selectedStatus === status.id
                                                    ? (theme.primary || '#3F51B5')
                                                    : (theme.text || '#000000')
                                            }
                                        ]}>
                                            {status.label}
                                        </Text>
                                    </TouchableOpacity>
                                ))}
                            </View>
                        </View>

                        {/* Date Range */}
                        <View style={[styles.filterSection, { backgroundColor: theme.card || '#F5F5F5' }]}>
                            <Text style={[styles.filterSectionTitle, { color: theme.text || '#000000' }]}>
                                Date Range
                            </Text>
                            <View style={styles.dateRangeContainer}>
                                <View style={styles.dateInput}>
                                    <Text style={[styles.dateLabel, { color: theme.textSecondary || '#757575' }]}>
                                        From Date
                                    </Text>
                                    <TouchableOpacity
                                        style={[styles.dateButton, {
                                            backgroundColor: theme.background || '#FFFFFF',
                                            borderColor: theme.border || '#E0E0E0'
                                        }]}
                                        onPress={() => setShowDateFromPicker(true)}
                                    >
                                        <Icon name="calendar" size={20} color={theme.text || '#000000'} />
                                        <Text style={[styles.dateButtonText, { color: theme.text || '#000000' }]}>
                                            {dateFrom ? dateFrom.toLocaleDateString() : 'Select date'}
                                        </Text>
                                    </TouchableOpacity>
                                </View>

                                <View style={styles.dateInput}>
                                    <Text style={[styles.dateLabel, { color: theme.textSecondary || '#757575' }]}>
                                        To Date
                                    </Text>
                                    <TouchableOpacity
                                        style={[styles.dateButton, {
                                            backgroundColor: theme.background || '#FFFFFF',
                                            borderColor: theme.border || '#E0E0E0'
                                        }]}
                                        onPress={() => setShowDateToPicker(true)}
                                    >
                                        <Icon name="calendar" size={20} color={theme.text || '#000000'} />
                                        <Text style={[styles.dateButtonText, { color: theme.text || '#000000' }]}>
                                            {dateTo ? dateTo.toLocaleDateString() : 'Select date'}

                            style={[styles.filterButton, { backgroundColor: theme.primary || '#3F51B5' }]}
                            onPress={handleApplyFilters}
                        >
                            <Icon name="check" size={20} color="#FFF" />
                            <Text style={[styles.filterButtonText, { color: '#FFFFFF' }]}>
                                Apply Filters
                            </Text>
                        </TouchableOpacity>
                    </View>
                </View>
            </View>
        </Modal>
    );

    const renderAttendanceItem = ({ item, index }) => {
        const statusColor = getStatusColor(item.status);
        const statusText = getStatusText(item.status);
        const statusIcon = getStatusIcon(item.status);

        return (
            <View style={[styles.attendanceItem, { backgroundColor: theme.card || '#F5F5F5' }]}>
                <View style={styles.attendanceDate}>
                    <Text style={[styles.dateText, { color: theme.text || '#000000' }]}>
                        {formatDate(item.date)}

const getGradeColor = (grade, theme) => {
    switch (grade) {
        case 'A+':
        case 'A':
            return theme.success;
        case 'B+':
        case 'B':
            return theme.warning;
        case 'C+':
        case 'C':
            return theme.secondary;
        case 'D':
        case 'F':
            return theme.error;
        default:
            return theme.textSecondary;
    }
};

                                    styles.subjectChip,
                                    { backgroundColor: theme.card },
                                    selectedSubject === subject.name && {
                                        backgroundColor: subject.colorCode,
                                    },
                                ]}
                                onPress={() => handleSubjectSelect(subject)}
                            >
                                <Text
                                    style={[
                                        styles.subjectChipText,
                                        { color: theme.textSecondary },
                                        selectedSubject === subject.name && { color: '#FFFFFF' },
                                    ]}
                                >
                                    {subject.name}
                                </Text>
                            </TouchableOpacity>
                        ))}
                    </ScrollView>
                )}

                {/* Stats Summary */}
                {hasGrades && (
                    <View style={[styles.statsCard, { backgroundColor: theme.card }]}>
                        <View style={styles.statsGrid}>
                            <View style={styles.statItem}>
                                <Text style={[styles.statValue, { color: theme.text }]}>
                                    {statsData?.averagePercentage || 0}%
                                </Text>
                                <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                                    Average
                                </Text>
                            </View>
                            <View style={styles.statItem}>
                                <Text style={[styles.statValue, { color: theme.text }]}>
                                    {statsData?.totalExams || 0}
                                </Text>
                                <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                                    Total Exams
                                </Text>
                            </View>
                            <View style={styles.statItem}>
                                <Text style={[styles.statValue, { color: theme.text }]}>
                                    {statsData?.totalSubjects || 0}
                                </Text>
                                <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                                    Subjects
                                </Text>
                            </View>
                            <View style={styles.statItem}>
                                <Text style={[styles.statValue, { color: theme.text }]}>
                                    {statsData?.passRate || 0}%
                                </Text>
                                <Text style={[styles.statLabel, { color: theme.textSecondary }]}>
                                    Pass Rate
                                </Text>
                            </View>
                        </View>

                        {statsData?.improvement && (
                            <View style={styles.improvementContainer}>
                                <Icon
                                    name={statsData.improvement.startsWith('+') ? 'trending-up' : 'trending-down'}
                                    size={16}
                                    color={statsData.improvement.startsWith('+') ? '#27ae60' : '#e74c3c'}
                                />
                                <Text style={[
                                    styles.improvementText,
                                    { color: statsData.improvement.startsWith('+') ? '#27ae60' : '#e74c3c' }
                                ]}>
                                    {statsData.improvement} overall improvement
                                </Text>
                            </View>
                        )}
                    </View>
                )}

                {/* Subject Performance */}
                {subjectPerformance.length > 0 && (
                    <View style={[styles.section, { backgroundColor: theme.card }]}>
                        <Text style={[styles.sectionTitle, { color: theme.text }]}>
                            Subject Performance
                        </Text>
                        {subjectPerformance.map((item, index) => (
                            <View key={index} style={styles.subjectProgressItem}>
                                <View style={styles.subjectProgressHeader}>
                                    <Text style={[styles.subjectProgressName, { color: theme.text }]}>
                                        {item.subject}
                                    </Text>
                                    <Text style={[styles.subjectProgressPercentage, { color: theme.text }]}>
                                        {item.percentage}%
                                    </Text>
                                </View>
                                <View style={[styles.subjectProgressBar, { backgroundColor: theme.border }]}>
                                    <View
                                        style={[
                                            styles.subjectProgressFill,
                                            {
                                                width: `${item.percentage}%`,
                                                backgroundColor: item.color
                                            }
                                        ]}
                                    />
                                </View>
                            </View>
                        ))}
                    </View>
                )}

                {/* Grades List */}
                <View style={styles.gradesSection}>
                    <View style={styles.gradesHeader}>
                        <Text style={[styles.gradesTitle, { color: theme.text }]}>
                            Grade History
                        </Text>
                        {hasGrades && (
                            <Text style={[styles.gradesCount, { color: theme.textSecondary }]}>
                                {gradesData.length} {gradesData.length === 1 ? 'exam' : 'exams'}
                            </Text>
                        )}
                    </View>

                    {hasGrades ? (
                        <FlatList
                            data={gradesData}
                            renderItem={renderGradeItem}
                            keyExtractor={(item) => item.id?.toString() || Math.random().toString()}
                            scrollEnabled={false}
                            contentContainerStyle={styles.gradesList}
                        />
                    ) : (
                        renderEmptyState()
                    )}
                </View>

                {/* Actions */}
                {hasGrades && (
                    <View style={styles.actionsContainer}>
                        <TouchableOpacity
                            style={[styles.actionButton, { backgroundColor: theme.primary }]}
                            onPress={handleViewReport}
                        >
                            <Icon name="file-document-outline" size={20} color="#FFFFFF" />
                            <Text style={styles.actionButtonText}>View Full Report</Text>
                        </TouchableOpacity>

                        <TouchableOpacity
                            style={[
                                styles.actionButton,
                                styles.secondaryButton,
                                { backgroundColor: theme.card, borderColor: theme.border }
                            ]}
                            onPress={handleDownloadReport}
                            disabled={downloading}
                        >
                            {downloading ? (
                                <ActivityIndicator size="small" color={theme.text} />
                            ) : (
                                <Icon name="download" size={20} color={theme.text} />
                            )}
                            <Text style={[styles.actionButtonText, { color: theme.text }]}>
                                {downloading ? 'Downloading...' : 'Download Report'}
                            </Text>
                        </TouchableOpacity>
                    </View>
                )}
            </ScrollView>

            {/* Filters Modal */}
            <Modal
                visible={showFilters}
                animationType="slide"
                transparent={true}
                onRequestClose={() => setShowFilters(false)}
            >
                <View style={[styles.modalOverlay, { backgroundColor: 'rgba(0,0,0,0.5)' }]}>
                    <View style={[styles.modalContent, { backgroundColor: theme.background }]}>
                        <View style={styles.modalHeader}>
                            <Text style={[styles.modalTitle, { color: theme.text }]}>
                                Filter Grades
                            </Text>
                            <TouchableOpacity onPress={() => setShowFilters(false)}>
                                <Icon name="close" size={24} color={theme.text} />
                            </TouchableOpacity>
                        </View>

                        <ScrollView style={styles.filterContainer}>
                            {/* Subject Filter */}
                            <View style={styles.filterGroup}>
                                <Text style={[styles.filterLabel, { color: theme.text }]}>
                                    Subject
                                </Text>
                                <View style={styles.filterOptions}>
                                    {subjectsList.map(subject => (
                                        <TouchableOpacity
                                            key={subject.id || 'all'}
                                            style={[
                                                styles.filterOptionChip,
                                                { backgroundColor: theme.card },
                                                filters.subjectId === subject.id && {
                                                    backgroundColor: subject.colorCode,
                                                }
                                            ]}
                                            onPress={() => applyFilters({ subjectId: subject.id })}
                                        >
                                            <Text style={[
                                                styles.filterOptionText,
                                                { color: theme.textSecondary },
                                                filters.subjectId === subject.id && { color: '#FFFFFF' }
                                            ]}>
                                                {subject.name}
                                            </Text>
                                        </TouchableOpacity>
                                    ))}
                                </View>
                            </View>

                            {/* Exam Batch Filter */}
                            {examBatches.length > 0 && (
                                <View style={styles.filterGroup}>
                                    <Text style={[styles.filterLabel, { color: theme.text }]}>
                                        Exam Type
                                    </Text>
                                    <View style={styles.filterOptions}>
                                        {examBatches.map(batch => (
                                            <TouchableOpacity
                                                key={batch.id}
                                                style={[
                                                    styles.filterOption,
                                                    { backgroundColor: theme.card },
                                                    filters.batchId === batch.id && {
                                                        borderLeftWidth: 3,
                                                        borderLeftColor: theme.primary
                                                    }
                                                ]}
                                                onPress={() => applyFilters({ batchId: batch.id })}
                                            >
                                                <Text style={[styles.filterOptionText, { color: theme.text }]}>
                                                    {batch.name}
                                                </Text>
                                                <Text style={[styles.filterOptionSubtext, { color: theme.textSecondary }]}>
                                                    {batch.type}
                                                </Text>
                                            </TouchableOpacity>
                                        ))}
                                    </View>
                                </View>
                            )}

                            {/* Academic Year Filter */}
                            {academicYears.length > 0 && (
                                <View style={styles.filterGroup}>
                                    <Text style={[styles.filterLabel, { color: theme.text }]}>
                                        Academic Year
                                    </Text>
                                    <View style={styles.filterOptions}>
                                        {academicYears.map(year => (
                                            <TouchableOpacity
                                                key={year.id}
                                                style={[
                                                    styles.filterOption,
                                                    { backgroundColor: theme.card },
                                                    filters.yearId === year.id && {
                                                        borderLeftWidth: 3,
                                                        borderLeftColor: theme.primary
                                                    }
                                                ]}
                                                onPress={() => applyFilters({ yearId: year.id })}
                                            >
                                                <Text style={[styles.filterOptionText, { color: theme.text }]}>
                                                    {year.title}
                                                </Text>
                                            </TouchableOpacity>
                                        ))}
                                    </View>
                                </View>
                            )}
                        </ScrollView>

                        <View style={styles.modalActions}>
                            <TouchableOpacity
                                style={[styles.modalButton, styles.modalButtonSecondary, { borderColor: theme.border }]}
                                onPress={resetFilters}
                            >
                                <Text style={[styles.modalButtonText, { color: theme.text }]}>
                                    Reset
                                </Text>
                            </TouchableOpacity>
                            <TouchableOpacity
                                style={[styles.modalButton, { backgroundColor: theme.primary }]}
                                onPress={() => setShowFilters(false)}
                            >
                                <Text style={[styles.modalButtonText, { color: '#FFFFFF' }]}>
                                    Apply
                                </Text>
                            </TouchableOpacity>
                        </View>
                    </View>
                </View>
            </Modal>
        </SafeAreaView>
    );
};

import React, { useState, useEffect, useCallback, useMemo } from 'react';
import {
    View,
    Text,
    StyleSheet,
    ScrollView,
    TouchableOpacity,
    Image,
    RefreshControl,
    ActivityIndicator,
    Alert,
} from 'react-native';
import { useSelector, useDispatch } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialCommunityIcons';
import { selectThemeData } from '../../store/slices/themeSlice';
import { selectChildren, setChildren, selectCurrentUser } from '../../store/slices/authSlice';
import ChildCard from '../../components/Parent/ChildCard';
import { useGetChildrenQuery } from '../../store/attendanceApi';

const ChildrenListScreen = ({ navigation }) => {
    const theme = useSelector(selectThemeData);
    const reduxChildren = useSelector(selectChildren);
    const currentUser = useSelector(selectCurrentUser);
    const dispatch = useDispatch();
    const [refreshing, setRefreshing] = useState(false);
    
    // Check if children exist in Redux store
    const hasChildrenInStore = useMemo(() => {
        return reduxChildren && Array.isArray(reduxChildren) && reduxChildren.length > 0;
    }, [reduxChildren]);

    console.log('📦 Children in Redux store:', {
        count: reduxChildren?.length || 0,
        hasChildren: hasChildrenInStore,
        firstChild: reduxChildren?.[0]?.name
    });

    // Only fetch from API if children not in store
    const {
        data: childrenData,
        isLoading,
        error,
        refetch,
    } = useGetChildrenQuery(currentUser?.id, {
        skip: !currentUser?.id || hasChildrenInStore, // Skip if children already in store
    });

    console.log('🌐 API Fetch status:', {
        skip: !currentUser?.id || hasChildrenInStore,
        isLoading: isLoading,
        hasData: !!childrenData,
        error: error
    });

    // Update Redux only when API data is fetched (only if store was empty)
    useEffect(() => {
        if (childrenData?.success === true && childrenData?.data && !hasChildrenInStore) {
            console.log('📥 Updating Redux store with API data');
            const transformedChildren = childrenData.data.map(child => ({
                reg_Id: child.reg_Id || child.id,
                reg_No: child.reg_No,
                name: child.name || child.childName,
                image: child.image || '',
                gender: child.gender,
                class_ID: child.class_ID || child.classId,
                className: child.className,
                section_ID: child.section_ID || child.sectionId,
                sectionName: child.sectionName,
                rollNo: child.rollNo,
                family_Code: child.family_Code || child.familyCode,
                campusName: child.campusName,
                status: child.status || 'A',
                dob: child.dob,
                bloodGroup: child.bloodGroup,
                address: child.address,
                admissionNo: child.admissionNo,
                diaryAccess: child.diaryAccess || 0, // Important: Store diary access
            }));
            dispatch(setChildren(transformedChildren));
        }
    }, [childrenData, dispatch, hasChildrenInStore]);

    const onRefresh = async () => {
        setRefreshing(true);
        // Force refetch from API on refresh
        const result = await refetch();
        if (result.data?.success === true && result.data?.data) {
            const transformedChildren = result.data.data.map(child => ({
                reg_Id: child.reg_Id || child.id,
                reg_No: child.reg_No,
                name: child.name || child.childName,
                image: child.image || '',
                gender: child.gender,
                class_ID: child.class_ID || child.classId,
                className: child.className,
                section_ID: child.section_ID || child.sectionId,
                sectionName: child.sectionName,
                rollNo: child.rollNo,
                family_Code: child.family_Code || child.familyCode,
                campusName: child.campusName,
                status: child.status || 'A',
                dob: child.dob,
                bloodGroup: child.bloodGroup,
                address: child.address,
                admissionNo: child.admissionNo,
                diaryAccess: child.diaryAccess || 0,
            }));
            dispatch(setChildren(transformedChildren));
        }
        setRefreshing(false);
    };

    const handleChildPress = (child) => {
        navigation.navigate('ChildDashboard', {
            childId: child.reg_Id,
            childName: child.name,
            childData: child,
        });
    };

    const handleViewAttendance = (child) => {
        navigation.navigate('ChildAttendance', {
            childId: child.reg_Id,
            childName: child.name,
        });
    };

    const handleViewGrades = (child) => {
        navigation.navigate('ChildGrades', {
            childId: child.reg_Id,
            childName: child.name,
        });
    };

    const handleViewDiary = (child) => {
        // Check if diary access is allowed (1 = allowed, 0 = not allowed)
        if (child.diaryAccess === 0) {
            Alert.alert(
                'Access Restricted',
                'Diary access is currently restricted. Please contact the school administration for more information.',
                [{ text: 'OK' }]
            );
            return;
        }
        
        navigation.navigate('ChildDiary', {
            childId: child.reg_Id,
            childName: child.name,
            childClass: child.className,
            childSection: child.sectionName,
        });
    };

    // Determine which children to display
    const displayChildren = useMemo(() => {
        // First try Redux store, then API data
        if (hasChildrenInStore) {
            return reduxChildren;
        }
        if (childrenData?.success === true && childrenData?.data) {
            return childrenData.data.map(child => ({
                ...child,
                diaryAccess: child.diaryAccess || 0,
            }));
        }
        return [];
    }, [hasChildrenInStore, reduxChildren, childrenData]);

    const isLoadingState = isLoading && !hasChildrenInStore && !childrenData;

    if (isLoadingState) {
        return (
            <View style={[styles.container, { backgroundColor: theme.background }]}>
                <ActivityIndicator size="large" color={theme.primary} />
                <Text style={[styles.loadingText, { color: theme.textSecondary }]}>
                    Loading children...
                </Text>
            </View>
        );
    }

    if (error && !hasChildrenInStore) {
        return (
            <View style={[styles.container, { backgroundColor: theme.background }]}>
                <Icon name="cloud-off-outline" size={60} color={theme.error} />
                <Text style={[styles.errorText, { color: theme.error }]}>
                    Failed to load children
                </Text>
                <TouchableOpacity
                    style={[styles.retryButton, { backgroundColor: theme.primary }]}
                    onPress={onRefresh}
                >
                    <Text style={styles.retryButtonText}>Try Again</Text>
                </TouchableOpacity>
            </View>
        );
    }

    return (
        <ScrollView
            style={[styles.container, { backgroundColor: theme.background }]}
            refreshControl={
                <RefreshControl
                    refreshing={refreshing}
                    onRefresh={onRefresh}
                    colors={[theme.primary]}
                />
            }
        >
            {/* Header */}
            <View style={[styles.header, { backgroundColor: theme.primary }]}>
                <View>
                    <Text style={styles.headerTitle}>My Children</Text>
                    <Text style={styles.headerSubtitle}>
                        {displayChildren.length} child{displayChildren.length !== 1 ? 'ren' : ''} enrolled
                    </Text>
                </View>
                <Icon name="account-group" size={40} color="#FFFFFF" />
            </View>

            {/* Children List */}
            <View style={styles.childrenList}>
                {displayChildren.map((child) => (
                    <ChildCard
                        key={child.reg_Id}
                        child={child}
                        onPress={() => handleChildPress(child)}
                        onViewAttendance={() => handleViewAttendance(child)}
                        onViewGrades={() => handleViewGrades(child)}
                        onViewDiary={() => handleViewDiary(child)}
                        theme={theme}
                    />
                ))}

                {displayChildren.length === 0 && !isLoadingState && (
                    <View style={styles.emptyContainer}>
                        <Icon
                            name="account-child-outline"
                            size={60}
                            color={theme.textSecondary}
                        />
                        <Text style={[styles.emptyText, { color: theme.textSecondary }]}>
                            No children found
                        </Text>
                        <Text style={[styles.emptySubtext, { color: theme.textSecondary }]}>
                            Contact school administration to enroll your child
                        </Text>
                    </View>
                )}
            </View>

            {/* Quick Info */}
            <View style={[styles.infoCard, { backgroundColor: theme.card }]}>
                <Icon name="information" size={24} color={theme.primary} />
                <View style={styles.infoContent}>
                    <Text style={[styles.infoTitle, { color: theme.text }]}>
                        Need Help?
                    </Text>
                    <Text style={[styles.infoText, { color: theme.textSecondary }]}>
                        Contact school office for any queries regarding your child's
                        education
                    </Text>
                </View>
            </View>
        </ScrollView>
    );
};

import React, { useRef, useState } from 'react';
import {
    View,
    Text,
    StyleSheet,
    SafeAreaView,
    ActivityIndicator,
} from 'react-native';
import { useSelector } from 'react-redux';
import Icon from 'react-native-vector-icons/MaterialCommunityIcons';
import { WebView } from 'react-native-webview';
import { selectThemeData } from '../../store/slices/themeSlice';

const DiaryDetailScreen = ({ route }) => {
    const theme = useSelector(selectThemeData);
    const { entry } = route.params;

    const webViewRef = useRef(null);
    const [isLoading, setIsLoading] = useState(true);

    const formatFullDate = (dateString) => {
        const date = new Date(dateString);
        return date.toLocaleDateString('en-US', {
            weekday: 'long',
            year: 'numeric',
            month: 'long',
            day: 'numeric',
        });
    };

    const getHTML = (content) => {
        return `
            <!DOCTYPE html>
            <html>
            <head>
                <meta name="viewport" content="width=device-width, initial-scale=1.0">
                <style>
                    * { margin:0; padding:0; box-sizing:border-box; }
                    body {
                        font-family: -apple-system, BlinkMacSystemFont, 'Segoe UI', Roboto;
                        font-size: 16px;
                        line-height: 1.7;
                        padding: 18px;
                        color: ${theme.isDarkMode ? '#e0e0e0' : '#2c3e50'};
                        background: ${theme.isDarkMode ? '#121212' : '#ffffff'};
                    }
                    p { margin-bottom: 14px; }
                    img {
                        max-width: 100%;
                        border-radius: 14px;
                        margin: 14px 0;
                    }
                </style>
            </head>
            <body>
                ${content || '<p style="text-align:center;color:#aaa">No content available</p>'}
            </body>
            </html>
        `;
    };

    return (
        <SafeAreaView style={[styles.container, { backgroundColor: theme.background }]}>

            {/* 🔥 Metadata Card */}
            <View style={styles.metaWrapper}>
                <View style={[
                    styles.metaCard,
                    {
                        backgroundColor: theme.card,
                        shadowColor: theme.isDarkMode ? '#000' : '#000',
                    }
                ]}>

                    {/* 📅 Date + Status */}
                    <View style={styles.metaTop}>
                        <Text style={[styles.metaDate, { color: theme.text }]}>
                            {formatFullDate(entry.date)}
                        </Text>

                        <View style={[
                            styles.statusBadge,
                            {
                                backgroundColor: entry.isPublished
                                    ? theme.success + '22'
                                    : theme.warning + '22'
                            }
                        ]}>
                            <View style={[
                                styles.statusDot,
                                {
                                    backgroundColor: entry.isPublished
                                        ? theme.success
                                        : theme.warning
                                }
                            ]}/>
                            <Text style={[
                                styles.statusText,
                                {
                                    color: entry.isPublished
                                        ? theme.success
                                        : theme.warning
                                }
                            ]}>
                                {entry.isPublished ? 'Published' : 'Draft'}
                            </Text>
                        </View>
                    </View>

                    {/* 📘 Subject */}
                    <View style={[
                        styles.subjectBadge,
                        { backgroundColor: theme.primary + '20' }
                    ]}>
                        <Icon name="book-open-variant" size={15} color={theme.primary} />
                        <Text style={[styles.subjectText, { color: theme.primary }]}>
                            {entry.subject}
                        </Text>
                    </View>

                    {/* Divider */}
                    <View style={[styles.divider, { backgroundColor: theme.border }]} />

                    {/* 👤 Teacher */}
                    <View style={styles.teacherRow}>
                        <View style={[
                            styles.avatar,
                            { backgroundColor: theme.primary + '25' }
                        ]}>
                            <Icon name="account" size={18} color={theme.primary} />
                        </View>

                        <View>
                            <Text style={[
                                styles.teacherLabel,
                                { color: theme.textSecondary }
                            ]}>
                                Teacher
                            </Text>

                            <Text style={[
                                styles.teacherName,
                                { color: theme.text }
                            ]}>
                                {entry.teacher}
                            </Text>
                        </View>
                    </View>

                </View>
            </View>

            {/* 🌐 WebView Content */}
            <View style={styles.webViewContainer}>
                {isLoading && (
                    <View style={styles.loader}>
                        <ActivityIndicator size="large" color={theme.primary} />
                        <Text style={[
                            styles.loaderText,
                            { color: theme.textSecondary }
                        ]}>
                            Loading content...
                        </Text>
                    </View>
                )}

                <WebView
                    ref={webViewRef}
                    source={{ html: getHTML(entry.content) }}
                    style={styles.webView}
                    onLoadEnd={() => setIsLoading(false)}
                    originWhitelist={['*']}
                />
            </View>

        </SafeAreaView>
    );
};

export default DiaryDetailScreen;
