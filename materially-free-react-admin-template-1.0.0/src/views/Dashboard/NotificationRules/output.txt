const CronConverter = ({ value, onChange, disabled = false, label = "Schedule Configuration" }) => {
    const [selectedDate, setSelectedDate] = useState(new Date());
    const [frequency, setFrequency] = useState('daily');
    const [customInterval, setCustomInterval] = useState(1);
    const [intervalUnit, setIntervalUnit] = useState('hours');
    const [generatedCron, setGeneratedCron] = useState(value || '0 9 * * *');
    const [copied, setCopied] = useState(false);
    const [showAdvanced, setShowAdvanced] = useState(false);
    const [manualCron, setManualCron] = useState(value || '');
    const [validationError, setValidationError] = useState('');

    const frequencies = [
        { value: 'once', label: 'Once at specific time', description: 'Runs only once at the specified date and time' },
        { value: 'daily', label: 'Daily', description: 'Runs every day at the specified time' },
        { value: 'weekly', label: 'Weekly', description: 'Runs every week on the selected day and time' },
        { value: 'monthly', label: 'Monthly', description: 'Runs every month on the selected day and time' },
        { value: 'yearly', label: 'Yearly', description: 'Runs every year on the selected date and time' },
        { value: 'hourly', label: 'Hourly', description: 'Runs every hour at the specified minute' },
        { value: 'every_minute', label: 'Every Minute', description: 'Runs every minute' },
        { value: 'custom', label: 'Custom Interval', description: 'Custom interval (every X minutes/hours/days)' },
    ];

    const presets = [
        { label: 'Daily at 9 AM', cron: '0 9 * * *' },
        { label: 'Daily at midnight', cron: '0 0 * * *' },
        { label: 'Every Monday at 9 AM', cron: '0 9 * * 1' },
        { label: '1st of every month', cron: '0 0 1 * *' },
        { label: 'Every hour', cron: '0 * * * *' },
        { label: 'Every 30 minutes', cron: '*/30 * * * *' },
        { label: 'Every 2 hours', cron: '0 */2 * * *' },
        { label: 'Weekdays at 9 AM', cron: '0 9 * * 1-5' },
        { label: 'Weekends at 10 AM', cron: '0 10 * * 6,0' },
        { label: 'Monthly on 15th at 2 PM', cron: '0 14 15 * *' },
    ];

    // Initialize with current value
    useEffect(() => {
        if (value && isValidCron(value)) {
            setGeneratedCron(value);
            setManualCron(value);
            // Try to guess frequency from cron
            guessFrequencyFromCron(value);
        }
    }, [value]);

    const guessFrequencyFromCron = (cron) => {
        const [minute, hour, day, month, dayOfWeek] = cron.split(' ');

        if (cron === '* * * * *') {
            setFrequency('every_minute');
        } else if (minute === '0' && hour === '0' && day === '1' && month === '*') {
            setFrequency('monthly');
        } else if (minute === '0' && hour === '0' && day === '1' && month === '1') {
            setFrequency('yearly');
        } else if (dayOfWeek !== '*' && day === '*') {
            setFrequency('weekly');
        } else if (minute !== '*' && hour !== '*' && day === '*' && month === '*') {
            setFrequency('daily');
        } else if (hour.startsWith('*/') && minute === '0') {
            setFrequency('custom');
            const interval = hour.substring(2);
            setCustomInterval(parseInt(interval));
            setIntervalUnit('hours');
        } else if (minute.startsWith('*/')) {
            setFrequency('custom');
            const interval = minute.substring(2);
            setCustomInterval(parseInt(interval));
            setIntervalUnit('minutes');
        }
    };

    const generateCron = () => {
        try {
            const date = selectedDate;
            let cron;

            if (frequency === 'custom') {
                cron = dateToCron(date, 'custom', {
                    interval: customInterval,
                    unit: intervalUnit
                });
            } else {
                cron = dateToCron(date, frequency);
            }

            setGeneratedCron(cron);
            setManualCron(cron);
            setValidationError('');

            if (onChange) {
                onChange(cron);
            }
        } catch (error) {
            console.error('Error generating cron:', error);
            setValidationError('Failed to generate cron expression');
        }
    };

    const handlePresetClick = (cron) => {
        setGeneratedCron(cron);
        setManualCron(cron);
        setValidationError('');
        guessFrequencyFromCron(cron);

        if (onChange) {
            onChange(cron);
        }
    };

    const handleManualCronChange = (e) => {
        const cron = e.target.value;
        setManualCron(cron);

        if (cron && isValidCron(cron)) {
            setGeneratedCron(cron);
            setValidationError('');
            guessFrequencyFromCron(cron);

            if (onChange) {
                onChange(cron);
            }
        } else if (cron) {
            setValidationError('Invalid cron format. Must be 5 parts: minute hour day month day-of-week');
        } else {
            setValidationError('');
        }
    };

    const copyToClipboard = () => {
        navigator.clipboard.writeText(generatedCron).then(() => {
            setCopied(true);
            setTimeout(() => setCopied(false), 2000);
        });
    };

    const formatDateTime = () => {
        return dayjs(selectedDate).format('dddd, MMMM D, YYYY [at] h:mm A');
    };

    const getSchedulePreviews = () => {
        if (!isValidCron(generatedCron)) return [];
        return generateSchedulePreview(generatedCron, 3);
    };

    const schedulePreviews = getSchedulePreviews();
    const nextRunTime = getNextRunTime(generatedCron);
    const humanReadable = parseCronToText(generatedCron);

    return (
        <Card variant="outlined" sx={{ mb: 3 }}>
            <CardContent>
                <Box display="flex" alignItems="center" gap={1} mb={2}>
                    <ScheduleIcon color="primary" />
                    <Typography variant="h6">{label}</Typography>
                </Box>

                <Grid container spacing={3}>
                    {/* Left Column - Configuration */}
                    <Grid item xs={12} md={6}>
                        <Paper sx={{ p: 2, height: '100%', bgcolor: 'grey.50' }}>
                            <Typography variant="subtitle2" gutterBottom color="text.secondary">
                                Configure Schedule
                            </Typography>

                            {/* Frequency Selection */}
                            <FormControl fullWidth sx={{ mb: 3 }}>
                                <InputLabel>Frequency</InputLabel>
                                <Select
                                    value={frequency}
                                    label="Frequency"
                                    onChange={(e) => setFrequency(e.target.value)}
                                    disabled={disabled}
                                >
                                    {frequencies.map((freq) => (
                                        <MenuItem key={freq.value} value={freq.value}>
                                            <Box>
                                                <Typography>{freq.label}</Typography>
                                                <Typography variant="caption" color="text.secondary">
                                                    {freq.description}
                                                </Typography>
                                            </Box>
                                        </MenuItem>
                                    ))}
                                </Select>
                            </FormControl>

                            {/* Date Time Picker */}
                            <Box sx={{ mb: 3 }}>
                                <Typography variant="body2" gutterBottom>
                                    Select Date & Time
                                </Typography>
                                <StyledDatePicker
                                    selected={selectedDate}
                                    onChange={(date) => setSelectedDate(date)}
                                    showTimeSelect
                                    timeFormat="HH:mm"
                                    timeIntervals={15}
                                    dateFormat="MMMM d, yyyy h:mm aa"
                                    disabled={disabled || frequency === 'every_minute'}
                                    customInput={
                                        <TextField
                                            fullWidth
                                            InputProps={{
                                                startAdornment: (
                                                    <InputAdornment position="start">
                                                        <DateIcon />
                                                    </InputAdornment>
                                                ),
                                            }}
                                        />
                                    }
                                />
                            </Box>

                            {/* Custom Interval Options */}
                            {frequency === 'custom' && (
                                <Box sx={{ mb: 3 }}>
                                    <Typography variant="body2" gutterBottom>
                                        Custom Interval
                                    </Typography>
                                    <Grid container spacing={2}>
                                        <Grid item xs={6}>
                                            <TextField
                                                fullWidth
                                                label="Interval"
                                                type="number"
                                                value={customInterval}
                                                onChange={(e) => setCustomInterval(e.target.value)}
                                                InputProps={{ inputProps: { min: 1, max: 59 } }}
                                            />
                                        </Grid>
                                        <Grid item xs={6}>
                                            <FormControl fullWidth>
                                                <InputLabel>Unit</InputLabel>
                                                <Select
                                                    value={intervalUnit}
                                                    label="Unit"
                                                    onChange={(e) => setIntervalUnit(e.target.value)}
                                                >
                                                    <MenuItem value="minutes">Minutes</MenuItem>
                                                    <MenuItem value="hours">Hours</MenuItem>
                                                    <MenuItem value="days">Days</MenuItem>
                                                </Select>
                                            </FormControl>
                                        </Grid>
                                    </Grid>
                                </Box>
                            )}

                            {/* Generate Button */}
                            <Button
                                variant="contained"
                                fullWidth
                                onClick={generateCron}
                                disabled={disabled}
                                startIcon={<RefreshIcon />}
                                sx={{ mt: 2 }}
                            >
                                Generate Cron Expression
                            </Button>

                            {/* Quick Presets */}
                            <Box sx={{ mt: 3 }}>
                                <Typography variant="body2" gutterBottom color="text.secondary">
                                    Quick Presets:
                                </Typography>
                                <Box display="flex" flexWrap="wrap" gap={1}>
                                    {presets.map((preset) => (
                                        <Chip
                                            key={preset.cron}
                                            label={preset.label}
                                            size="small"
                                            onClick={() => handlePresetClick(preset.cron)}
                                            variant="outlined"
                                            sx={{ cursor: 'pointer', mb: 0.5 }}
                                        />
                                    ))}
                                </Box>
                            </Box>
                        </Paper>
                    </Grid>

                    {/* Right Column - Results */}
                    <Grid item xs={12} md={6}>
                        <Paper sx={{ p: 2, height: '100%' }}>
                            <Typography variant="subtitle2" gutterBottom color="text.secondary">
                                Generated Cron Expression
                            </Typography>

                            {/* Cron Display */}
                            <Paper
                                sx={{
                                    p: 2,
                                    bgcolor: 'primary.50',
                                    mb: 2,
                                    border: '1px solid',
                                    borderColor: 'primary.100',
                                    borderRadius: 1,
                                    position: 'relative',
                                }}
                            >
                                <Box display="flex" alignItems="center" justifyContent="space-between">
                                    <Typography
                                        variant="h6"
                                        fontFamily="monospace"
                                        color="primary.main"
                                        sx={{ wordBreak: 'break-all' }}
                                    >
                                        {generatedCron}
                                    </Typography>
                                    <Tooltip title={copied ? "Copied!" : "Copy to clipboard"}>
                                        <IconButton
                                            size="small"
                                            onClick={copyToClipboard}
                                            color={copied ? "success" : "primary"}
                                        >
                                            {copied ? <CheckIcon /> : <CopyIcon />}
                                        </IconButton>
                                    </Tooltip>
                                </Box>
                            </Paper>

                            {/* Human Readable Description */}
                            <Alert
                                severity="info"
                                icon={<InfoIcon />}
                                sx={{ mb: 2 }}
                            >
                                <Box>
                                    <Typography variant="body2" fontWeight="medium">
                                        {humanReadable}
                                    </Typography>
                                    {frequency !== 'every_minute' && frequency !== 'once' && (
                                        <Typography variant="caption" display="block" sx={{ mt: 0.5 }}>
                                            Based on: {formatDateTime()}
                                        </Typography>
                                    )}
                                </Box>
                            </Alert>

                            {/* Next Run Time */}
                            {nextRunTime && (
                                <Alert
                                    severity="success"
                                    sx={{ mb: 2 }}
                                    icon={<ArrowForwardIcon />}
                                >
                                    <Typography variant="body2">
                                        Next run: {dayjs(nextRunTime).format('dddd, MMMM D, YYYY [at] h:mm A')}
                                    </Typography>
                                </Alert>
                            )}

                            {/* Schedule Previews */}
                            {schedulePreviews.length > 0 && (
                                <Box sx={{ mb: 2 }}>
                                    <Typography variant="caption" color="text.secondary" display="block" gutterBottom>
                                        Upcoming runs:
                                    </Typography>
                                    <Stack spacing={0.5}>
                                        {schedulePreviews.map((time, index) => (
                                            <Box
                                                key={index}
                                                display="flex"
                                                alignItems="center"
                                                gap={1}
                                                sx={{ px: 1, py: 0.5 }}
                                            >
                                                <TimeIcon fontSize="small" color="action" />
                                                <Typography variant="caption">
                                                    {dayjs(time).format('MMM D, h:mm A')}
                                                </Typography>
                                            </Box>
                                        ))}
                                    </Stack>
                                </Box>
                            )}

                            {/* Validation Status */}
                            <Box sx={{ mb: 2 }}>
                                {isValidCron(generatedCron) ? (
                                    <Alert severity="success" icon={<CheckIcon />}>
                                        Valid cron expression
                                    </Alert>
                                ) : (
                                    <Alert severity="error">
                                        Invalid cron expression
                                    </Alert>
                                )}
                            </Box>

                            {/* Advanced Manual Input */}
                            <Box>
                                <FormControlLabel
                                    control={
                                        <Switch
                                            size="small"
                                            checked={showAdvanced}
                                            onChange={() => setShowAdvanced(!showAdvanced)}
                                        />
                                    }
                                    label="Advanced: Edit manually"
                                />

                                <Collapse in={showAdvanced}>
                                    <Box sx={{ mt: 2 }}>
                                        <TextField
                                            fullWidth
                                            label="Manual Cron Expression"
                                            value={manualCron}
                                            onChange={handleManualCronChange}
                                            placeholder="e.g., 0 9 * * *"
                                            error={!!validationError}
                                            helperText={validationError || 'Format: minute hour day-of-month month day-of-week'}
                                            InputProps={{
                                                startAdornment: (
                                                    <InputAdornment position="start">
                                                        <TimeIcon sx={{ color: 'action.active' }} />
                                                    </InputAdornment>
                                                ),
                                            }}
                                        />
                                        <Typography variant="caption" color="text.secondary" sx={{ mt: 1, display: 'block' }}>
                                            Examples:
                                            <br />• Daily at 9 AM: "0 9 * * *"
                                            <br />• Every Monday at 2 PM: "0 14 * * 1"
                                            <br />• Every 30 minutes: "*/30 * * * *"
                                            <br />• 1st of every month: "0 0 1 * *"
                                        </Typography>
                                    </Box>
                                </Collapse>
                            </Box>
                        </Paper>
                    </Grid>
                </Grid>

                {/* Cron Explanation */}
                <Box sx={{ mt: 3, pt: 2, borderTop: 1, borderColor: 'divider' }}>
                    <Typography variant="body2" color="text.secondary" gutterBottom>
                        <strong>Cron Format:</strong> minute(0-59) hour(0-23) day-of-month(1-31) month(1-12) day-of-week(0-6, 0=Sunday)
                    </Typography>
                    <Typography variant="caption" color="text.secondary">
                        * = any value, */n = every n units, 1-5 = range, 1,3,5 = list
                    </Typography>
                </Box>
            </CardContent>
        </Card>
    );
};

export default CronConverter;

const NotificationRulesList = () => {
    const navigate = useNavigate();
    const { isAdmin, userRole } = usePermissions();
    const theme = useTheme();
    const isMobile = useMediaQuery(theme.breakpoints.down('sm'));

    const [page, setPage] = useState(0);
    const [rowsPerPage, setRowsPerPage] = useState(10);
    const [searchTerm, setSearchTerm] = useState('');
    const [statusFilter, setStatusFilter] = useState('all');
    const [typeFilter, setTypeFilter] = useState('all');
    const [categoryFilter, setCategoryFilter] = useState('all');
    const [deleteDialog, setDeleteDialog] = useState(null);
    const [executeDialog, setExecuteDialog] = useState(null);

    const { data, isLoading, error, refetch } = useGetNotificationRulesQuery();
    const [deleteRule] = useDeleteNotificationRuleMutation();
    const [activateRule] = useActivateNotificationRuleMutation();
    const [executeRule] = useExecuteNotificationRuleMutation();

    const rules = data?.data || [];

    const handleChangePage = (event, newPage) => {
        setPage(newPage);
    };

    const handleChangeRowsPerPage = (event) => {
        setRowsPerPage(parseInt(event.target.value, 10));
        setPage(0);
    };

    const handleDelete = async () => {
        if (deleteDialog) {
            try {
                await deleteRule(deleteDialog.ruleID).unwrap();
                refetch();
                setDeleteDialog(null);
            } catch (error) {
                console.error('Failed to delete rule:', error);
            }
        }
    };

    const handleActivate = async (ruleId, isActive) => {
        try {
            await activateRule({ id: ruleId, isActive: !isActive }).unwrap();
            refetch();
        } catch (error) {
            console.error('Failed to update rule status:', error);
        }
    };

    const handleExecute = async () => {
        if (executeDialog) {
            try {
                await executeRule(executeDialog.ruleID).unwrap();
                refetch();
                setExecuteDialog(null);
            } catch (error) {
                console.error('Failed to execute rule:', error);
            }
        }
    };

    const filteredRules = useMemo(() => {
        return rules.filter(rule => {
            const matchesSearch =
                rule.ruleName?.toLowerCase().includes(searchTerm.toLowerCase()) ||
                rule.description?.toLowerCase().includes(searchTerm.toLowerCase()) ||
                rule.category?.toLowerCase().includes(searchTerm.toLowerCase());

            const matchesStatus = statusFilter === 'all' ||
                (statusFilter === 'active' && rule.isActive) ||
                (statusFilter === 'inactive' && !rule.isActive);

            const matchesType = typeFilter === 'all' || rule.ruleType === typeFilter;
            const matchesCategory = categoryFilter === 'all' || rule.category === categoryFilter;

            return matchesSearch && matchesStatus && matchesType && matchesCategory;
        });
    }, [rules, searchTerm, statusFilter, typeFilter, categoryFilter]);

    const paginatedRules = filteredRules.slice(
        page * rowsPerPage,
        page * rowsPerPage + rowsPerPage
    );

    const getStatusColor = (isActive) => {
        return isActive ? 'success' : 'error';
    };

    const getStatusIcon = (isActive) => {
        return isActive ? <ActiveIcon fontSize="small" /> : <InactiveIcon fontSize="small" />;
    };

    const getRuleTypeIcon = (type) => {
        switch (type) {
            case 'EVENT': return <EventIcon fontSize="small" />;
            case 'SCHEDULED': return <ScheduleIcon fontSize="small" />;
            case 'CRON': return <CronIcon fontSize="small" />;
            case 'MANUAL': return <ManualIcon fontSize="small" />;
            default: return <EventIcon fontSize="small" />;
        }
    };

    const getRuleTypeColor = (type) => {
        switch (type) {
            case 'EVENT': return 'primary';
            case 'SCHEDULED': return 'info';
            case 'CRON': return 'warning';
            case 'MANUAL': return 'secondary';
            default: return 'default';
        }
    };

    const getCategoryColor = (category) => {
        switch (category) {
            case 'ATTENDANCE': return 'error';
            case 'FEE': return 'warning';
            case 'EXAM': return 'info';
            case 'GENERAL': return 'success';
            case 'ALERT': return 'error';
            case 'ANNOUNCEMENT': return 'primary';
            default: return 'default';
        }
    };

    const getPriorityIcon = (priority) => {
        switch (priority) {
            case 1: return <HighPriorityIcon fontSize="small" />;
            case 2: return <MediumPriorityIcon fontSize="small" />;
            case 3: return <LowPriorityIcon fontSize="small" />;
            default: return <MediumPriorityIcon fontSize="small" />;
        }
    };

    const getPriorityColor = (priority) => {
        switch (priority) {
            case 1: return 'error';
            case 2: return 'warning';
            case 3: return 'info';
            default: return 'default';
        }
    };

    const formatLastRun = (lastRunTime) => {
        if (!lastRunTime) return 'Never';
        return dayjs(lastRunTime).fromNow();
    };

    const formatNextRun = (nextRunTime) => {
        if (!nextRunTime) return 'N/A';
        return dayjs(nextRunTime).format('MMM D, h:mm A');
    };

    if (!isAdmin()) {
        return (
            <Box display="flex" justifyContent="center" alignItems="center" minHeight="60vh">
                <Alert severity="error" sx={{ maxWidth: 400 }}>
                    <Typography variant="h6">Access Denied</Typography>
                    <Typography>
                        Notification rule management is only available to administrators.
                    </Typography>
                </Alert>
            </Box>
        );
    }

    if (error) {
        return (
            <Box display="flex" justifyContent="center" alignItems="center" minHeight="60vh">
                <Alert severity="error" sx={{ maxWidth: 400 }}>
                    <Typography variant="h6">Error Loading Rules</Typography>
                    <Typography>{error.data?.message || 'Failed to load notification rules'}</Typography>
                    <Button onClick={refetch} sx={{ mt: 2 }}>Retry</Button>
                </Alert>
            </Box>
        );
    }

    return (
        <Box sx={{ p: { xs: 2, sm: 3 } }}>
            {/* Header */}
            <Box display="flex" flexDirection={{ xs: 'column', sm: 'row' }}
                justifyContent="space-between" alignItems={{ xs: 'stretch', sm: 'center' }}
                gap={2} mb={4}>
                <Box>

                        Notification Rules
                    </Typography>

                <Box display="flex" gap={2}>
                    <Button
                        variant="outlined"
                        startIcon={<RefreshIcon />}
                        onClick={() => refetch()}
                        disabled={isLoading}
                    >
                        Refresh
                    </Button>
                    <Button
                        variant="contained"
                        startIcon={<AddIcon />}
                        onClick={() => navigate('/dashboard/notification-rules/create')}
                    >
                        New Rule
                    </Button>
                </Box>
            </Box>

            {/* Stats Cards */}
            <Box display="flex" gap={2} mb={3} flexWrap="wrap">
                <Card sx={{ flex: 1, minWidth: 150 }}>
                    <Box p={2}>
                        <Typography variant="body2" color="text.secondary">
                            Total Rules
                        </Typography>
                        <Typography variant="h4">
                            {rules.length}
                        </Typography>
                    </Box>
                </Card>

                <Card sx={{ flex: 1, minWidth: 150 }}>
                    <Box p={2}>
                        <Typography variant="body2" color="text.secondary">
                            Active
                        </Typography>
                        <Typography variant="h4" color="success.main">
                            {rules.filter(r => r.isActive).length}
                        </Typography>
                    </Box>
                </Card>

                <Card sx={{ flex: 1, minWidth: 150 }}>
                    <Box p={2}>
                        <Typography variant="body2" color="text.secondary">
                            Scheduled
                        </Typography>
                        <Typography variant="h4" color="info.main">
                            {rules.filter(r => r.ruleType === 'SCHEDULED' || r.ruleType === 'CRON').length}
                        </Typography>
                    </Box>
                </Card>

                <Card sx={{ flex: 1, minWidth: 150 }}>
                    <Box p={2}>
                        <Typography variant="body2" color="text.secondary">
                            Events
                        </Typography>
                        <Typography variant="h4" color="primary.main">
                            {rules.filter(r => r.ruleType === 'EVENT').length}
                        </Typography>
                    </Box>
                </Card>
            </Box>

            {/* Filters */}
            <Card sx={{ mb: 3 }}>
                <Box p={3}>
                    <Typography variant="h6" gutterBottom>
                        Filters
                    </Typography>
                    <Box display="flex" flexDirection={{ xs: 'column', sm: 'row' }} gap={2} flexWrap="wrap">
                        <TextField
                            placeholder="Search rules..."
                            value={searchTerm}
                            onChange={(e) => setSearchTerm(e.target.value)}
                            size="small"
                            sx={{ minWidth: 250 }}
                            InputProps={{
                                startAdornment: (
                                    <InputAdornment position="start">
                                        <SearchIcon fontSize="small" color="action" />
                                    </InputAdornment>
                                ),
                                endAdornment: searchTerm && (
                                    <InputAdornment position="end">
                                        <IconButton size="small" onClick={() => setSearchTerm('')}>
                                            <CancelIcon fontSize="small" />
                                        </IconButton>
                                    </InputAdornment>
                                ),
                            }}
                        />

                        <FormControl size="small" sx={{ minWidth: 120 }}>
                            <InputLabel>Status</InputLabel>
                            <Select
                                value={statusFilter}
                                label="Status"
                                onChange={(e) => setStatusFilter(e.target.value)}
                            >
                                <MenuItem value="all">All Status</MenuItem>
                                <MenuItem value="active">Active</MenuItem>
                                <MenuItem value="inactive">Inactive</MenuItem>
                            </Select>
                        </FormControl>

                        <FormControl size="small" sx={{ minWidth: 120 }}>
                            <InputLabel>Type</InputLabel>
                            <Select
                                value={typeFilter}
                                label="Type"
                                onChange={(e) => setTypeFilter(e.target.value)}
                            >
                                <MenuItem value="all">All Types</MenuItem>
                                <MenuItem value="EVENT">Event</MenuItem>
                                <MenuItem value="SCHEDULED">Scheduled</MenuItem>
                                <MenuItem value="CRON">Cron</MenuItem>
                                <MenuItem value="MANUAL">Manual</MenuItem>
                            </Select>
                        </FormControl>

                        <FormControl size="small" sx={{ minWidth: 120 }}>
                            <InputLabel>Category</InputLabel>
                            <Select
                                value={categoryFilter}
                                label="Category"
                                onChange={(e) => setCategoryFilter(e.target.value)}
                            >
                                <MenuItem value="all">All Categories</MenuItem>
                                <MenuItem value="ATTENDANCE">Attendance</MenuItem>
                                <MenuItem value="FEE">Fee</MenuItem>
                                <MenuItem value="EXAM">Exam</MenuItem>
                                <MenuItem value="GENERAL">General</MenuItem>
                                <MenuItem value="ALERT">Alert</MenuItem>
                                <MenuItem value="ANNOUNCEMENT">Announcement</MenuItem>
                            </Select>
                        </FormControl>

                        <Button
                            variant="outlined"
                            startIcon={<FilterIcon />}
                            onClick={() => {
                                setSearchTerm('');
                                setStatusFilter('all');
                                setTypeFilter('all');
                                setCategoryFilter('all');
                            }}
                        >
                            Clear Filters
                        </Button>
                    </Box>
                </Box>
            </Card>

            {/* Rules Table */}
            <Card>
                {isLoading ? (
                    <Box display="flex" justifyContent="center" alignItems="center" minHeight="200px">
                        <CircularProgress />
                    </Box>
                ) : (
                    <>
                        <TableContainer>
                            <Table>
                                <TableHead>
                                    <TableRow>
                                        <TableCell>Name</TableCell>
                                        {!isMobile && (
                                            <>
                                                <TableCell>Type</TableCell>
                                                <TableCell>Category</TableCell>
                                                <TableCell>Priority</TableCell>
                                                <TableCell>Last Run</TableCell>
                                                <TableCell>Next Run</TableCell>
                                            </>
                                        )}
                                        <TableCell>Status</TableCell>
                                        <TableCell align="right">Actions</TableCell>
                                    </TableRow>
                                </TableHead>
                                <TableBody>
                                    {paginatedRules.length === 0 ? (
                                        <TableRow>
                                            <TableCell colSpan={isMobile ? 3 : 8} align="center" sx={{ py: 4 }}>
                                                <Box textAlign="center">
                                                    <Typography variant="body1" color="text.secondary" gutterBottom>
                                                        No rules found
                                                    </Typography>
                                                    <Typography variant="body2" color="text.secondary">
                                                        {searchTerm || statusFilter !== 'all' || typeFilter !== 'all'
                                                            ? 'Try adjusting your filters'
                                                            : 'Create your first notification rule'}
                                                    </Typography>
                                                    {!searchTerm && statusFilter === 'all' && typeFilter === 'all' && (
                                                        <Button
                                                            variant="outlined"
                                                            startIcon={<AddIcon />}
                                                            onClick={() => navigate('/dashboard/notification-rules/create')}
                                                            sx={{ mt: 2 }}
                                                        >
                                                            Create Rule
                                                        </Button>
                                                    )}
                                                </Box>
                                            </TableCell>
                                        </TableRow>
                                    ) : (
                                        paginatedRules.map((rule) => (
                                            <TableRow
                                                key={rule.ruleID}
                                                hover
                                                sx={{
                                                    '&:hover': {
                                                        bgcolor: 'action.hover'
                                                    },
                                                    cursor: 'pointer'
                                                }}
                                                onClick={() => navigate(`/dashboard/notification-rules/${rule.ruleID}`)}
                                            >
                                                <TableCell>
                                                    <Box display="flex" flexDirection="column">
                                                        <Typography fontWeight="medium">
                                                            {rule.ruleName}
                                                        </Typography>
                                                        {isMobile && (
                                                            <Box display="flex" gap={1} flexWrap="wrap" mt={1}>
                                                                <Chip
                                                                    label={rule.ruleType}
                                                                    size="small"
                                                                    icon={getRuleTypeIcon(rule.ruleType)}
                                                                    color={getRuleTypeColor(rule.ruleType)}
                                                                />
                                                                <Chip
                                                                    label={rule.category}
                                                                    size="small"
                                                                    color={getCategoryColor(rule.category)}
                                                                />
                                                                <Chip
                                                                    label={`P${rule.priority}`}
                                                                    size="small"
                                                                    icon={getPriorityIcon(rule.priority)}
                                                                    color={getPriorityColor(rule.priority)}
                                                                />
                                                            </Box>
                                                        )}
                                                    </Box>
                                                </TableCell>

                                                {!isMobile && (
                                                    <>
                                                        <TableCell>
                                                            <Chip
                                                                label={rule.ruleType}
                                                                size="small"
                                                                icon={getRuleTypeIcon(rule.ruleType)}
                                                                color={getRuleTypeColor(rule.ruleType)}
                                                                variant="outlined"
                                                            />
                                                        </TableCell>
                                                        <TableCell>
                                                            <Chip
                                                                label={rule.category}
                                                                size="small"
                                                                color={getCategoryColor(rule.category)}
                                                            />
                                                        </TableCell>
                                                        <TableCell>
                                                            <Tooltip title={`Priority ${rule.priority}`}>
                                                                <Chip
                                                                    label={`P${rule.priority}`}
                                                                    size="small"
                                                                    icon={getPriorityIcon(rule.priority)}
                                                                    color={getPriorityColor(rule.priority)}
                                                                />
                                                            </Tooltip>
                                                        </TableCell>
                                                        <TableCell>
                                                            <Typography variant="body2" color="text.secondary">
                                                                {formatLastRun(rule.lastRunTime)}
                                                            </Typography>
                                                        </TableCell>
                                                        <TableCell>
                                                            <Typography variant="body2" color="text.secondary">
                                                                {formatNextRun(rule.nextRunTime)}
                                                            </Typography>
                                                        </TableCell>
                                                    </>
                                                )}

                                                <TableCell>
                                                    <Box display="flex" alignItems="center" gap={1}>
                                                        <Tooltip title={rule.isActive ? 'Active' : 'Inactive'}>
                                                            <IconButton
                                                                size="small"
                                                                onClick={(e) => {
                                                                    e.stopPropagation();
                                                                    handleActivate(rule.ruleID, rule.isActive);
                                                                }}
                                                                color={getStatusColor(rule.isActive)}
                                                            >
                                                                {getStatusIcon(rule.isActive)}
                                                            </IconButton>
                                                        </Tooltip>
                                                        <Switch
                                                            size="small"
                                                            checked={rule.isActive}
                                                            onClick={(e) => e.stopPropagation()}
                                                            onChange={() => handleActivate(rule.ruleID, rule.isActive)}
                                                            color="success"
                                                        />
                                                    </Box>
                                                </TableCell>

                                                <TableCell align="right" onClick={(e) => e.stopPropagation()}>
                                                    <Box display="flex" justifyContent="flex-end" gap={1}>
                                                        <Tooltip title="Edit Rule">
                                                            <IconButton
                                                                size="small"
                                                                onClick={() => navigate(`/dashboard/notification-rules/${rule.ruleID}`)}
                                                                color="primary"
                                                            >
                                                                <EditIcon fontSize="small" />
                                                            </IconButton>
                                                        </Tooltip>

                                                        <Tooltip title="Execute Now">
                                                            <IconButton
                                                                size="small"
                                                                onClick={() => setExecuteDialog(rule)}
                                                                color="secondary"
                                                            >
                                                                <ExecuteIcon fontSize="small" />
                                                            </IconButton>
                                                        </Tooltip>

                                                        <Tooltip title="Delete Rule">
                                                            <IconButton
                                                                size="small"
                                                                color="error"
                                                                onClick={() => setDeleteDialog(rule)}
                                                            >
                                                                <DeleteIcon fontSize="small" />
                                                            </IconButton>
                                                        </Tooltip>
                                                    </Box>
                                                </TableCell>
                                            </TableRow>
                                        ))
                                    )}
                                </TableBody>
                            </Table>
                        </TableContainer>

                        <TablePagination
                            rowsPerPageOptions={[5, 10, 25, 50]}
                            component="div"
                            count={filteredRules.length}
                            rowsPerPage={rowsPerPage}
                            page={page}
                            onPageChange={handleChangePage}
                            onRowsPerPageChange={handleChangeRowsPerPage}
                            labelRowsPerPage="Rules per page:"
                            labelDisplayedRows={({ from, to, count }) => `${from}-${to} of ${count}`}
                        />
                    </>
                )}
            </Card>

            {/* Delete Confirmation Dialog */}
            <Dialog open={!!deleteDialog} onClose={() => setDeleteDialog(null)} maxWidth="xs" fullWidth>
                <DialogTitle>Delete Rule</DialogTitle>
                <DialogContent>
                    <Box display="flex" alignItems="center" gap={2} mb={2}>
                        <DeleteIcon color="error" />
                        <Typography variant="body1">
                            Are you sure you want to delete "{deleteDialog?.ruleName}"?
                        </Typography>
                    </Box>
                    <Typography variant="body2" color="text.secondary">
                        This action cannot be undone. All related notification history will be removed.
                    </Typography>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setDeleteDialog(null)}>Cancel</Button>
                    <Button onClick={handleDelete} color="error" variant="contained" autoFocus>
                        Delete Rule
                    </Button>
                </DialogActions>
            </Dialog>

            {/* Execute Confirmation Dialog */}
            <Dialog open={!!executeDialog} onClose={() => setExecuteDialog(null)} maxWidth="xs" fullWidth>
                <DialogTitle>Execute Rule</DialogTitle>
                <DialogContent>
                    <Box display="flex" alignItems="center" gap={2} mb={2}>
                        <ExecuteIcon color="primary" />
                        <Typography variant="body1">
                            Execute "{executeDialog?.ruleName}" now?
                        </Typography>
                    </Box>
                    <Typography variant="body2" color="text.secondary">
                        This will trigger the rule immediately and send notifications to all configured recipients.
                    </Typography>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setExecuteDialog(null)}>Cancel</Button>
                    <Button onClick={handleExecute} color="primary" variant="contained" autoFocus>
                        Execute Now
                    </Button>
                </DialogActions>
            </Dialog>
        </Box>
    );
};

export default NotificationRulesList;
