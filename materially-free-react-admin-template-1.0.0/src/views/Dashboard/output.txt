const SalesLineCard = ({ bgColor, chartData, footerData, icon, title, percentage }) => {
  const theme = useTheme();

  let footerHtml;

  if (footerData) {
    footerHtml = footerData.map((item, index) => {
      return (
        <Grid item key={index}>
          <Box mt={3} mb={3} p={1}>
            <Grid container direction="column" spacing={1} alignItems="center">
              <Typography variant="h4">{item.value}</Typography>
              <Typography variant="subtitle2" color="secondary">
                {item.label}
              </Typography>
            </Grid>
          </Box>
        </Grid>
      );
    });
  }

  return (
    <Card>
      <CardContent sx={{ padding: 0, paddingBottom: '0 !important' }}>
        <Box bgcolor={bgColor ? bgColor : theme.palette.primary.main} p={3}>
          <Grid container direction="column" spacing={1}>
            <Grid item container justifyContent="space-between" alignItems="center">
              {title && (
                <Grid item>
                  <Typography variant="subtitle1" color="#fff">
                    {title}
                  </Typography>
                </Grid>
              )}
              <Grid item>
                <Grid container alignItems="center">
                  {icon && (
                    <Box component="span" mr={2} color="#fff">
                      {icon}
                    </Box>
                  )}
                  {percentage && (
                    <Typography variant="subtitle1" color="#fff">
                      {percentage}
                    </Typography>
                  )}
                </Grid>
              </Grid>
            </Grid>
            {chartData && (
              <Grid item>
                <Chart {...chartData} />
              </Grid>
            )}
          </Grid>
        </Box>
        {footerData && (
          <Grid container justifyContent="space-around" alignItems="center">
            {footerHtml}
          </Grid>
        )}
      </CardContent>
    </Card>
  );
};

const StatusDistributionChart = ({ tasks }) => {
  // Calculate task status distribution
  const statusData = [
    { name: 'Completed', value: tasks.filter((t) => t.status === 'Completed').length, color: '#4CAF50' },
    { name: 'In Progress', value: tasks.filter((t) => t.status === 'InProgress').length, color: '#2196F3' },
    { name: 'Pending', value: tasks.filter((t) => t.status === 'Pending').length, color: '#FF9800' },
    { name: 'To Do', value: tasks.filter((t) => t.status === 'ToDo').length, color: '#9C27B0' }
  ].filter((item) => item.value > 0);

  const CustomTooltip = ({ active, payload }) => {
    if (active && payload && payload.length) {
      return (
        <Box
          sx={{
            backgroundColor: 'background.paper',
            p: 1,
            border: 1,
            borderColor: 'divider',
            borderRadius: 1
          }}
        >
          <Typography variant="body2">{`${payload[0].name}: ${payload[0].value} tasks`}</Typography>
        </Box>
      );
    }
    return null;
  };

  return (
    <Card>
      <CardContent>
        <Typography variant="h6" gutterBottom>
          Task Status Distribution
        </Typography>

        {statusData.length > 0 ? (
          <ResponsiveContainer width="100%" height={300}>
            <PieChart>
              <Pie
                data={statusData}
                cx="50%"
                cy="50%"
                labelLine={false}
                label={({ name, percent }) => `${name} (${(percent * 100).toFixed(0)}%)`}
                outerRadius={80}
                fill="#8884d8"
                dataKey="value"
              >
                {statusData.map((entry, index) => (
                  <Cell key={`cell-${index}`} fill={entry.color} />
                ))}
              </Pie>
              <Tooltip content={<CustomTooltip />} />
              <Legend />
            </PieChart>
          </ResponsiveContainer>
        ) : (
          <Box display="flex" justifyContent="center" alignItems="center" height={200}>
            <Typography color="textSecondary">No task data available</Typography>
          </Box>
        )}
      </CardContent>
    </Card>
  );
};

        data: apiResponse,
        isLoading,
        error,
        refetch
    } = useGetNotificationHistoryQuery({
        page: page + 1,
        pageSize: rowsPerPage,
        category: categoryFilter !== 'all' ? categoryFilter : undefined,
        fromDate: dateFilter ? dayjs(dateFilter).format('YYYY-MM-DD') : undefined,
    });

    const notifications = apiResponse?.data || [];
    const pagination = apiResponse?.pagination || {};
    const totalCount = pagination.totalCount || 0;

        if (error) {
            console.error('API Error:', error);

    };

    const handleViewDetails = (notification) => {
        setViewDialog(notification);
    };

        if (window.confirm('Are you sure you want to delete this archived notification?')) {
            try {
                await deleteNotification(historyID).unwrap();
                setSnackbar({
                    open: true,
                    message: 'Archived notification deleted successfully',

                });
                refetch();
            } catch (error) {

            (notification.historyID?.toString() || '').includes(searchTerm) ||
            (notification.ruleID?.toString() || '').includes(searchTerm)
        );

const CronConverter = ({ value, onChange, disabled = false, label = "Schedule Configuration" }) => {
    const [selectedDate, setSelectedDate] = useState(new Date());
    const [frequency, setFrequency] = useState('daily');
    const [customInterval, setCustomInterval] = useState(1);
    const [intervalUnit, setIntervalUnit] = useState('hours');
    const [generatedCron, setGeneratedCron] = useState(value || '0 9 * * *');
    const [copied, setCopied] = useState(false);
    const [showAdvanced, setShowAdvanced] = useState(false);
    const [manualCron, setManualCron] = useState(value || '');
    const [validationError, setValidationError] = useState('');

    const frequencies = [
        { value: 'once', label: 'Once at specific time', description: 'Runs only once at the specified date and time' },
        { value: 'daily', label: 'Daily', description: 'Runs every day at the specified time' },
        { value: 'weekly', label: 'Weekly', description: 'Runs every week on the selected day and time' },
        { value: 'monthly', label: 'Monthly', description: 'Runs every month on the selected day and time' },
        { value: 'yearly', label: 'Yearly', description: 'Runs every year on the selected date and time' },
        { value: 'hourly', label: 'Hourly', description: 'Runs every hour at the specified minute' },
        { value: 'every_minute', label: 'Every Minute', description: 'Runs every minute' },
        { value: 'custom', label: 'Custom Interval', description: 'Custom interval (every X minutes/hours/days)' },
    ];

    const presets = [
        { label: 'Daily at 9 AM', cron: '0 9 * * *' },
        { label: 'Daily at midnight', cron: '0 0 * * *' },
        { label: 'Every Monday at 9 AM', cron: '0 9 * * 1' },
        { label: '1st of every month', cron: '0 0 1 * *' },
        { label: 'Every hour', cron: '0 * * * *' },
        { label: 'Every 30 minutes', cron: '*/30 * * * *' },
        { label: 'Every 2 hours', cron: '0 */2 * * *' },
        { label: 'Weekdays at 9 AM', cron: '0 9 * * 1-5' },
        { label: 'Weekends at 10 AM', cron: '0 10 * * 6,0' },
        { label: 'Monthly on 15th at 2 PM', cron: '0 14 15 * *' },
    ];

    // Initialize with current value
    useEffect(() => {
        if (value && isValidCron(value)) {
            setGeneratedCron(value);
            setManualCron(value);
            // Try to guess frequency from cron
            guessFrequencyFromCron(value);
        }
    }, [value]);

    const guessFrequencyFromCron = (cron) => {
        const [minute, hour, day, month, dayOfWeek] = cron.split(' ');

        if (cron === '* * * * *') {
            setFrequency('every_minute');
        } else if (minute === '0' && hour === '0' && day === '1' && month === '*') {
            setFrequency('monthly');
        } else if (minute === '0' && hour === '0' && day === '1' && month === '1') {
            setFrequency('yearly');
        } else if (dayOfWeek !== '*' && day === '*') {
            setFrequency('weekly');
        } else if (minute !== '*' && hour !== '*' && day === '*' && month === '*') {
            setFrequency('daily');
        } else if (hour.startsWith('*/') && minute === '0') {
            setFrequency('custom');
            const interval = hour.substring(2);
            setCustomInterval(parseInt(interval));
            setIntervalUnit('hours');
        } else if (minute.startsWith('*/')) {
            setFrequency('custom');
            const interval = minute.substring(2);
            setCustomInterval(parseInt(interval));
            setIntervalUnit('minutes');
        }
    };

    const generateCron = () => {
        try {
            const date = selectedDate;
            let cron;

            if (frequency === 'custom') {
                cron = dateToCron(date, 'custom', {
                    interval: customInterval,
                    unit: intervalUnit
                });
            } else {
                cron = dateToCron(date, frequency);
            }

            setGeneratedCron(cron);
            setManualCron(cron);
            setValidationError('');

            if (onChange) {
                onChange(cron);
            }
        } catch (error) {
            console.error('Error generating cron:', error);
            setValidationError('Failed to generate cron expression');
        }
    };

    const handlePresetClick = (cron) => {
        setGeneratedCron(cron);
        setManualCron(cron);
        setValidationError('');
        guessFrequencyFromCron(cron);

        if (onChange) {
            onChange(cron);
        }
    };

    const handleManualCronChange = (e) => {
        const cron = e.target.value;
        setManualCron(cron);

        if (cron && isValidCron(cron)) {
            setGeneratedCron(cron);
            setValidationError('');
            guessFrequencyFromCron(cron);

            if (onChange) {
                onChange(cron);
            }
        } else if (cron) {
            setValidationError('Invalid cron format. Must be 5 parts: minute hour day month day-of-week');
        } else {
            setValidationError('');
        }
    };

    const copyToClipboard = () => {
        navigator.clipboard.writeText(generatedCron).then(() => {
            setCopied(true);
            setTimeout(() => setCopied(false), 2000);
        });
    };

    const formatDateTime = () => {
        return dayjs(selectedDate).format('dddd, MMMM D, YYYY [at] h:mm A');
    };

    const getSchedulePreviews = () => {
        if (!isValidCron(generatedCron)) return [];
        return generateSchedulePreview(generatedCron, 3);
    };

    const schedulePreviews = getSchedulePreviews();
    const nextRunTime = getNextRunTime(generatedCron);
    const humanReadable = parseCronToText(generatedCron);

    return (
        <Card variant="outlined" sx={{ mb: 3 }}>
            <CardContent>
                <Box display="flex" alignItems="center" gap={1} mb={2}>
                    <ScheduleIcon color="primary" />
                    <Typography variant="h6">{label}</Typography>
                </Box>

                <Grid container spacing={3}>
                    {/* Left Column - Configuration */}
                    <Grid item xs={12} md={6}>
                        <Paper sx={{ p: 2, height: '100%', bgcolor: 'grey.50' }}>
                            <Typography variant="subtitle2" gutterBottom color="text.secondary">
                                Configure Schedule
                            </Typography>

                            {/* Frequency Selection */}
                            <FormControl fullWidth sx={{ mb: 3 }}>
                                <InputLabel>Frequency</InputLabel>
                                <Select
                                    value={frequency}
                                    label="Frequency"
                                    onChange={(e) => setFrequency(e.target.value)}
                                    disabled={disabled}
                                >
                                    {frequencies.map((freq) => (
                                        <MenuItem key={freq.value} value={freq.value}>
                                            <Box>
                                                <Typography>{freq.label}</Typography>
                                                <Typography variant="caption" color="text.secondary">
                                                    {freq.description}
                                                </Typography>
                                            </Box>
                                        </MenuItem>
                                    ))}
                                </Select>
                            </FormControl>

                            {/* Date Time Picker */}
                            <Box sx={{ mb: 3 }}>
                                <Typography variant="body2" gutterBottom>
                                    Select Date & Time
                                </Typography>
                                <StyledDatePicker
                                    selected={selectedDate}
                                    onChange={(date) => setSelectedDate(date)}
                                    showTimeSelect
                                    timeFormat="HH:mm"
                                    timeIntervals={15}
                                    dateFormat="MMMM d, yyyy h:mm aa"
                                    disabled={disabled || frequency === 'every_minute'}
                                    customInput={
                                        <TextField
                                            fullWidth
                                            InputProps={{
                                                startAdornment: (
                                                    <InputAdornment position="start">
                                                        <DateIcon />
                                                    </InputAdornment>
                                                ),
                                            }}
                                        />
                                    }
                                />
                            </Box>

                            {/* Custom Interval Options */}
                            {frequency === 'custom' && (
                                <Box sx={{ mb: 3 }}>
                                    <Typography variant="body2" gutterBottom>
                                        Custom Interval
                                    </Typography>
                                    <Grid container spacing={2}>
                                        <Grid item xs={6}>
                                            <TextField
                                                fullWidth
                                                label="Interval"
                                                type="number"
                                                value={customInterval}
                                                onChange={(e) => setCustomInterval(e.target.value)}
                                                InputProps={{ inputProps: { min: 1, max: 59 } }}
                                            />
                                        </Grid>
                                        <Grid item xs={6}>
                                            <FormControl fullWidth>
                                                <InputLabel>Unit</InputLabel>
                                                <Select
                                                    value={intervalUnit}
                                                    label="Unit"
                                                    onChange={(e) => setIntervalUnit(e.target.value)}
                                                >
                                                    <MenuItem value="minutes">Minutes</MenuItem>
                                                    <MenuItem value="hours">Hours</MenuItem>
                                                    <MenuItem value="days">Days</MenuItem>
                                                </Select>
                                            </FormControl>
                                        </Grid>
                                    </Grid>
                                </Box>
                            )}

                            {/* Generate Button */}
                            <Button
                                variant="contained"
                                fullWidth
                                onClick={generateCron}
                                disabled={disabled}
                                startIcon={<RefreshIcon />}
                                sx={{ mt: 2 }}
                            >
                                Generate Cron Expression
                            </Button>

                            {/* Quick Presets */}
                            <Box sx={{ mt: 3 }}>
                                <Typography variant="body2" gutterBottom color="text.secondary">
                                    Quick Presets:
                                </Typography>
                                <Box display="flex" flexWrap="wrap" gap={1}>
                                    {presets.map((preset) => (
                                        <Chip
                                            key={preset.cron}
                                            label={preset.label}
                                            size="small"
                                            onClick={() => handlePresetClick(preset.cron)}
                                            variant="outlined"
                                            sx={{ cursor: 'pointer', mb: 0.5 }}
                                        />
                                    ))}
                                </Box>
                            </Box>
                        </Paper>
                    </Grid>

                    {/* Right Column - Results */}
                    <Grid item xs={12} md={6}>
                        <Paper sx={{ p: 2, height: '100%' }}>
                            <Typography variant="subtitle2" gutterBottom color="text.secondary">
                                Generated Cron Expression
                            </Typography>

                            {/* Cron Display */}
                            <Paper
                                sx={{
                                    p: 2,
                                    bgcolor: 'primary.50',
                                    mb: 2,
                                    border: '1px solid',
                                    borderColor: 'primary.100',
                                    borderRadius: 1,
                                    position: 'relative',
                                }}
                            >
                                <Box display="flex" alignItems="center" justifyContent="space-between">
                                    <Typography
                                        variant="h6"
                                        fontFamily="monospace"
                                        color="primary.main"
                                        sx={{ wordBreak: 'break-all' }}
                                    >
                                        {generatedCron}
                                    </Typography>
                                    <Tooltip title={copied ? "Copied!" : "Copy to clipboard"}>
                                        <IconButton
                                            size="small"
                                            onClick={copyToClipboard}

                                        Next run: {dayjs(nextRunTime).format('dddd, MMMM D, YYYY [at] h:mm A')}

                            {/* Schedule Previews */}
                            {schedulePreviews.length > 0 && (
                                <Box sx={{ mb: 2 }}>
                                    <Typography variant="caption" color="text.secondary" display="block" gutterBottom>
                                        Upcoming runs:
                                    </Typography>
                                    <Stack spacing={0.5}>
                                        {schedulePreviews.map((time, index) => (
                                            <Box

                        </Paper>
                    </Grid>
                </Grid>

                {/* Cron Explanation */}

                        <strong>Cron Format:</strong> minute(0-59) hour(0-23) day-of-month(1-31) month(1-12) day-of-week(0-6, 0=Sunday)
                    </Typography>
                    <Typography variant="caption" color="text.secondary">
                        * = any value, */n = every n units, 1-5 = range, 1,3,5 = list
                    </Typography>
                </Box>
            </CardContent>
        </Card>
    );
};

                        </Typography>
                    </Box>
                    <Typography variant="body2" color="text.secondary">
                        This will trigger the rule immediately and send notifications to all configured recipients.
                    </Typography>
                </DialogContent>
                <DialogActions>
                    <Button onClick={() => setExecuteDialog(null)}>Cancel</Button>
                    <Button onClick={handleExecute} color="primary" variant="contained" autoFocus>
                        Execute Now
                    </Button>
                </DialogActions>
            </Dialog>
        </Box>
    );
};

                        <Typography color="textPrimary">{new Date(user.createdAt).toLocaleDateString()}</Typography>
                      </TableCell>
                      <TableCell align="center" sx={tableCellStyle}>
                        <Box display="flex" justifyContent="center" gap={1}>
                          <Tooltip title="Edit User">
                            <IconButton
                              size="small"
                              color="primary"
                              onClick={() => openEditUserDialog(user)}
                              disabled={updatingUser}
                              sx={{
                                color: theme.palette.primary.main,
                                '&:hover': {
                                  backgroundColor: theme.palette.primary.light,
                                  color: '#ffffff'
                                }
                              }}
                            >
                              <EditIcon />
                            </IconButton>
                          </Tooltip>
                          <Tooltip title="Reset Password">
                            <IconButton
                              size="small"
                              color="secondary"
                              onClick={() => handleOpenResetPasswordDialog(user)}
                              disabled={resettingPassword}
                              sx={{
                                color: theme.palette.secondary.main,
                                '&:hover': {
                                  backgroundColor: theme.palette.secondary.light,
                                  color: '#ffffff'
                                }
                              }}
                            >
                              <LockResetIcon />
                            </IconButton>
                          </Tooltip>
                          <Tooltip title="Delete User">
                            <IconButton
                              size="small"
                              color="error"
                              onClick={() => handleDeleteUser(user)}
                              disabled={user.userType === 'Admin' || deletingUser}
                              sx={{
                                color: theme.palette.error.main,
                                '&:hover': {
                                  backgroundColor: theme.palette.error.light,
                                  color: '#ffffff'
                                }
                              }}
                            >
                              <DeleteIcon />
                            </IconButton>
                          </Tooltip>
                        </Box>
                      </TableCell>
                    </TableRow>
                  ))
                )}
              </TableBody>
            </Table>
          </TableContainer>
        </CardContent>
      </Card>

      {/* Create User Dialog */}
      <Dialog
        open={openCreateDialog}
        onClose={() => {
          setOpenCreateDialog(false);
          resetCreateForm();
        }}
        maxWidth="md"
        fullWidth
        PaperProps={{
          sx: {
            borderRadius: '10px'
          }
        }}
      >
        <DialogTitle>
          <Typography variant="h6" color="textDark" fontWeight={600}>
            Create New User
          </Typography>
        </DialogTitle>
        <DialogContent>
          <Grid container spacing={2} sx={{ mt: 1 }}>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Username *"
                value={createForm.username}
                onChange={(e) => setCreateForm((prev) => ({ ...prev, username: e.target.value }))}
                error={!!formErrors.create.username}
                helperText={formErrors.create.username}
                onBlur={() => validateCreateForm()}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>User Type *</InputLabel>
                <Select
                  value={createForm.userType}
                  label="User Type *"
                  onChange={(e) => setCreateForm((prev) => ({ ...prev, userType: e.target.value }))}
                >
                  <MenuItem value="User">User</MenuItem>
                  <MenuItem value="Manager">Manager</MenuItem>
                  <MenuItem value="Admin">Admin</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12}>
              <TextField
                fullWidth
                label="Full Name *"
                value={createForm.fullName}
                onChange={(e) => setCreateForm((prev) => ({ ...prev, fullName: e.target.value }))}
                error={!!formErrors.create.fullName}
                helperText={formErrors.create.fullName}
                onBlur={() => validateCreateForm()}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Email"
                type="email"
                value={createForm.email}
                onChange={(e) => setCreateForm((prev) => ({ ...prev, email: e.target.value }))}
                error={!!formErrors.create.email}
                helperText={formErrors.create.email || 'Only Gmail addresses are allowed (optional)'}
                onBlur={() => validateCreateForm()}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Phone"
                value={createForm.phone}
                onChange={(e) => setCreateForm((prev) => ({ ...prev, phone: e.target.value }))}
                error={!!formErrors.create.phone}
                helperText={formErrors.create.phone || 'Optional - Enter digits only (e.g., 1234567890)'}
                onBlur={() => validateCreateForm()}
              />
            </Grid>
            {/* Project Selection Dropdown */}
            <Grid item xs={12}>
              <FormControl fullWidth>
                <InputLabel>Assign to Project (Optional)</InputLabel>
                <Select
                  value={createForm.projectName}
                  label="Assign to Project (Optional)"
                  onChange={(e) => setCreateForm((prev) => ({ ...prev, projectName: e.target.value }))}
                >
                  <MenuItem value="">No Project</MenuItem>
                  {projectsLoading ? (
                    <MenuItem disabled>Loading projects...</MenuItem>
                  ) : projects.map((project) => (
                    <MenuItem key={project.id} value={project.name}>
                      {project.name}
                    </MenuItem>
                  ))}
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Password *"
                type={showPassword ? 'text' : 'password'}
                value={createForm.password}
                onChange={(e) => setCreateForm((prev) => ({ ...prev, password: e.target.value }))}
                error={!!formErrors.create.password}
                helperText={formErrors.create.password}
                InputProps={{
                  endAdornment: (
                    <IconButton onClick={() => setShowPassword(!showPassword)} edge="end">
                      {showPassword ? <VisibilityOffIcon /> : <VisibilityIcon />}
                    </IconButton>
                  )
                }}
                onBlur={() => validateCreateForm()}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Confirm Password *"
                type={showConfirmPassword ? 'text' : 'password'}
                value={createForm.confirmPassword}
                onChange={(e) => setCreateForm((prev) => ({ ...prev, confirmPassword: e.target.value }))}
                error={!!formErrors.create.confirmPassword}
                helperText={formErrors.create.confirmPassword}
                InputProps={{
                  endAdornment: (
                    <IconButton onClick={() => setShowConfirmPassword(!showConfirmPassword)} edge="end">
                      {showConfirmPassword ? <VisibilityOffIcon /> : <VisibilityIcon />}
                    </IconButton>
                  )
                }}
                onBlur={() => validateCreateForm()}
              />
            </Grid>
          </Grid>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => {
            setOpenCreateDialog(false);
            resetCreateForm();
          }}>
            Cancel
          </Button>
          <Button
            onClick={handleCreateUser}
            variant="contained"
            disabled={creatingUser || isLoading || projectsLoading}
            sx={{
              backgroundColor: theme.palette.primary.main,
              '&:hover': {
                backgroundColor: theme.palette.primary.dark
              }
            }}
          >
            {creatingUser ? <CircularProgress size={24} /> : 'Create User'}
          </Button>
        </DialogActions>
      </Dialog>

      {/* Edit User Dialog */}
      <Dialog
        open={openEditDialog}
        onClose={() => {
          setOpenEditDialog(false);
          setOriginalEditForm(null);
        }}
        maxWidth="md"
        fullWidth
        PaperProps={{
          sx: {
            borderRadius: '10px'
          }
        }}
      >
        <DialogTitle>
          <Typography variant="h6" color="textDark" fontWeight={600}>
            Edit User - {selectedUser?.username}
          </Typography>
        </DialogTitle>
        <DialogContent>
          <Grid container spacing={2} sx={{ mt: 1 }}>
            <Grid item xs={12}>
              <TextField 
                fullWidth 
                label="Username" 
                value={selectedUser?.username || ''} 
                disabled 
                helperText="Username cannot be changed" 
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>User Type</InputLabel>
                <Select
                  value={editForm.userType}
                  label="User Type"
                  onChange={(e) => setEditForm((prev) => ({ ...prev, userType: e.target.value }))}
                >
                  <MenuItem value="User">User</MenuItem>
                  <MenuItem value="Manager">Manager</MenuItem>
                  <MenuItem value="Admin">Admin</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12} sm={6}>
              <FormControl fullWidth>
                <InputLabel>Status</InputLabel>
                <Select
                  value={editForm.isActive}
                  label="Status"
                  onChange={(e) => setEditForm((prev) => ({ ...prev, isActive: e.target.value }))}
                >
                  <MenuItem value={true}>Active</MenuItem>
                  <MenuItem value={false}>Inactive</MenuItem>
                </Select>
              </FormControl>
            </Grid>
            <Grid item xs={12}>
              <TextField
                fullWidth
                label="Full Name *"
                value={editForm.fullName}
                onChange={(e) => setEditForm((prev) => ({ ...prev, fullName: e.target.value }))}
                error={!!formErrors.edit.fullName}
                helperText={formErrors.edit.fullName}
                onBlur={() => validateEditForm()}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Email"
                type="email"
                value={editForm.email}
                onChange={(e) => setEditForm((prev) => ({ ...prev, email: e.target.value }))}
                error={!!formErrors.edit.email}
                helperText={formErrors.edit.email || 'Only Gmail addresses are allowed (optional)'}
                onBlur={() => validateEditForm()}
              />
            </Grid>
            <Grid item xs={12} sm={6}>
              <TextField
                fullWidth
                label="Phone"
                value={editForm.phone}
                onChange={(e) => setEditForm((prev) => ({ ...prev, phone: e.target.value }))}
                error={!!formErrors.edit.phone}
                helperText={formErrors.edit.phone || 'Optional - Enter digits only (e.g., 1234567890)'}
                onBlur={() => validateEditForm()}
              />
            </Grid>
            {/* Project Selection Dropdown for Edit */}
            <Grid item xs={12}>
              <FormControl fullWidth>
                <InputLabel>Assign to Project (Optional)</InputLabel>
                <Select
                  value={editForm.projectName}
                  label="Assign to Project (Optional)"
                  onChange={(e) => setEditForm((prev) => ({ ...prev, projectName: e.target.value }))}
                >
                  <MenuItem value="">No Project</MenuItem>
                  {projectsLoading ? (
                    <MenuItem disabled>Loading projects...</MenuItem>
                  ) : projects.map((project) => (
                    <MenuItem key={project.id} value={project.name}>
                      {project.name}
                    </MenuItem>
                  ))}
                </Select>
              </FormControl>
            </Grid>
          </Grid>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => {
            setOpenEditDialog(false);
            setOriginalEditForm(null);
          }}>
            Cancel
          </Button>
          <Button
            onClick={handleEditUser}
            variant="contained"
            disabled={updatingUser || isLoading || projectsLoading}
            sx={{
              backgroundColor: theme.palette.primary.main,
              '&:hover': {
                backgroundColor: theme.palette.primary.dark
              }
            }}
          >
            {updatingUser ? <CircularProgress size={24} /> : 'Update User'}
          </Button>
        </DialogActions>
      </Dialog>

      {/* Admin Reset Password Dialog */}
      <Dialog
        open={openResetPasswordDialog}
        onClose={() => {
          setOpenResetPasswordDialog(false);
          resetResetPasswordForm();
        }}
        maxWidth="sm"
        fullWidth
        PaperProps={{
          sx: {
            borderRadius: '10px'
          }
        }}
      >
        <DialogTitle>
          <Typography variant="h6" color="textDark" fontWeight={600}>
            Reset Password for {selectedUser?.username}
          </Typography>
        </DialogTitle>
        <DialogContent>
          <Alert severity="info" sx={{ mb: 2 }}>
            As an administrator, you can reset this user's password without knowing their current password.
          </Alert>
          <Grid container spacing={2} sx={{ mt: 1 }}>
            <Grid item xs={12}>
              <TextField
                fullWidth
                label="New Password *"
                type="password"
                value={resetPasswordForm.newPassword}
                onChange={(e) => setResetPasswordForm((prev) => ({ ...prev, newPassword: e.target.value }))}
                error={!!formErrors.resetPassword.newPassword}
                helperText={formErrors.resetPassword.newPassword || 'Password must be at least 6 characters long'}
                onBlur={() => validateResetPasswordForm()}
              />
            </Grid>
            <Grid item xs={12}>
              <TextField
                fullWidth
                label="Confirm New Password *"
                type="password"
                value={resetPasswordForm.confirmNewPassword}
                onChange={(e) => setResetPasswordForm((prev) => ({ ...prev, confirmNewPassword: e.target.value }))}
                error={!!formErrors.resetPassword.confirmNewPassword}
                helperText={formErrors.resetPassword.confirmNewPassword}
                onBlur={() => validateResetPasswordForm()}
              />
            </Grid>
          </Grid>
        </DialogContent>
        <DialogActions>
          <Button onClick={() => {
            setOpenResetPasswordDialog(false);
            resetResetPasswordForm();
          }}>
            Cancel
          </Button>
          <Button
            onClick={handleResetPassword}
            variant="contained"
            disabled={resettingPassword || isLoading}
            sx={{
              backgroundColor: theme.palette.primary.main,
              '&:hover': {
                backgroundColor: theme.palette.primary.dark
              }
            }}
          >
            {resettingPassword ? <CircularProgress size={24} /> : 'Reset Password'}
          </Button>
        </DialogActions>
      </Dialog>

      {/* Snackbar for notifications */}
      <Snackbar open={snackbar.open} autoHideDuration={6000} onClose={() => setSnackbar((prev) => ({ ...prev, open: false }))}>
        <Alert onClose={() => setSnackbar((prev) => ({ ...prev, open: false }))} severity={snackbar.severity}>
          {snackbar.message}
        </Alert>
      </Snackbar>
    </Box>
  );
};

export default UserManagement;
