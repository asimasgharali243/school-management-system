export const getFileUrl = (filePath, download = false) => {
  if (!filePath) return '#';

  console.log('📁 Original file path:', filePath);

  // Case 1: If it already starts with http:// or https://
  if (filePath.startsWith('http://') || filePath.startsWith('https://')) {
    return download ? `${filePath}?download=true` : filePath;
  }

  // Case 2: If it starts with /uploads/ (like your data shows)
  if (filePath.startsWith('/uploads/')) {
    return download
      ? `${API_BASE_URL_FILE}${filePath}?download=true`
      : `${API_BASE_URL_FILE}${filePath}`;
  }

  // Case 3: If it's just a filename without path
  const cleanFileName = filePath.replace(/^\/+/, '');

  // Default to tasks directory
  return download
    ? `${API_BASE_URL_FILE}/uploads/tasks/${cleanFileName}?download=true`
    : `${API_BASE_URL_FILE}/uploads/tasks/${cleanFileName}`;
};

export const showErrorToast = (error) => {
  let errorMessage = 'An error occurred';

  if (typeof error === 'string') {
    errorMessage = error;
  } else if (error?.data?.message) {
    errorMessage = error.data.message;
  } else if (error?.message) {
    errorMessage = error.message;
  } else if (error?.error) {
    errorMessage = error.error;
  }

  if (typeof window !== 'undefined' && window.showToast) {
    window.showToast({ type: 'error', message: errorMessage });
  } else {
    console.error('❌ Error:', errorMessage);
    alert(`❌ ${errorMessage}`);
  }
};

// ✅ Determine if file can be opened in browser
export const canOpenInBrowser = (fileName) => {
  if (!fileName) return false;

  const extension = fileName.split('.').pop().toLowerCase();
  const viewableExtensions = [
    // Images
    'jpg', 'jpeg', 'png', 'gif', 'bmp', 'webp', 'svg',
    // Documents
    'pdf',
    // Text
    'txt', 'csv', 'json', 'xml', 'html', 'htm',
    // Audio
    'mp3', 'wav', 'm4a', 'aac', 'ogg',
    // Video
    'mp4', 'webm', 'ogg'
  ];

  return viewableExtensions.includes(extension);
};

// ✅ Get file URL based on type
export const getSmartFileUrl = (filePath) => {
  const fileName = filePath.split('/').pop();

  if (canOpenInBrowser(fileName)) {
    // For viewable files, return direct URL
    return getFileUrl(filePath, false);
  } else {
    // For non-viewable files, force download
    return getFileUrl(filePath, true);
  }
};

// ✅ Get file size in readable format
export const formatFileSize = (bytes) => {
  if (bytes === 0) return '0 Bytes';

  const k = 1024;
  const sizes = ['Bytes', 'KB', 'MB', 'GB'];
  const i = Math.floor(Math.log(bytes) / Math.log(k));

  return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
};

import { createApi, fetchBaseQuery } from '@reduxjs/toolkit/query/react';
import { API_BASE_URL, AUTH_ENDPOINTS } from './apiConfig';

export const authApi = createApi({
  reducerPath: 'authApi',
  baseQuery: fetchBaseQuery({
    baseUrl: API_BASE_URL,
    
    prepareHeaders: (headers) => {
      const token = localStorage.getItem('token');
      if (token) headers.set('authorization', `Bearer ${token}`);
      headers.set('Content-Type', 'application/json');
      return headers;
    },
    credentials: 'include'
  }),
  tagTypes: ['Auth'],
  endpoints: (builder) => ({
    login: builder.mutation({
      query: (credentials) => ({
        url: AUTH_ENDPOINTS.LOGIN,
        method: 'POST',
        body: credentials,
      }),
      invalidatesTags: ['Auth'],
    }),
    register: builder.mutation({
      query: (userData) => ({
        url: AUTH_ENDPOINTS.REGISTER,
        method: 'POST',
        body: userData,
      }),
      invalidatesTags: ['Auth'],
    }),
    logout: builder.mutation({
      query: () => ({
        url: AUTH_ENDPOINTS.LOGOUT,
        method: 'POST',
      }),
      invalidatesTags: ['Auth'],
    }),
    getProfile: builder.query({
      query: () => AUTH_ENDPOINTS.PROFILE,
      providesTags: ['Auth'],
    }),
    validateUsername: builder.mutation({
      query: (username) => ({
        url: AUTH_ENDPOINTS.VALIDATE_USERNAME,
        method: 'POST',
        body: { username },
      }),
    }),
  }),
});

export const {
  useLoginMutation,
  useRegisterMutation,
  useLogoutMutation,
  useGetProfileQuery,
  useValidateUsernameMutation,
} = authApi;

let isRefreshing = false;
let failedQueue = [];

const processQueue = (error, token = null) => {
  failedQueue.forEach(prom => {
    if (error) {
      prom.reject(error);
    } else {
      prom.resolve(token);
    }
  });
  failedQueue = [];
};

const baseQuery = fetchBaseQuery({
  baseUrl: API_BASE_URL,
  prepareHeaders: (headers, { getState }) => {
    const token = localStorage.getItem('token');
    if (token) {
      headers.set('authorization', `Bearer ${token}`);
    }
    headers.set('Content-Type', 'application/json');
    return headers;
  },
  credentials: 'include',
});

const baseQueryWithReauth = async (args, api, extraOptions) => {
  let result = await baseQuery(args, api, extraOptions);

  if (result.error && (result.error.status === 401 || result.error.originalStatus === 401)) {
    const token = localStorage.getItem('token');

    if (!token) {
      localStorage.removeItem('token');
      localStorage.removeItem('user');
      localStorage.removeItem('tokenExpiry');
      window.location.href = '/application/login';
      return result;
    }

    try {
      const decoded = jwtDecode(token);
      const currentTime = Date.now() / 1000;
      const timeUntilExpiry = decoded.exp - currentTime;

      if (timeUntilExpiry < 300) {
        if (!isRefreshing) {
          isRefreshing = true;

          try {
            const refreshResult = await baseQuery(
              {
                url: '/auth/refresh-token',
                method: 'POST',
                body: { token }
              },
              api,
              extraOptions
            );

            if (refreshResult.data && refreshResult.data.token) {
              const newToken = refreshResult.data.token;
              localStorage.setItem('token', newToken);

              const newDecoded = jwtDecode(newToken);
              localStorage.setItem('tokenExpiry', newDecoded.exp.toString());

              isRefreshing = false;
              processQueue(null, newToken);

              result = await baseQuery(args, api, extraOptions);
            } else {
              throw new Error('Token refresh failed');
            }
          } catch (error) {
            isRefreshing = false;
            processQueue(error, null);
            localStorage.removeItem('token');
            localStorage.removeItem('user');
            localStorage.removeItem('tokenExpiry');
            window.location.href = '/application/login';
            return result;
          }
        } else {
          return new Promise((resolve, reject) => {
            failedQueue.push({ resolve, reject });
          }).then(() => {
            return baseQuery(args, api, extraOptions);
          }).catch(() => {
            return result;
          });
        }
      }
    } catch (error) {
      console.error('Token decode error:', error);
      localStorage.removeItem('token');
      localStorage.removeItem('user');
      localStorage.removeItem('tokenExpiry');
      window.location.href = '/application/login';
    }
  }

  return result;
};

      firstName: userData.firstName,
      lastName: userData.lastName,
      email: userData.email,
      username: userData.username || userData.email,
      password: '***' // Don't log actual password
    });

    const response = await fetch(`${API_BASE_URL}/api/Auth/register`, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'Accept': 'application/json',
      },
      body: JSON.stringify(userData),
    });

    console.log('🔍 DEBUG - Response status:', response.status);
    console.log('🔍 DEBUG - Response ok:', response.ok);

    // Check if response is OK
    if (!response.ok) {
      const errorText = await response.text();
      console.error('🔍 DEBUG - Response error text:', errorText);

      let errorMessage = `Registration failed: ${response.status}`;
      try {
        const errorData = JSON.parse(errorText);
        errorMessage = errorData.message || errorData.title || errorMessage;
      } catch (e) {
        // If not JSON, use text
        errorMessage = errorText || errorMessage;
      }

      return rejectWithValue(errorMessage);
    }

    // Parse successful response
    const data = await response.json();
    console.log('🔍 DEBUG - Registration successful:', {
      token: data.token ? 'Token received' : 'No token',
      user: data.user ? 'User data received' : 'No user data'
    });

    // Store data if auto-login after registration
    if (data.token) {
      localStorage.setItem('token', data.token);
    }

    if (data.user) {
      localStorage.setItem('user', JSON.stringify(data.user));
    }

    // Store token expiry if token exists
    if (data.token) {
      try {
        const decoded = jwtDecode(data.token);
        localStorage.setItem('tokenExpiry', decoded.exp.toString());
        console.log('🔍 DEBUG - Token expiry set:', new Date(decoded.exp * 1000));
      } catch (error) {
        console.error('🔍 DEBUG - Error decoding token:', error);
      }
    }

    return data;
  } catch (error) {
    console.error('🔍 DEBUG - Network/connection error during registration:', error);

    // Specific error handling
    if (error.name === 'TypeError' && error.message.includes('Failed to fetch')) {
      return rejectWithValue('Cannot connect to server. Please check if backend is running');
    }

    if (error.message.includes('CORS')) {
      return rejectWithValue('CORS error: Check backend CORS configuration');
    }

    return rejectWithValue(error.message || 'Network error occurred during registration');
  }
});

const authSlice = createSlice({
  name: 'auth',
  initialState,
  reducers: {
    clearError: (state) => {
      state.error = null;
    },
    forceLogout: (state) => {
      state.user = null;
      state.token = null;
      state.isAuthenticated = false;
      state.tokenValid = false;
      localStorage.removeItem('token');
      localStorage.removeItem('user');
      localStorage.removeItem('tokenExpiry');
    },
    updateUser: (state, action) => {
      state.user = { ...state.user, ...action.payload };
      localStorage.setItem('user', JSON.stringify(state.user));
    },
  },
  extraReducers: (builder) => {
    builder
      // Login
      .addCase(loginUser.pending, (state) => {
        state.loading = true;
        state.error = null;
      })
      .addCase(loginUser.fulfilled, (state, action) => {
        state.loading = false;
        state.user = action.payload.user || null;
        state.token = action.payload.token || null;
        state.isAuthenticated = true;
        state.tokenValid = true;
        state.error = null;
      })
      .addCase(loginUser.rejected, (state, action) => {
        state.loading = false;
        state.error = action.payload;
        state.isAuthenticated = false;
        state.tokenValid = false;
      })
      // Validate token
      .addCase(validateToken.pending, (state) => {
        state.loading = true;
      })
      .addCase(validateToken.fulfilled, (state, action) => {
        state.loading = false;
        state.user = action.payload.user || state.user;
        state.isAuthenticated = true;
        state.tokenValid = true;
      })
      .addCase(validateToken.rejected, (state) => {
        state.loading = false;
        state.isAuthenticated = false;
        state.tokenValid = false;
      })
      // Logout
      .addCase(logoutUser.fulfilled, (state) => {
        state.user = null;
        state.token = null;
        state.isAuthenticated = false;
        state.tokenValid = false;
        state.error = null;
      });
  },
});

export const { clearError, forceLogout, updateUser } = authSlice.actions;
export default authSlice.reducer;
