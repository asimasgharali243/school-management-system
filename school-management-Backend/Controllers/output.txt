        // GET: api/adminnotification/logs
        [HttpGet("logs")]
        public async Task<IActionResult> GetNotificationLogs(
            [FromQuery] DateTime? fromDate = null,
            [FromQuery] DateTime? toDate = null,
            [FromQuery] string action = null,
            [FromQuery] int page = 1,
            [FromQuery] int pageSize = 50)
        {
            try
            {
                var query = _context.NotificationLogs.AsQueryable();

                if (fromDate.HasValue)
                    query = query.Where(l => l.Timestamp >= fromDate.Value);

                if (toDate.HasValue)
                    query = query.Where(l => l.Timestamp <= toDate.Value);

                if (!string.IsNullOrEmpty(action))
                    query = query.Where(l => l.Action == action);

                var totalCount = await query.CountAsync();
                var logs = await query
                    .OrderByDescending(l => l.Timestamp)
                    .Skip((page - 1) * pageSize)
                    .Take(pageSize)
                    .Select(l => new
                    {
                        l.LogID,
                        l.QueueID,
                        l.UserID,
                        l.Action,
                        l.Timestamp,
                        l.Details
                    })
                    .ToListAsync();

                return Ok(new
                {
                    Success = true,
                    Data = logs,
                    Pagination = new
                    {
                        Page = page,
                        PageSize = pageSize,
                        TotalCount = totalCount,
                        TotalPages = (int)Math.Ceiling(totalCount / (double)pageSize)
                    }
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error getting notification logs");
                return StatusCode(500, new { Success = false, Message = "Internal server error" });
            }
        }

[HttpGet("system-stats")]
public async Task<IActionResult> GetSystemStats([FromQuery] DateTime? fromDate = null, [FromQuery] DateTime? toDate = null)
{
    try
    {
        fromDate ??= DateTime.Now.AddDays(-30);
        toDate ??= DateTime.Now;

        // Rule statistics
        var ruleStats = await _context.NotificationRules
            .Select(r => new
            {
                r.RuleID,
                r.RuleName,
                r.RuleType,
                r.Category,
                r.IsActive,
                r.TotalRuns,
                r.LastRunTime,
                r.NextRunTime
            })
            .ToListAsync();

        // Notification statistics
        var notificationStats = await _context.NotificationQueue
            .Where(n => n.CreatedTime >= fromDate && n.CreatedTime <= toDate)
            .GroupBy(n => n.Status)
            .Select(g => new
            {
                Status = g.Key,
                Count = g.Count(),
                AverageAttempts = g.Average(n => n.AttemptCount)
            })
            .ToListAsync();

        var totalNotifications = notificationStats.Sum(s => s.Count);
        var sentNotifications = notificationStats.FirstOrDefault(s => s.Status == "SENT")?.Count ?? 0;

        // Calculate average attempts - SIMPLE AND WORKING
        double averageAttemptsValue = 0;
        if (totalNotifications > 0)
        {
            averageAttemptsValue = await _context.NotificationQueue
                .Where(n => n.CreatedTime >= fromDate && n.CreatedTime <= toDate)
                .AverageAsync(n => (double)n.AttemptCount);
        }

        // Device statistics
        var deviceStats = await _context.UserDevices
            .GroupBy(d => d.Platform)
            .Select(g => new
            {
                Platform = g.Key,
                Total = g.Count(),
                Active = g.Count(d => d.IsActive)
            })
            .ToListAsync();

        // History statistics
        var historyStats = await _context.NotificationHistory
            .Where(h => h.ArchivedDate >= fromDate && h.ArchivedDate <= toDate)
            .GroupBy(h => h.Category)
            .Select(g => new
            {
                Category = g.Key,
                Count = g.Count()
            })
            .ToListAsync();

        // Log statistics
        var logStats = await _context.NotificationLogs
            .Where(l => l.Timestamp >= fromDate && l.Timestamp <= toDate)
            .GroupBy(l => l.Action)
            .Select(g => new
            {
                Action = g.Key,
                Count = g.Count()
            })
            .ToListAsync();

        return Ok(new
        {
            Success = true,
            Data = new
            {
                Period = new { From = fromDate, To = toDate },
                Rules = new
                {
                    Total = ruleStats.Count,
                    Active = ruleStats.Count(r => r.IsActive),
                    ByType = ruleStats.GroupBy(r => r.RuleType)
                        .Select(g => new { Type = g.Key, Count = g.Count() }),
                    ByCategory = ruleStats.GroupBy(r => r.Category)
                        .Select(g => new { Category = g.Key, Count = g.Count() })
                },
                Notifications = new
                {
                    Total = totalNotifications,
                    ByStatus = notificationStats,
                    SuccessRate = totalNotifications > 0 ? (sentNotifications / (double)totalNotifications) * 100 : 0,
                    AverageAttempts = Math.Round(averageAttemptsValue, 2)
                },
                Devices = new
                {
                    Total = deviceStats.Sum(d => d.Total),
                    Active = deviceStats.Sum(d => d.Active),
                    ByPlatform = deviceStats
                },
                History = new
                {
                    Total = historyStats.Sum(h => h.Count),
                    ByCategory = historyStats
                },
                Logs = new
                {
                    Total = logStats.Sum(l => l.Count),
                    ByAction = logStats
                }
            }
        });
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error getting system stats");
        return StatusCode(500, new { Success = false, Message = "Internal server error" });
    }
}

        [HttpPost("broadcast")]
        public async Task<IActionResult> SendBroadcastNotification([FromBody] BroadcastNotificationVM model)
        {
            try
            {
                if (!ModelState.IsValid)
                    return BadRequest(new { Success = false, Errors = ModelState.Values.SelectMany(v => v.Errors) });

                if (string.IsNullOrEmpty(model.Title))
                    return BadRequest(new { Success = false, Message = "Title is required" });

                if (string.IsNullOrEmpty(model.Message))
                    return BadRequest(new { Success = false, Message = "Message is required" });

                // Get target users based on broadcast type
                var users = await GetBroadcastTargetUsersAsync(model);

                if (users.Count == 0)
                    return BadRequest(new { Success = false, Message = "No users found for the specified criteria" });

                // Create a batch notification record
                var batch = new BatchNotification
                {
                    RuleID = 0, // Manual broadcast
                    Title = model.Title,
                    Message = model.Message,
                    DataPayload = model.Data?.ToString(),
                    TargetType = model.TargetType,
                    TargetFilter = model.TargetFilter?.ToString(),
                    TotalRecipients = users.Count,
                    SentCount = 0,
                    FailedCount = 0,
                    Status = "PROCESSING",
                    CreatedTime = DateTime.Now
                };

                _context.BatchNotifications.Add(batch);
                await _context.SaveChangesAsync();

                // Create notifications for each user
                var notifications = new List<NotificationQueue>();
                foreach (var user in users)
                {
                    var notification = new NotificationQueue
                    {
                        RuleID = 0,
                        UserID = user.UserID,
                        UserType = user.UserType,
                        DeviceToken = user.ExpoToken,
                        Title = model.Title,
                        Message = model.Message,
                        DataPayload = model.Data?.ToString(),
                        Status = "PENDING",
                        Priority = model.Priority,
                        ScheduledTime = DateTime.Now,
                        CreatedTime = DateTime.Now,
                        AttemptCount = 0
                    };

                    notifications.Add(notification);
                }

                _context.NotificationQueue.AddRange(notifications);
                await _context.SaveChangesAsync();

                // Update batch status
                batch.Status = "PENDING";
                await _context.SaveChangesAsync();

                _logger.LogInformation("Created broadcast notification for {Count} users (Batch ID: {BatchId})", users.Count, batch.BatchID);

                return Ok(new
                {
                    Success = true,
                    Message = $"Broadcast notification created for {users.Count} users",
                    BatchId = batch.BatchID,
                    NotificationsCreated = notifications.Count
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error sending broadcast notification");
                return StatusCode(500, new { Success = false, Message = "Internal server error" });
            }
        }

        private async Task<List<BroadcastUser>> GetBroadcastTargetUsersAsync(BroadcastNotificationVM model)
        {
            var users = new List<BroadcastUser>();

            switch (model.TargetType?.ToUpper())
            {
                case "ALL":
                    var allUsers = await _context.UserDevices
                        .Where(d => d.IsActive && !string.IsNullOrEmpty(d.ExpoToken))
                        .Select(d => new BroadcastUser
                        {
                            UserID = d.UserID,
                            UserType = d.UserType,
                            ExpoToken = d.ExpoToken
                        })
                        .ToListAsync();
                    users.AddRange(allUsers);
                    break;

                case "PARENTS":
                    var parents = await _context.UserDevices
                        .Where(d => d.IsActive && !string.IsNullOrEmpty(d.ExpoToken) && d.UserType == "PARENT")
                        .Select(d => new BroadcastUser
                        {
                            UserID = d.UserID,
                            UserType = d.UserType,
                            ExpoToken = d.ExpoToken
                        })
                        .ToListAsync();
                    users.AddRange(parents);
                    break;

                case "TEACHERS":
                    var teachers = await _context.UserDevices
                        .Where(d => d.IsActive && !string.IsNullOrEmpty(d.ExpoToken) && d.UserType == "TEACHER")
                        .Select(d => new BroadcastUser
                        {
                            UserID = d.UserID,
                            UserType = d.UserType,
                            ExpoToken = d.ExpoToken
                        })
                        .ToListAsync();
                    users.AddRange(teachers);
                    break;

                case "ADMIN":
                    var admins = await _context.UserDevices
                        .Where(d => d.IsActive && !string.IsNullOrEmpty(d.ExpoToken) && d.UserType == "ADMIN")
                        .Select(d => new BroadcastUser
                        {
                            UserID = d.UserID,
                            UserType = d.UserType,
                            ExpoToken = d.ExpoToken
                        })
                        .ToListAsync();
                    users.AddRange(admins);
                    break;

                case "CLASS":
                    if (model.TargetFilter != null && model.TargetFilter.ContainsKey("ClassId"))
                    {
                        var classId = model.TargetFilter["ClassId"].Value<int>();
                        // This query depends on your actual database structure
                        // Adjust based on your actual student-parent relationships
                        var classUsers = await _context.UserDevices
                            .Where(d => d.IsActive && !string.IsNullOrEmpty(d.ExpoToken))
                            .Take(10) // Limit for demo - adjust based on your actual query
                            .Select(d => new BroadcastUser
                            {
                                UserID = d.UserID,
                                UserType = d.UserType,
                                ExpoToken = d.ExpoToken
                            })
                            .ToListAsync();
                        users.AddRange(classUsers);
                    }
                    break;

                default:
                    // Default to all active users with tokens
                    var defaultUsers = await _context.UserDevices
                        .Where(d => d.IsActive && !string.IsNullOrEmpty(d.ExpoToken))
                        .Select(d => new BroadcastUser
                        {
                            UserID = d.UserID,
                            UserType = d.UserType,
                            ExpoToken = d.ExpoToken
                        })
                        .ToListAsync();
                    users.AddRange(defaultUsers);
                    break;
            }

            return users.DistinctBy(u => u.UserID).ToList();
        }

        // GET: api/attendance/child/{childId}/stats
        [HttpGet("child/{childId}/stats")]
        public async Task<ActionResult<ApiResponse<AttendanceStatsDto>>> GetChildAttendanceStats(
            int childId,
            [FromQuery] AttendanceStatsFilters filters)
        {
            try
            {
                _logger.LogInformation($"Fetching attendance stats for child ID: {childId}");
                
                // Get student
                var student = await _context.tbl_StudentRegistration
                    .FirstOrDefaultAsync(s => s.Reg_Id == childId);

                if (student == null)
                {
                    return NotFound(ApiResponse<AttendanceStatsDto>.CreateError("Student not found"));
                }

                // Get attendance for the student
                var query = from detail in _context.tbl_StudentAttDetail
                            join master in _context.tbl_StudentAttMaster on detail.MasterId equals master.ID
                            where detail.RegID == childId && master.AttDate.HasValue
                            select new { detail, master };

                // Apply filters
                if (filters.DateFrom.HasValue)
                {
                    var dateFrom = DateOnly.FromDateTime(filters.DateFrom.Value);
                    query = query.Where(x => x.master.AttDate >= dateFrom);
                }

                if (filters.DateTo.HasValue)
                {
                    var dateTo = DateOnly.FromDateTime(filters.DateTo.Value);
                    query = query.Where(x => x.master.AttDate <= dateTo);
                }

                if (filters.Month.HasValue && filters.Year.HasValue)
                {
                    query = query.Where(x => x.master.AttDate.HasValue && 
                                             x.master.AttDate.Value.Month == filters.Month.Value && 
                                             x.master.AttDate.Value.Year == filters.Year.Value);
                }

                var attendanceRecords = await query.ToListAsync();
                var totalDays = attendanceRecords.Count;

                // Calculate stats
                var presentDays = attendanceRecords.Count(d => d.detail.Status == "P");
                var absentDays = attendanceRecords.Count(d => d.detail.Status == "A");
                var lateDays = attendanceRecords.Count(d => d.detail.LateStatus == "Late");
                var leaveDays = attendanceRecords.Count(d => d.detail.Status == "L");
                
                var presentPercentage = totalDays > 0 ? Math.Round((presentDays / (double)totalDays) * 100, 1) : 0;

                var stats = new AttendanceStatsDto
                {
                    TotalDays = totalDays,
                    PresentDays = presentDays,
                    AbsentDays = absentDays,
                    LateDays = lateDays,
                    LeaveDays = leaveDays,
                    PresentPercentage = presentPercentage,
                    AttendanceRate = presentPercentage
                };

                _logger.LogInformation($"Attendance stats calculated for child {childId}: Present {presentDays}/{totalDays}");
                
                return Ok(ApiResponse<AttendanceStatsDto>.CreateSuccess(stats));
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, $"Error fetching attendance stats for child {childId}: {ex.Message}");
                return StatusCode(500, ApiResponse<AttendanceStatsDto>.CreateError($"Error fetching attendance stats: {ex.Message}"));
            }
        }

            })
            .ToListAsync();

        var regIds = children.Select(x => x.Reg_Id).ToList();
        var now = DateTime.Now;

        // ==============================
        // 2. TEMPORARY ACCESS (FAST)
        // ==============================
        var tempAccessIds = await _context.tbl_DiaryAccessControl
            .Where(x =>
                regIds.Contains(x.RegID) &&
                x.IsActive &&
                now >= x.AllowFrom &&
                now <= x.AllowTo
            )
            .Select(x => x.RegID)
            .Distinct()
            .ToListAsync();

        // ==============================
        // 3. UNPAID FEE CHECK (CORRECT LOGIC)
        // ==============================
        var unpaidIds = await (
            from p in _context.tbl_FeePosting
            where regIds.Contains((int)p.Reg_ID)

            where !_context.tbl_FeeReceiveMaster
                .Where(m => m.RegID == p.Reg_ID)
                .Join(_context.tbl_FeeReceiveDetail,
                    m => m.Receipt_ID,
                    d => d.Receipt_ID,
                    (m, d) => new { m, d })
                .Any(x =>
                    x.d.FeeMonth == p.FeeMonth &&
                    x.d.FeeYear == p.FeeYear
                )

            select p.Reg_ID
        )
        .Distinct()
        .ToListAsync();

        // ==============================
        // 4. APPLY BUSINESS LOGIC (NO DB CALLS)
        // ==============================
        foreach (var child in children)
        {
            bool hasUnpaid = unpaidIds.Contains(child.Reg_Id);
            bool hasTempAccess = tempAccessIds.Contains(child.Reg_Id);

            // FINAL RULE
         child.DiaryAccess = hasTempAccess ? 1 : (hasUnpaid ? 0 : 1);
        }

        _logger.LogInformation($"Found {children.Count} children");

        return Ok(ApiResponse<List<StudentDto>>.CreateSuccess(
            children,
            "Children retrieved successfully"
        ));
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error getting children");
        return StatusCode(500,
            ApiResponse<List<StudentDto>>.CreateError(ex.Message));
    }

            return userId;
        }
        return null;
    }
    catch (Exception ex)

        return null;
    }

[HttpGet("child/{childId}")]
public async Task<ActionResult<ApiResponse<DiaryResponseDto>>> GetChildDiary(
    int childId,
    [FromQuery] DiaryFilters filters)
{
    try
    {
        _logger.LogInformation($"Fetching diary for child {childId}");

        // Get student info to get class and section
        var student = await _context.tbl_StudentRegistration
            .Include(s => s.Class)
            .Include(s => s.Section)
            .FirstOrDefaultAsync(s => s.Reg_Id == childId);

        if (student == null)
        {
            return NotFound(ApiResponse<DiaryResponseDto>.CreateError("Student not found"));
        }

        // Get class and section names
        string className = student.Class?.Class_Name ?? "N/A";
        string sectionName = student.Section?.Section ?? "N/A";
        // Fix: Use correct property name (adjust based on your actual column names)
        string studentName = !string.IsNullOrEmpty(student.Frist_Name)?student.Frist_Name:"NA";
                           
        // Get diary entries from tbl_ClassDiary for student's class and section
        var query = from diary in _context.tbl_ClassDiary
                    join subject in _context.tbl_Subject 
                        on diary.SubjectID equals subject.Subject_Id into subjectJoin
                    from subject in subjectJoin.DefaultIfEmpty()
                    join user in _context.tbl_User 
                        on diary.CreatedBy equals user.ID into userJoin
                    from user in userJoin.DefaultIfEmpty()
                    where diary.ClassID == student.Class_ID && 
                          diary.SectionID == student.Section_ID &&
                          diary.HomeWork != null &&
                          diary.HomeWork != "" &&
                          (diary.isPublished == null || diary.isPublished == true)
                    select new 
                    {
                        Diary = diary,
                        Subject = subject,
                        Teacher = user
                    };

        // Apply date filters
        if (!string.IsNullOrEmpty(filters.Date))
        {
            if (DateOnly.TryParse(filters.Date, out var date))
            {
                query = query.Where(x => x.Diary.Date == date);
            }
        }

        // Apply month/year filters
        if (filters.Month.HasValue && filters.Year.HasValue)
        {
            query = query.Where(x => x.Diary.Date.HasValue && 
                                     x.Diary.Date.Value.Month == filters.Month.Value && 
                                     x.Diary.Date.Value.Year == filters.Year.Value);
        }

        // Apply subject filter
        if (filters.SubjectId.HasValue)
        {
            query = query.Where(x => x.Diary.SubjectID == filters.SubjectId.Value);
        }

        var diaryEntries = await query
            .OrderByDescending(x => x.Diary.Date)
            .Select(x => new DiaryEntryDto
            {
                Id = x.Diary.dairyID,
                Date = x.Diary.Date.HasValue ? x.Diary.Date.Value.ToString("yyyy-MM-dd") : "",
                Content = x.Diary.HomeWork ?? "",
                Subject = x.Subject != null ? x.Subject.Subject_Name : "General",
                SubjectId = x.Diary.SubjectID,
                Teacher = x.Teacher != null ? x.Teacher.UserName : "Teacher",
                TeacherId = x.Diary.CreatedBy,
                ClassId = x.Diary.ClassID,
                ClassName = className,      // Add this field to DTO
                SectionId = x.Diary.SectionID,
                SectionName = sectionName,  // Add this field to DTO
                IsPublished = x.Diary.isPublished ?? false,
                CreatedAt = x.Diary.CreatedDT
            })
            .ToListAsync();

        var response = new DiaryResponseDto
        {
            Entries = diaryEntries,
            Count = diaryEntries.Count,
            StudentId = childId,
            StudentName = studentName,
            ClassName = className,
            SectionName = sectionName,
            GeneratedAt = DateTime.Now
        };

        return Ok(ApiResponse<DiaryResponseDto>.CreateSuccess(response));
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error fetching diary for child {ChildId}", childId);
        return StatusCode(500, ApiResponse<DiaryResponseDto>.CreateError($"Error fetching diary: {ex.Message}"));
    }
}

[HttpGet("teacher/classes")]
[Authorize(Roles = "Admin,Teacher,Manager")]
public async Task<IActionResult> GetTeacherClassesForDiary()
{
    try
    {
        _logger.LogInformation("GetTeacherClassesForDiary called");
        
        var userId = GetUserIdFromToken();
        if (!userId.HasValue)
        {
            return Unauthorized(new { success = false, message = "Invalid token" });
        }

        // Get ALL active classes with ALL their sections (no filtering by teacher)
        var allClassesWithSections = await _context.tbl_AddClass
            .Where(c => c.Active == true)
            .Select(c => new
            {
                ClassId = c.Class_ID,
                ClassName = c.Class_Name ?? string.Empty,
                Sections = _context.tbl_AddSection
                    .Where(s => s.Class_ID == c.Class_ID && s.Active == true)
                    .Select(s => new
                    {
                        SectionId = s.ID,
                        SectionName = s.Section ?? string.Empty,
                        TotalStudents = _context.tbl_StudentRegistration
                            .Count(st => st.Section_ID == s.ID && 
                                        (st.cur_Status == null || st.cur_Status == "A"))
                    })
                    .OrderBy(s => s.SectionName)
                    .ToList()
            })
            .Where(c => c.Sections.Any())
            .OrderBy(c => c.ClassName)
            .ToListAsync();

        _logger.LogInformation($"Returning {allClassesWithSections.Count} classes");
        
        // Log sections for debugging
        foreach (var cls in allClassesWithSections)
        {
            _logger.LogInformation($"Class {cls.ClassName} has {cls.Sections.Count} sections");
            foreach (var sec in cls.Sections)
            {
                _logger.LogInformation($"  - Section: {sec.SectionName} (ID: {sec.SectionId})");
            }
        }
        
        return Ok(new { success = true, data = allClassesWithSections });
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error in GetTeacherClassesForDiary");
        return StatusCode(500, new { success = false, message = ex.Message });
    }
}
        // GET: api/diary/class/{classId}/subjects
        [HttpGet("class/{classId}/subjects")]
        [Authorize(Roles = "Admin,Teacher,Manager")]
        public async Task<IActionResult> GetSubjectsByClass(int classId, [FromQuery] int? sectionId = null)
        {
            try
            {
                var userId = GetUserIdFromToken();
                if (!userId.HasValue)
                {
                    return Unauthorized(new { success = false, message = "Invalid token" });
                }

                // Get teacher's employee code
                var user = await _context.tbl_User
                    .FirstOrDefaultAsync(u => u.ID == userId.Value);

                if (user == null || !user.EmployeeCode.HasValue)
                {
                    return BadRequest(new { success = false, message = "Teacher employee code not found" });
                }

                var query = _context.tbl_EmployeeSubject
                    .Where(es => es.Employee_Code == user.EmployeeCode.Value && es.Class_ID == classId)
                    .Include(es => es.Subject)
                    .AsQueryable();

                if (sectionId.HasValue && sectionId.Value > 0)
                {
                    query = query.Where(es => es.Section_ID == sectionId.Value);
                }

                var subjects = await query
                    .Select(es => new
                    {
                        SubjectId = es.Subject_Id,
                        SubjectName = es.Subject != null ? es.Subject.Subject_Name : string.Empty,
                        SubjectCode = es.Subject != null ? es.Subject.SubjectCode : string.Empty,
                        ColorCode = es.Subject != null ? es.Subject.ColorCode : "#2196F3"
                    })
                    .Distinct()
                    .ToListAsync();

                return Ok(new { success = true, data = subjects });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error getting subjects");
                return StatusCode(500, new { success = false, message = ex.Message });
            }
        }

            try
            {
                if (!DateOnly.TryParse(date, out var diaryDate))

                var exists = await _context.tbl_ClassDiary

                                   d.SectionID == sectionId && 
                                   d.SubjectID == subjectId && 
                                   d.Date == diaryDate);

                return Ok(new { success = true, exists = exists });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error checking diary exists");
                return StatusCode(500, new { success = false, message = ex.Message });
            }
        }

        // POST: api/diary
        [HttpPost]
        [Authorize(Roles = "Admin,Teacher,Manager")]
        public async Task<IActionResult> CreateOrUpdateDiary([FromBody] CreateDiaryDto model)
        {
            try
            {
                _logger.LogInformation($"Creating/Updating diary for Class:{model.ClassID}, Section:{model.SectionID}, Subject:{model.SubjectID}, Date:{model.Date}");

                // Check if diary already exists
                var existingDiary = await _context.tbl_ClassDiary
                    .FirstOrDefaultAsync(d => d.Date == model.Date &&
                                               d.ClassID == model.ClassID &&
                                               d.SectionID == model.SectionID &&
                                               d.SubjectID == model.SubjectID);

                if (existingDiary != null)
                {
                    // Update existing diary
                    existingDiary.HomeWork = model.HomeWork;
                    existingDiary.ClassWork = model.ClassWork;
                    existingDiary.ModifyBy = model.CreatedBy;
                    existingDiary.ModifyDT = DateTime.Now;
                    existingDiary.isPublished = model.isPublished;

                    await _context.SaveChangesAsync();
                    _logger.LogInformation($"Updated diary entry ID: {existingDiary.dairyID}");

                    return Ok(new { 
                        success = true, 
                        message = "Diary updated successfully",
                        data = new { existingDiary.dairyID, existingDiary.isPublished }
                    });
                }
                else
                {
                    // Create new diary
                    var newDiary = new tbl_ClassDiary
                    {
                        Date = model.Date,
                        ClassID = model.ClassID,
                        SectionID = model.SectionID,
                        SubjectID = model.SubjectID,
                        ClassWork = model.ClassWork,
                        HomeWork = model.HomeWork,
                        CreatedBy = model.CreatedBy,
                        CreatedDT = DateTime.Now,
                        ModifyBy = null,
                        ModifyDT = null,
                        isPublished = model.isPublished,
                        ApprovedBy = null,
                        ApprovedDT = null
                    };

                    _context.tbl_ClassDiary.Add(newDiary);
                    await _context.SaveChangesAsync();
                    _logger.LogInformation($"Created new diary entry ID: {newDiary.dairyID}");

                    return Ok(new { 
                        success = true, 
                        message = "Diary created successfully",
                        data = new { newDiary.dairyID, newDiary.isPublished }
                    });
                }
            }
            catch (Exception ex)
            {

                        existingDiary.ClassWork = model.ClassWork;
                        existingDiary.ModifyBy = model.CreatedBy;
                        existingDiary.ModifyDT = DateTime.Now;
                        existingDiary.isPublished = model.isPublished;
                        updateCount++;
                        results.Add(new { sectionId, action = "updated", id = existingDiary.dairyID });
                    }
                    else
                    {
                        // Create new
                        var newDiary = new tbl_ClassDiary
                        {
                            Date = model.Date,

[HttpGet("teacher/entries")]
[Authorize(Roles = "Admin,Teacher,Manager")]
public async Task<IActionResult> GetTeacherDiaries(
    [FromQuery] int? classId,
    [FromQuery] string? sectionIds,
    [FromQuery] int? subjectId,
    [FromQuery] string? dateFrom,
    [FromQuery] string? dateTo,
    [FromQuery] int page = 1,
    [FromQuery] int limit = 20)
{
    try
    {
        _logger.LogInformation("GetTeacherDiaries called");
        
        var userId = GetUserIdFromToken();
        if (!userId.HasValue)
        {
            return Unauthorized(new { success = false, message = "Invalid token" });
        }

        // Get teacher's user info
        var user = await _context.tbl_User
            .FirstOrDefaultAsync(u => u.ID == userId.Value);

        if (user == null)
        {
            return BadRequest(new { success = false, message = "User not found" });
        }

        // Build query for diaries created by this teacher
        var query = _context.tbl_ClassDiary
            .Where(d => d.CreatedBy == userId.Value)
            .AsQueryable();

        // Apply filters
        if (classId.HasValue && classId.Value > 0)
        {
            query = query.Where(d => d.ClassID == classId.Value);
        }

        // Parse sectionIds from comma-separated string
        if (!string.IsNullOrEmpty(sectionIds))
        {
            var sectionIdList = sectionIds.Split(',').Select(int.Parse).ToList();
            if (sectionIdList.Any())
            {
                query = query.Where(d => sectionIdList.Contains(d.SectionID ?? 0));
            }
        }

        if (subjectId.HasValue && subjectId.Value > 0)
        {
            query = query.Where(d => d.SubjectID == subjectId.Value);
        }

        if (!string.IsNullOrEmpty(dateFrom))
        {
            if (DateOnly.TryParse(dateFrom, out var fromDate))
            {
                query = query.Where(d => d.Date >= fromDate);
            }
        }

        if (!string.IsNullOrEmpty(dateTo))
        {
            if (DateOnly.TryParse(dateTo, out var toDate))
            {
                query = query.Where(d => d.Date <= toDate);
            }
        }

        // Get total count for pagination
        var totalCount = await query.CountAsync();

        // Get paginated entries with joins to get class, section, subject names
        var entries = await query
            .OrderByDescending(d => d.Date)
            .ThenByDescending(d => d.CreatedDT)
            .Skip((page - 1) * limit)
            .Take(limit)
            .Select(d => new
            {
                d.dairyID,
                d.Date,
                d.ClassID,
                d.SectionID,
                d.SubjectID,
                d.HomeWork,
                d.ClassWork,
                d.isPublished,
                d.CreatedBy,
                d.CreatedDT,
                ClassName = _context.tbl_AddClass
                    .Where(c => c.Class_ID == d.ClassID)
                    .Select(c => c.Class_Name)
                    .FirstOrDefault() ?? "",
                SectionName = _context.tbl_AddSection
                    .Where(s => s.ID == d.SectionID)
                    .Select(s => s.Section)
                    .FirstOrDefault() ?? "",
                SubjectName = _context.tbl_Subject
                    .Where(s => s.Subject_Id == d.SubjectID)
                    .Select(s => s.Subject_Name)
                    .FirstOrDefault() ?? "",
                // Get all section names for this class (if multiple sections are saved separately)
                // For listing, we'll group by class and subject later
            })
            .ToListAsync();

        // Group entries by class, subject, date to show combined sections
        var groupedEntries = entries
            .GroupBy(e => new { e.Date, e.ClassID, e.SubjectID })
            .Select(g => new
            {
                id = g.First().dairyID,
                date = g.Key.Date.HasValue ? g.Key.Date.Value.ToString("yyyy-MM-dd") : "",
                classId = g.Key.ClassID,
                className = g.First().ClassName,
                subjectId = g.Key.SubjectID,
                subjectName = g.First().SubjectName,
                sectionIds = g.Select(e => e.SectionID).ToList(),
                sectionNames = g.Select(e => e.SectionName).Where(s => !string.IsNullOrEmpty(s)).ToList(),
                content = g.First().HomeWork ?? "",
                isPublished = g.First().isPublished ?? false,
                createdAt = g.First().CreatedDT
            })
            .OrderByDescending(e => e.date)
            .ToList();

        var response = new
        {
            success = true,
            data = new
            {
                entries = groupedEntries,
                count = groupedEntries.Count,
                total = totalCount,
                page = page,
                limit = limit,
                totalPages = (int)Math.Ceiling(totalCount / (double)limit)
            }
        };

        return Ok(response);
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error in GetTeacherDiaries");
        return StatusCode(500, new { success = false, message = ex.Message });
    }

                    .FirstOrDefaultAsync(s => s.Reg_Id == childId);

                if (student == null)
                {
                    return NotFound(ApiResponse<List<ChildGradeDto>>.CreateError("Student not found"));
                }

                // Get results for the student with all necessary joins
                var query = from resultEntry in _context.tbl_ResultEntry
                            join resultMaster in _context.tbl_ResultMaster 
                                on resultEntry.ResultID equals resultMaster.ResultID
                            join subject in _context.tbl_Subject 
                                on resultMaster.SubjectID equals subject.Subject_Id into subjectJoin
                            from subject in subjectJoin.DefaultIfEmpty()
                            join examBatch in _context.tbl_ResultBatch 
                                on resultMaster.BatchID equals examBatch.BatchID into batchJoin
                            from examBatch in batchJoin.DefaultIfEmpty()
                            join teacher in _context.tbl_Employee 
                                on resultMaster.TeacherID equals teacher.Employee_Code into teacherJoin
                            from teacher in teacherJoin.DefaultIfEmpty()
                            where resultEntry.Reg_Id == childId
                            select new
                            {
                                ResultEntry = resultEntry,
                                ResultMaster = resultMaster,
                                Subject = subject,
                                ExamBatch = examBatch,

                    query = query.Where(x => x.ResultMaster.BatchID == filters.BatchId.Value);

                    var dateFrom = DateOnly.FromDateTime(filters.DateFrom.Value);
                    query = query.Where(x => x.ResultMaster.ExamDate >= dateFrom);

        private string CalculateGradeFromScale(double obtainedMarks, double totalMarks, List<tbl_Grade> gradeScale)
        {
            if (totalMarks == 0) return "N/A";
            
            var percentage = (obtainedMarks / totalMarks) * 100;
            
            var grade = gradeScale
                .Where(g => percentage >= g.Grade_Percent)
                .OrderByDescending(g => g.Grade_Percent)
                .FirstOrDefault();

            return grade?.Grade ?? "F";
        }

        private string GetGradeRemarks(double obtainedMarks, double totalMarks, List<tbl_Grade> gradeScale, string gender)
        {
            if (totalMarks == 0) return "No remarks available";
            
            var percentage = (obtainedMarks / totalMarks) * 100;
            
            var grade = gradeScale
                .Where(g => percentage >= g.Grade_Percent)
                .OrderByDescending(g => g.Grade_Percent)
                .FirstOrDefault();

            if (grade == null) return "Needs Improvement";

            return gender?.ToLower() == "female" 
                ? grade.RemarksFemale ?? grade.Remarks1 ?? "Good"
                : grade.RemarksMale ?? grade.Remarks1 ?? "Good";
        }

        private string GetBestGradeFromPercentages(List<double> percentages, List<tbl_Grade> gradeScale)
        {
            if (!percentages.Any()) return "N/A";
            
            var bestPercentage = percentages.Max();
            
            var grade = gradeScale
                .Where(g => bestPercentage >= g.Grade_Percent)
                .OrderByDescending(g => g.Grade_Percent)
                .FirstOrDefault();

            return grade?.Grade ?? "N/A";
        }

        private string CalculateImprovement(List<GradeItem> grades)
        {
            if (grades.Count < 2) return "0%";
            
            var orderedGrades = grades
                .Where(g => g.ExamDate != null)
                .OrderBy(g => g.ExamDate)
                .ToList();
            
            if (!orderedGrades.Any()) return "0%";
            
            var firstPercentage = orderedGrades.First().Percentage;
            var lastPercentage = orderedGrades.Last().Percentage;
            
            var improvement = lastPercentage - firstPercentage;
            return improvement >= 0 ? $"+{improvement:F1}%" : $"{improvement:F1}%";
        }

        private double CalculateGPA(List<GradeItem> grades, List<tbl_Grade> gradeScale)
        {
            if (!grades.Any()) return 0;

            var totalPoints = 0.0;
            var count = 0;

            foreach (var grade in grades)
            {
                var gradeRecord = gradeScale
                    .Where(g => grade.Percentage >= g.Grade_Percent)
                    .OrderByDescending(g => g.Grade_Percent)
                    .FirstOrDefault();

                if (gradeRecord?.CGPA.HasValue == true)
                {
                    totalPoints += gradeRecord.CGPA.Value;
                    count++;
                }
            }

            return count > 0 ? Math.Round(totalPoints / count, 2) : 0;
        }

        private double CalculatePassRate(List<GradeItem> grades)
        {
            if (!grades.Any()) return 0;
            
            var passed = grades.Count(g => g.Percentage >= 40);
            return Math.Round((double)passed / grades.Count * 100, 1);
        }

        private List<TrendDataDto> CalculatePerformanceTrend(List<GradeItem> grades)
        {
            return grades
                .Where(g => g.ExamDate != null)
                .GroupBy(g => g.ExamDate?.ToString("MMM yyyy") ?? "Unknown")
                .Select(g => new TrendDataDto
                {
                    Label = g.Key,
                    Value = Math.Round(g.Average(x => x.Percentage), 1)
                })
                .OrderBy(t => {
                    try 
                    {
                        return DateTime.ParseExact(t.Label, "MMM yyyy", 
                            System.Globalization.CultureInfo.InvariantCulture);
                    }
                    catch
                    {
                        return DateTime.MinValue;
                    }
                })
                .ToList();
        }

        private string GetSubjectColor(int subjectId)
        {
            var colors = new[] { "#3498db", "#e74c3c", "#2ecc71", "#f39c12", "#9b59b6", "#1abc9c", "#e67e22", "#34495e" };
            return colors[subjectId % colors.Length];
        }

        private async Task<string> ProcessTargetFilterAsync(string targetFilter, string targetType, string category)
        {
            if (string.IsNullOrEmpty(targetFilter) || targetFilter == "{}")
                return "{}";

            try
            {
                JToken parsed;
                try
                {
                    parsed = JToken.Parse(targetFilter);
                }
                catch (JsonException)
                {
                    _logger.LogWarning("Invalid JSON in TargetFilter: {TargetFilter}", targetFilter);
                    return "{}";
                }
                
                if (targetType == "CLASS")
                {
                    List<int> classIds = new List<int>();
                    
                    if (parsed["ClassID"] is JArray classArray)
                    {
                        foreach (var item in classArray)
                        {
                            if (item.Type == JTokenType.Integer)
                            {
                                classIds.Add(item.Value<int>());
                            }
                            else if (item.Type == JTokenType.String)
                            {
                                if (int.TryParse(item.Value<string>(), out int val))
                                    classIds.Add(val);
                            }
                        }
                    }
                    else if (parsed["classId"] != null)
                    {
                        if (parsed["classId"].Type == JTokenType.Integer)
                        {
                            classIds.Add(parsed["classId"].Value<int>());
                        }
                        else if (parsed["classId"].Type == JTokenType.String)
                        {
                            if (int.TryParse(parsed["classId"].Value<string>(), out int val))
                                classIds.Add(val);
                        }
                    }
                    else if (parsed["classes"] is JArray classesArray)
                    {
                        foreach (var item in classesArray)
                        {
                            if (item.Type == JTokenType.Integer)
                            {
                                classIds.Add(item.Value<int>());
                            }
                        }
                    }
                    else if (parsed.Type == JTokenType.Array)
                    {
                        var array = parsed as JArray;
                        foreach (var item in array)
                        {
                            if (item.Type == JTokenType.Integer)
                            {
                                classIds.Add(item.Value<int>());
                            }
                        }
                    }
                    
                    if (classIds.Any())
                    {
                        var result = new JObject
                        {
                            ["ClassID"] = JArray.FromObject(classIds)
                        };
                        return result.ToString(Formatting.None);
                    }
                }
                else if (targetType == "SECTION" && parsed["SectionID"] != null)
                {
                    var transformed = new JObject
                    {
                        ["SectionID"] = parsed["SectionID"]
                    };
                    return transformed.ToString(Formatting.None);
                }
                
                return parsed.ToString(Formatting.None);
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "Error processing TargetFilter");
                return "{}";
            }
        }

        private object? SafeParseTargetFilter(string targetFilter)
        {
            if (string.IsNullOrEmpty(targetFilter) || targetFilter == "{}")
                return null;
            
            try
            {
                var parsed = JObject.Parse(targetFilter);
                
                if (parsed["ClassID"] is JArray classArray)
                {
                    var actualValues = new List<int>();
                    foreach (var item in classArray)
                    {
                        if (item.Type == JTokenType.Integer)
                        {
                            actualValues.Add(item.Value<int>());
                        }
                        else if (item.Type == JTokenType.String)
                        {
                            if (int.TryParse(item.Value<string>(), out int val))
                                actualValues.Add(val);
                        }
                    }
                    
                    if (actualValues.Any())
                    {
                        return new Dictionary<string, object>
                        {
                            ["ClassID"] = actualValues
                        };
                    }
                }
                
                var dict = new Dictionary<string, object>();
                foreach (var prop in parsed.Properties())
                {
                    var value = prop.Value;
                    if (value.Type == JTokenType.Integer)
                    {
                        dict[prop.Name] = value.Value<int>();
                    }
                    else if (value.Type == JTokenType.String)
                    {
                        dict[prop.Name] = value.Value<string>() ?? string.Empty;
                    }
                    else if (value.Type == JTokenType.Boolean)
                    {
                        dict[prop.Name] = value.Value<bool>();
                    }
                    else if (value.Type == JTokenType.Array)
                    {
                        var array = value as JArray;
                        if (array != null && array.Count > 0 && array[0].Type == JTokenType.Integer)
                        {
                            dict[prop.Name] = array.Select(x => x.Value<int>()).ToList();
                        }
                        else
                        {
                            dict[prop.Name] = array?.Select(x => x.ToString()).ToList() ?? new List<string>();
                        }
                    }
                    else
                    {
                        dict[prop.Name] = value.ToString();
                    }
                }
                
                return dict;
            }
            catch (JsonException ex)
            {
                _logger.LogWarning(ex, "Failed to parse TargetFilter as JObject: {TargetFilter}", targetFilter);
                
                try
                {
                    var array = JArray.Parse(targetFilter);
                    var values = new List<int>();
                    foreach (var item in array)
                    {
                        if (item.Type == JTokenType.Integer)
                        {
                            values.Add(item.Value<int>());
                        }
                    }
                    if (values.Any())
                    {
                        return new Dictionary<string, object>
                        {
                            ["ClassID"] = values
                        };
                    }
                    return values;
                }
                catch
                {
                    _logger.LogWarning("Failed to parse TargetFilter: {TargetFilter}", targetFilter);
                    return null;
                }
            }
        }

                            }

        public async Task<IActionResult> GetUserNotifications(int userId, [FromQuery] int page = 1, [FromQuery] int pageSize = 20)
        {
            try
            {
                var query = _context.NotificationQueue
                    .Where(n => n.UserID == userId)
                    .OrderByDescending(n => n.CreatedTime);

                var totalCount = await query.CountAsync();
                var notifications = await query
                    .Skip((page - 1) * pageSize)
                    .Take(pageSize)
                    .ToListAsync();
                
                var notificationDtos = new List<NotificationDTO>();
                
                foreach (var n in notifications)
                {
                    object parsedPayload = null;
                    
                    if (!string.IsNullOrEmpty(n.DataPayload))
                    {
                        try
                        {
                            // Parse with JObject first
                            var jObject = JObject.Parse(n.DataPayload);
                            
                            // Convert JObject to a regular Dictionary that System.Text.Json can handle
                            var dict = new Dictionary<string, object>();
                            foreach (var property in jObject.Properties())
                            {
                                var value = property.Value;
                                
                                // Handle different value types
                                if (value.Type == JTokenType.Object)
                                {
                                    dict[property.Name] = ConvertJObjectToDictionary((JObject)value);
                                }
                                else if (value.Type == JTokenType.Array)
                                {
                                    dict[property.Name] = ConvertJArrayToList((JArray)value);
                                }
                                else
                                {
                                    dict[property.Name] = ((JValue)value).Value;
                                }
                            }
                            
                            parsedPayload = dict;
                            _logger.LogInformation("Successfully parsed payload for notification {Id}", n.QueueID);
                        }
                        catch (Exception ex)
                        {
                            _logger.LogError(ex, "Failed to parse payload for notification {Id}", n.QueueID);
                            parsedPayload = new Dictionary<string, object>
                            {
                                ["raw"] = n.DataPayload,
                                ["error"] = ex.Message
                            };
                        }
                    }
                    
                    notificationDtos.Add(new NotificationDTO
                    {
                        QueueID = n.QueueID,
                        RuleID = n.RuleID,
                        UserID = n.UserID,
                        UserType = n.UserType ?? "",
                        Title = n.Title ?? "",
                        Message = n.Message ?? "",
                        DataPayload = parsedPayload,
                        Status = n.Status ?? "",
                        Priority = n.Priority,
                        ScheduledTime = n.ScheduledTime,
                        ProcessedTime = n.ProcessedTime,
                        CreatedTime = n.CreatedTime,
                        AttemptCount = n.AttemptCount,
                        ErrorMessage = n.ErrorMessage
                    });

                {

                    Data = notificationDtos,
                    Pagination = new
                    {
                        Page = page,

                        TotalPages = (int)Math.Ceiling(totalCount / (double)pageSize)

                return Ok(new
                {
                    Success = true,
                    Data = new
                    {
                        Period = new { From = fromDate, To = toDate },
                        Total = total,
                        ByStatus = stats,
                        SuccessRate = total > 0 ? (sentCount / (double)total * 100) : 0,
                        FailureRate = total > 0 ? (failedCount / (double)total * 100) : 0
                    }
                });
            }
            catch (Exception ex)

                _logger.LogError(ex, "Error getting notification stats");

        // Helper methods to convert JObject to Dictionary
        private Dictionary<string, object> ConvertJObjectToDictionary(JObject jObject)
        {
            var dict = new Dictionary<string, object>();
            foreach (var property in jObject.Properties())
            {
                var value = property.Value;
                
                if (value.Type == JTokenType.Object)
                {
                    dict[property.Name] = ConvertJObjectToDictionary((JObject)value);
                }
                else if (value.Type == JTokenType.Array)
                {
                    dict[property.Name] = ConvertJArrayToList((JArray)value);
                }
                else
                {
                    dict[property.Name] = ((JValue)value).Value;
                }
            }
            return dict;
        }

        private List<object> ConvertJArrayToList(JArray jArray)
        {
            var list = new List<object>();
            foreach (var item in jArray)
            {
                if (item.Type == JTokenType.Object)
                {
                    list.Add(ConvertJObjectToDictionary((JObject)item));
                }
                else if (item.Type == JTokenType.Array)
                {
                    list.Add(ConvertJArrayToList((JArray)item));
                }
                else
                {
                    list.Add(((JValue)item).Value);
                }
            }
            return list;
        }

        private object TryParseJson(string json)
        {
            if (string.IsNullOrEmpty(json))
            {
                return null;
            }
            
            try
            {
                var jObject = JObject.Parse(json);
                
                // Convert to Dictionary for System.Text.Json compatibility
                var dict = new Dictionary<string, object>();
                foreach (var property in jObject.Properties())
                {
                    var value = property.Value;
                    
                    if (value.Type == JTokenType.Object)
                    {
                        dict[property.Name] = ConvertJObjectToDictionary((JObject)value);
                    }
                    else if (value.Type == JTokenType.Array)
                    {
                        dict[property.Name] = ConvertJArrayToList((JArray)value);
                    }
                    else
                    {
                        dict[property.Name] = ((JValue)value).Value;
                    }
                }
                
                return dict;
            }
            catch
            {
                return new Dictionary<string, object>
                {
                    ["raw"] = json,
                    ["error"] = "Failed to parse JSON"
                };
            }
        }

[HttpPost("login")]
public async Task<ActionResult<AuthResponse>> Login(LoginRequest request)
{
    try
    {
        _logger.LogInformation("Login attempt for username: {Username}", request.Username);

        // =========================
        // 1. GET USER
        // =========================
        var user = await _context.tbl_User
            .Include(u => u.Role)
            .Include(u => u.Campus)
            .FirstOrDefaultAsync(u =>
                u.UserName.ToLower() == request.Username.ToLower() &&
                u.Status == true);

        if (user == null)
            return Unauthorized(new { message = "Invalid username or password" });

        // =========================
        // 2. CHECK PASSWORD
        // =========================
        bool passwordValid = CheckPassword(user.Password, request.Password);

        if (!passwordValid)
            return Unauthorized(new { message = "Invalid username or password" });

        // =========================
        // 3. JWT TOKEN
        // =========================
        var token = GenerateJwtToken(user);

        var permissions = GetUserPermissions(user.RoleID ?? 0);

        var userDto = new UserDto
        {
            Id = user.ID,
            Name = user.Name ?? user.UserName,
            UserName = user.UserName,
            Designation = user.Designation ?? "",
            RoleID = user.RoleID,
            RoleName = user.Role?.Role ?? "",
            CampusID = user.CampusID,
            CampusName = user.Campus?.CampusName ?? "",
            UserImage = user.UserImage ?? "",
            FamilyID = user.FamilyID,
            Permissions = permissions
        };

        // =========================
        // 4. CHILDREN (ONLY FOR PARENT)
        // =========================
        List<StudentDto> children = new();

        if (user.RoleID == 4 && user.FamilyID.HasValue)
        {
            children = await _context.tbl_StudentRegistration
                .Where(s => s.Family_Code == user.FamilyID && s.cur_Status == "A")
                .Include(s => s.Class)
                .Include(s => s.Section)
                .Include(s => s.Campus)
                .Select(s => new StudentDto
                {
                    Reg_Id = s.Reg_Id,
                    Reg_No = s.Reg_No,
                    Name = s.Frist_Name ?? "",
                    Image = s.Image ?? "",
                    Gender = s.Gender ?? "",
                    Class_ID = s.Class_ID,
                    ClassName = s.Class.Class_Name,
                    Section_ID = s.Section_ID,
                    SectionName = s.Section.Section,
                    RollNo = s.RollNo.HasValue ? s.RollNo.Value.ToString() : "",
                    Family_Code = s.Family_Code,
                    CampusName = s.Campus.CampusName,
                    Status = s.cur_Status,
                    DiaryAccess = 0
                })
                .ToListAsync();

            // =========================
            // 5. BULK DIARY ACCESS LOGIC (FAST)
            // =========================
            var regIds = children.Select(x => x.Reg_Id).ToList();
            var now = DateTime.Now;

            // TEMP ACCESS
            var tempAccessIds = await _context.tbl_DiaryAccessControl
                .Where(x =>
                    regIds.Contains(x.RegID) &&
                    x.IsActive &&
                    now >= x.AllowFrom &&
                    now <= x.AllowTo
                )
                .Select(x => x.RegID)
                .Distinct()
                .ToListAsync();

            // UNPAID CHECK
            var unpaidIds = await (
                from p in _context.tbl_FeePosting
                where regIds.Contains((int)p.Reg_ID)

                where !_context.tbl_FeeReceiveMaster
                    .Where(m => m.RegID == p.Reg_ID)
                    .Join(_context.tbl_FeeReceiveDetail,
                        m => m.Receipt_ID,
                        d => d.Receipt_ID,
                        (m, d) => new { m, d })
                    .Any(x =>
                        x.d.FeeMonth == p.FeeMonth &&
                        x.d.FeeYear == p.FeeYear
                    )

                select p.Reg_ID
            )
            .Distinct()
            .ToListAsync();

            // =========================
            // 6. APPLY SAME BUSINESS RULE
            // =========================
            foreach (var child in children)
            {
                bool hasUnpaid = unpaidIds.Contains(child.Reg_Id);
                bool hasTempAccess = tempAccessIds.Contains(child.Reg_Id);

                // FINAL RULE (same as Children API)
              child.DiaryAccess = hasTempAccess ? 1 : (hasUnpaid ? 0 : 1);
            }
        }

        // =========================
        // 7. RESPONSE
        // =========================
        var response = new AuthResponse
        {
            Token = token,
            User = userDto,
            Children = children
        };

        _logger.LogInformation("Login successful for user: {Username}", request.Username);

        return Ok(response);
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error during login");
        return StatusCode(500, new { message = "Internal server error", error = ex.Message });
    }
}

        private bool CheckPassword(string storedPassword, string inputPassword)
        {
            try
            {
                if (string.IsNullOrEmpty(storedPassword))
                    return false;

                // Try double base64 decode
                try
                {
                    string decodedDbPassword = Decode(storedPassword);
                    if (decodedDbPassword == inputPassword)
                    {
                        _logger.LogDebug("Password validated using double base64 decode");
                        return true;
                    }
                }
                catch { }

                // Try single base64 decode
                try
                {
                    byte[] decodedBytes = Convert.FromBase64String(storedPassword);
                    string decodedPassword = Encoding.UTF8.GetString(decodedBytes);
                    
                    if (decodedPassword == inputPassword)
                    {
                        _logger.LogDebug("Password validated using single base64 decode");
                        return true;
                    }
                }
                catch { }

                // Try direct comparison
                if (storedPassword == inputPassword)
                {
                    _logger.LogDebug("Password validated using direct comparison");
                    return true;
                }

                return false;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error checking password");
                return false;
            }
        }

        private string GenerateJwtToken(tbl_User user)
        {
            var tokenHandler = new JwtSecurityTokenHandler();
            
            var jwtSecret = _configuration["Jwt:Secret"] ?? 
                           "YourSuperSecretKeyHereThatIsAtLeast32CharactersLong12345";
            var key = Encoding.UTF8.GetBytes(jwtSecret);
            
            var issuer = _configuration["Jwt:Issuer"] ?? "Backend";
            var audience = _configuration["Jwt:Audience"] ?? "BackendUsers";
            
            _logger.LogDebug("Generating JWT token with Issuer: {Issuer}, Audience: {Audience}", issuer, audience);
string roleName =user.Role != null ? user.Role.Role : string.Empty;
            // FIX: Use standard claim types to match middleware configuration
            var claims = new List<Claim>
            {
                new Claim(ClaimTypes.NameIdentifier, user.ID.ToString()),
                new Claim(ClaimTypes.Name, user.UserName ?? string.Empty),
                 new Claim(ClaimTypes.Role, roleName), 
                 new Claim("RoleID", (user.RoleID ?? 0).ToString()),
                new Claim("FamilyID", user.FamilyID?.ToString() ?? "0"),
                new Claim("CampusID", user.CampusID?.ToString() ?? "0"),
                new Claim(JwtRegisteredClaimNames.Jti, Guid.NewGuid().ToString())
            };

            var tokenDescriptor = new SecurityTokenDescriptor
            {
                Subject = new ClaimsIdentity(claims),
                Expires = DateTime.UtcNow.AddDays(7),
                Issuer = issuer,
                Audience = audience,
                SigningCredentials = new SigningCredentials(
                    new SymmetricSecurityKey(key), 
                    SecurityAlgorithms.HmacSha256Signature)
            };

            var token = tokenHandler.CreateToken(tokenDescriptor);
            var tokenString = tokenHandler.WriteToken(token);
            
            _logger.LogDebug("Generated token (first 50 chars): {Token}", 
                tokenString.Substring(0, Math.Min(50, tokenString.Length)));
            
            return tokenString;
        }

        private int? GetUserIdFromToken()
        {
            try
            {
                var authHeader = Request.Headers["Authorization"].ToString();
                _logger.LogDebug("Authorization header: {AuthHeader}", authHeader);
                
                if (string.IsNullOrEmpty(authHeader))
                {
                    _logger.LogDebug("No Authorization header found");
                    return null;
                }

                // Handle both "Bearer token" and just "token" formats
                string token;
                if (authHeader.StartsWith("Bearer ", StringComparison.OrdinalIgnoreCase))
                {
                    token = authHeader.Substring("Bearer ".Length).Trim();
                }
                else
                {
                    token = authHeader.Trim();
                }

                if (string.IsNullOrEmpty(token))
                {
                    _logger.LogDebug("Empty token");
                    return null;
                }

                _logger.LogDebug("Token extracted, length: {TokenLength}", token.Length);
                
                var tokenHandler = new JwtSecurityTokenHandler();
                var key = Encoding.UTF8.GetBytes(_configuration["Jwt:Secret"] ?? 
                    "YourSuperSecretKeyHereThatIsAtLeast32CharactersLong12345");
                
                // Validate token with the same parameters as middleware
                var validationParameters = new TokenValidationParameters
                {
                    ValidateIssuerSigningKey = true,
                    IssuerSigningKey = new SymmetricSecurityKey(key),
                    ValidateIssuer = true,
                    ValidIssuer = _configuration["Jwt:Issuer"] ?? "Backend",
                    ValidateAudience = true,
                    ValidAudience = _configuration["Jwt:Audience"] ?? "BackendUsers",
                    ValidateLifetime = true,
                    ClockSkew = TimeSpan.FromMinutes(5),
                    // Use standard claim types to match middleware
                    NameClaimType = ClaimTypes.Name,
                    RoleClaimType = ClaimTypes.Role
                };

                try
                {
                    var principal = tokenHandler.ValidateToken(token, validationParameters, out SecurityToken validatedToken);
                    _logger.LogDebug("Token validated successfully");
                    
                    var jwtToken = (JwtSecurityToken)validatedToken;
                    
                    // Look for user ID in standard NameIdentifier claim
                    var userIdClaim = principal.FindFirst(ClaimTypes.NameIdentifier);
                    
                    if (userIdClaim != null)
                    {
                        _logger.LogDebug("Found user ID claim: {ClaimType} = {ClaimValue}", userIdClaim.Type, userIdClaim.Value);
                        if (int.TryParse(userIdClaim.Value, out int userId))
                        {
                            _logger.LogDebug("Parsed user ID: {UserId}", userId);
                            return userId;
                        }
                        else
                        {
                            _logger.LogWarning("Could not parse user ID from claim value: {ClaimValue}", userIdClaim.Value);
                        }
                    }
                    else
                    {
                        _logger.LogWarning("No user ID claim found in token");
                    }
                }
                catch (SecurityTokenExpiredException ex)
                {
                    _logger.LogWarning("Token expired: {Message}", ex.Message);
                }
                catch (SecurityTokenInvalidAudienceException ex)
                {
                    _logger.LogWarning("Invalid audience: {Message}", ex.Message);
                }
                catch (SecurityTokenInvalidIssuerException ex)
                {
                    _logger.LogWarning("Invalid issuer: {Message}", ex.Message);
                }
                catch (SecurityTokenInvalidSignatureException ex)
                {
                    _logger.LogWarning("Invalid signature: {Message}", ex.Message);
                }
                catch (Exception ex)
                {
                    _logger.LogError(ex, "Error validating token");
                }
                
                return null;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error getting user ID from token");
                return null;
            }
        }

                var userId = GetUserIdFromToken();
                if (!userId.HasValue)
                {
                    return Unauthorized(new { success = false, message = "Invalid token" });
                }

                var today = DateOnly.FromDateTime(DateTime.Now);
                var existingAttendance = await _context.tbl_StudentAttMaster
                    .FirstOrDefaultAsync(m => 
                        m.AttDate == today && 
                        m.ClassID == request.ClassId && 
                        m.SectionID == request.SectionId);

                if (existingAttendance != null)
                {
                    var existingDetails = await _context.tbl_StudentAttDetail
                        .Where(d => d.MasterId == existingAttendance.ID)
                        .ToListAsync();
                    
                    _context.tbl_StudentAttDetail.RemoveRange(existingDetails);
                    _context.tbl_StudentAttMaster.Remove(existingAttendance);
                    await _context.SaveChangesAsync();
                }

                var attendanceMaster = new tbl_StudentAttMaster
                {
                    AttDate = today,
                    Type = "Daily",
                    Remarks = request.Remarks,
                    CreatedBy = userId.Value,
                    CreatedDT = DateTime.Now,
                    CampusID = 1,
                    ClassID = request.ClassId,
                    SectionID = request.SectionId
                };

                await _context.tbl_StudentAttMaster.AddAsync(attendanceMaster);
                await _context.SaveChangesAsync();

                var attendanceDetails = new List<tbl_StudentAttDetail>();
                foreach (var student in request.AttendanceRecords)
                {
                    var detail = new tbl_StudentAttDetail
                    {
                        MasterId = attendanceMaster.ID,
                        RegID = student.StudentId,
                        Status = student.Status,
                        LateStatus = student.LateStatus,
                        Remarks = student.Remarks,
                        Is_Manual = true,
                        SmsSend = false
                    };
                    attendanceDetails.Add(detail);
                }

                await _context.tbl_StudentAttDetail.AddRangeAsync(attendanceDetails);
                await _context.SaveChangesAsync();

                await transaction.CommitAsync();

                return Ok(new
                {
                    success = true,
                    message = "Attendance marked successfully",
                    attendanceId = attendanceMaster.ID,
                    count = attendanceDetails.Count
                });
            }
            catch (Exception ex)
            {
                await transaction.RollbackAsync();
                _logger.LogError(ex, "Error in MarkAttendance");
                return StatusCode(500, new { success = false, message = ex.Message });
            }
        }

        // GET: api/Teacher/AttendanceHistory/{classId}/{sectionId}/{month}/{year}
        [HttpGet("AttendanceHistory/{classId}/{sectionId}/{month}/{year}")]
        public async Task<ActionResult> GetAttendanceHistory(int classId, int sectionId, int month, int year)
        {
            try
            {
                var userId = GetUserIdFromToken();
                if (!userId.HasValue)
                {
                    return Unauthorized(new { success = false, message = "Invalid token" });
                }

                var attendanceHistory = await _context.tbl_StudentAttMaster
                    .Include(m => m.tbl_StudentAttDetail)
                    .Where(m => m.ClassID == classId && 
                                m.SectionID == sectionId &&
                                m.AttDate.HasValue &&
                                m.AttDate.Value.Month == month &&
                                m.AttDate.Value.Year == year)
                    .OrderBy(m => m.AttDate)
                    .Select(m => new
                    {
                        Date = m.AttDate.Value.ToString("yyyy-MM-dd"),
                        TotalStudents = m.tbl_StudentAttDetail.Count,
                        PresentCount = m.tbl_StudentAttDetail.Count(d => d.Status == "P"),
                        AbsentCount = m.tbl_StudentAttDetail.Count(d => d.Status == "A"),
                        LeaveCount = m.tbl_StudentAttDetail.Count(d => d.Status == "L"),
                        LateCount = m.tbl_StudentAttDetail.Count(d => d.LateStatus == "Late")
                    })
                    .ToListAsync();

                return Ok(new
                {
                    success = true,
                    history = attendanceHistory,
                    month = month,
                    year = year
                });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in GetAttendanceHistory");
                return StatusCode(500, new { success = false, message = ex.Message });
            }
        }

        private int? GetUserIdFromToken()
        {
            try
            {
                _logger.LogDebug("GetUserIdFromToken called");
                
                // IMPORTANT: Use standard claim type that matches middleware configuration
                var userIdClaim = User.FindFirst(ClaimTypes.NameIdentifier);
                
                if (userIdClaim != null)
                {
                    _logger.LogDebug("Found user ID claim: {ClaimType} = {ClaimValue}", userIdClaim.Type, userIdClaim.Value);
                    if (int.TryParse(userIdClaim.Value, out int userId))
                    {
                        _logger.LogDebug("Parsed user ID: {UserId}", userId);
                        return userId;
                    }
                    else
                    {
                        _logger.LogWarning("Could not parse user ID from claim value: {ClaimValue}", userIdClaim.Value);
                    }
                }
                else
                {
                    // Fallback for debugging
                    var allClaims = User.Claims.Select(c => $"{c.Type}: {c.Value}").ToList();
                    _logger.LogWarning("No NameIdentifier claim found. Available claims: {Claims}", 
                        string.Join(", ", allClaims));
                }
                
                return null;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error in GetUserIdFromToken");
                return null;
            }
        }

        [HttpGet("validate-username")]

                return Ok(new { exists });

                _logger.LogError(ex, "Error validating username");

        // POST: api/Users
        [HttpPost]
        public async Task<IActionResult> CreateUser([FromBody] CreateUserRequest request)
        {
            try
            {
                // Check if username exists
                var existingUser = await _context.tbl_User
                    .AnyAsync(u => u.UserName != null && u.UserName.ToLower() == request.Username.ToLower());

                if (existingUser)
                    return BadRequest(new { Success = false, Message = "Username already exists" });

                // Encode password using the same method as AuthController
                var encodedPassword = EncodePassword(request.Password);

                var user = new tbl_User
                {
                    UserName = request.Username,
                    Name = request.FullName,
                    Password = encodedPassword,
                    Designation = request.UserType,
                    RoleID = GetRoleIdFromType(request.UserType),
                    Status = true,
                    CreateDT = DateTime.Now,
                    CreatedBy = GetCurrentUserId()
                };

                _context.tbl_User.Add(user);
                await _context.SaveChangesAsync();

                return Ok(new { Success = true, Message = "User created successfully", Data = new { user.ID } });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error creating user");
                return StatusCode(500, new { Success = false, Message = ex.Message });
            }
        }

        // POST: api/Users/{id}/update
        [HttpPost("{id}/update")]
        public async Task<IActionResult> UpdateUser(int id, [FromBody] UpdateUserRequest request)
        {
            try
            {
                var user = await _context.tbl_User.FindAsync(id);
                if (user == null)
                    return NotFound(new { Success = false, Message = "User not found" });

                user.Name = request.FullName ?? user.Name;
                user.Designation = request.UserType ?? user.Designation;
                user.RoleID = GetRoleIdFromType(request.UserType);
                user.Status = request.IsActive;
                user.ModifiedDT = DateTime.Now;
                user.ModifiedBy = GetCurrentUserId();

                await _context.SaveChangesAsync();

                return Ok(new { Success = true, Message = "User updated successfully" });
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error updating user");
                return StatusCode(500, new { Success = false, Message = ex.Message });
            }
        }

        [HttpPost("{id}/reset-password")]

            {

        [HttpPost("{id}/status")]
        public async Task<IActionResult> ToggleUserStatus(int id, [FromBody] ToggleStatusRequest request)

            try
            {

        [HttpDelete("{id}")]
        public async Task<IActionResult> DeleteUser(int id)
        {
            try
            {
                var user = await _context.tbl_User.FindAsync(id);
                if (user == null)
                    return NotFound(new { Success = false, Message = "User not found" });

                // Don't allow deleting admin users (RoleID = 1)
                if (user.RoleID == 1)
                    return BadRequest(new { Success = false, Message = "Cannot delete admin users" });

                _context.tbl_User.Remove(user);
                await _context.SaveChangesAsync();

                return Ok(new { Success = true, Message = "User deleted successfully" });

            catch (Exception ex)
            {
                _logger.LogError(ex, "Error deleting user");
                return StatusCode(500, new { Success = false, Message = ex.Message });

        #region Helper Methods

        private string EncodePassword(string password)
        {
            // Use the same encoding as AuthController
            byte[] bt = Encoding.ASCII.GetBytes(password);
            string str2 = Convert.ToBase64String(bt);
            byte[] bt2 = Encoding.ASCII.GetBytes(str2);
            return Convert.ToBase64String(bt2);
        }

        private int GetRoleIdFromType(string userType)
        {
            return userType?.ToLower() switch
            {
                "admin" => 1,
                "manager" => 2,
                "teacher" => 3,
                "parent" => 4,
                "student" => 5,
                "user" => 3,
                _ => 3 // Default to User/Teacher
            };
        }

        private int GetCurrentUserId()
        {
            var userIdClaim = User.FindFirst(ClaimTypes.NameIdentifier)?.Value;
            if (int.TryParse(userIdClaim, out int userId))
                return userId;
            return 1;
        }
