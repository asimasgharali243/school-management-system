    public class BackgroundNotificationService : IBackgroundNotificationService
    {
        private readonly ApplicationDbContext _context;
        private readonly INotificationRuleProcessor _ruleProcessor;
        private readonly INotificationService _notificationService;
        private readonly ILogger<BackgroundNotificationService> _logger;
        private readonly TimeZoneInfo _pakistanTimeZone;

        public BackgroundNotificationService(
            ApplicationDbContext context,
            INotificationRuleProcessor ruleProcessor,
            INotificationService notificationService,
            ILogger<BackgroundNotificationService> logger)
        {
            _context = context;
            _ruleProcessor = ruleProcessor;
            _notificationService = notificationService;
            _logger = logger;
            
            try
            {
                _pakistanTimeZone = TimeZoneInfo.FindSystemTimeZoneById("Pakistan Standard Time");
            }
            catch
            {
                try
                {
                    _pakistanTimeZone = TimeZoneInfo.FindSystemTimeZoneById("Asia/Karachi");
                }
                catch
                {
                    _pakistanTimeZone = TimeZoneInfo.Local;
                    _logger.LogWarning("⚠️ Pakistan timezone not found, using local timezone");
                }
            }
        }

        private DateTime GetPakistanTime()
        {
            return TimeZoneInfo.ConvertTime(DateTime.UtcNow, _pakistanTimeZone);
        }

        public async Task ProcessScheduledNotificationsAsync()
        {
            try
            {
                var pakistanNow = GetPakistanTime();
                _logger.LogInformation("🕐 Processing scheduled notifications at Pakistan time: {Time}", 
                    pakistanNow.ToString("yyyy-MM-dd HH:mm:ss"));
                
                var dueRules = await _context.NotificationRules
                    .Where(r => r.IsActive && 
                           (r.RuleType == "SCHEDULED" || r.RuleType == "CRON") &&
                           !r.IsManualTrigger &&
                           (r.NextRunTime == null || r.NextRunTime <= pakistanNow))
                    .ToListAsync();

                _logger.LogInformation("📋 Found {Count} scheduled rules due for processing", dueRules.Count);

                foreach (var rule in dueRules)
                {
                    try
                    {
                        _logger.LogInformation("⚡ Processing scheduled rule: {RuleName} (ID: {RuleID}, Type: {RuleType})", 
                            rule.RuleName, rule.RuleID, rule.RuleType);
                        
                        await ProcessRuleBasedOnCategory(rule);
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(ex, "❌ Error processing scheduled rule {RuleID}", rule.RuleID);
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error in ProcessScheduledNotificationsAsync");
            }
        }

        public async Task CheckEventRulesAsync()
        {
            try
            {
                var pakistanNow = GetPakistanTime();
                
                var eventRules = await _context.NotificationRules
                    .Where(r => r.IsActive && 
                           r.RuleType == "EVENT" && 
                           !r.IsManualTrigger)
                    .ToListAsync();

                _logger.LogInformation("📋 Found {Count} active automatic event rules", eventRules.Count);

                foreach (var rule in eventRules)
                {
                    try
                    {
                        if (rule.TimingType == "IMMEDIATE")
                        {
                            if (rule.LastRunTime.HasValue && 
                                (pakistanNow - rule.LastRunTime.Value).TotalMinutes < 5)
                            {
                                _logger.LogInformation("⏸️ Event rule {RuleID} ran {Minutes} minutes ago, skipping", 
                                    rule.RuleID, (int)(pakistanNow - rule.LastRunTime.Value).TotalMinutes);
                                continue;
                            }
                        }
                        
                        await CheckSpecificEventRuleAsync(rule);
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(ex, "❌ Error checking event rule {RuleID}", rule.RuleID);
                    }
                }
                
                var manualRules = await _context.NotificationRules
                    .Where(r => r.IsActive && r.IsManualTrigger)
                    .ToListAsync();
                    
                if (manualRules.Any())
                {
                    _logger.LogInformation("🔧 Found {Count} manual trigger rules (will only run when manually executed)", 
                        manualRules.Count);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error in CheckEventRulesAsync");
            }
        }

        public async Task CheckSpecificEventRuleAsync(NotificationRule rule)
        {
            _logger.LogInformation("🔍 Processing event rule: {RuleName} (Category: {Category})", rule.RuleName, rule.Category);

            if (!await ShouldRuleRunNowAsync(rule))
            {
                _logger.LogInformation("⏸️ Event rule {RuleID} should not run now based on timing, skipping", rule.RuleID);
                return;
            }

            switch (rule.Category?.ToUpper())
            {
                case "ATTENDANCE":
                    await ProcessAttendanceRuleAsync(rule);
                    break;
                case "FEE":
                    await ProcessFeeRuleAsync(rule);
                    break;
                case "EXAM":
                    await ProcessExamRuleAsync(rule);
                    break;
                case "HOMEWORK":
                case "ANNOUNCEMENT":
                case "GENERAL":
                case "ALERT":
                    await ProcessGenericRuleAsync(rule);
                    break;
                default:
                    _logger.LogWarning("⚠️ Unknown category '{Category}' for rule {RuleID}", rule.Category, rule.RuleID);
                    break;
            }
        }

        private async Task<bool> ShouldRuleRunNowAsync(NotificationRule rule)
        {
            try
            {
                var pakistanNow = GetPakistanTime();
                
                if (rule.RuleType == "MANUAL" || rule.IsManualTrigger)
                    return false;
                
                if (rule.RuleType == "EVENT")
                {
                    if (rule.TimingType == "IMMEDIATE")
                    {
                        if (!rule.LastRunTime.HasValue)
                            return true;
                            
                        var minutesSinceLastRun = (pakistanNow - rule.LastRunTime.Value).TotalMinutes;
                        return minutesSinceLastRun >= 5;
                    }
                    
                    return true;
                }
                
                if (rule.RuleType == "CRON" || rule.RuleType == "SCHEDULED")
                {
                    return await CheckScheduleTimingAsync(rule, pakistanNow);
                }
                
                return true;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error checking if rule should run now");
                return false;
            }
        }

        private async Task<bool> CheckScheduleTimingAsync(NotificationRule rule, DateTime pakistanNow)
        {
            switch (rule.TimingType?.ToUpper())
            {
                case "IMMEDIATE":
                    return true;
                    
                case "FIXED_TIME":
                    if (!rule.FixedTime.HasValue) return false;
                    var scheduledTime = pakistanNow.Date.Add(rule.FixedTime.Value);
                    var timeDifference = Math.Abs((pakistanNow - scheduledTime).TotalMinutes);
                    return timeDifference <= 15;
                    
                case "DAYS_OFFSET":
                    if (!rule.DaysOffset.HasValue || !rule.LastRunTime.HasValue) return false;
                    var daysSinceLastRun = (pakistanNow.Date - rule.LastRunTime.Value.Date).Days;
                    return daysSinceLastRun >= rule.DaysOffset.Value;
                    
                case "CRON":
                    if (string.IsNullOrEmpty(rule.CronExpression)) return false;
                    return await CheckCronExpressionAsync(rule.CronExpression, pakistanNow);
                    
                case "DAILY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 1;
                    
                case "WEEKLY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 7;
                    
                case "BIWEEKLY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 14;
                    
                case "MONTHLY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 30;
                    
                case "WEEKDAY":
                    return pakistanNow.DayOfWeek != DayOfWeek.Saturday && pakistanNow.DayOfWeek != DayOfWeek.Sunday;
                    
                case "WEEKEND":
                    return pakistanNow.DayOfWeek == DayOfWeek.Saturday || pakistanNow.DayOfWeek == DayOfWeek.Sunday;
                    
                default:
                    return true;
            }
        }

        private async Task<bool> CheckCronExpressionAsync(string cronExpression, DateTime pakistanNow)
        {
            try
            {
                var utcNow = pakistanNow.ToUniversalTime();
                var parts = cronExpression.Trim().Split(' ', StringSplitOptions.RemoveEmptyEntries);
                
                if (parts.Length != 5 && parts.Length != 6)
                {
                    _logger.LogError("❌ Invalid cron expression format: {Cron}", cronExpression);
                    return false;
                }
                
                var format = parts.Length == 6 ? CronFormat.IncludeSeconds : CronFormat.Standard;
                var parsedExpression = CronExpression.Parse(cronExpression, format);
                
                var nextOccurrence = parsedExpression.GetNextOccurrence(utcNow.AddMinutes(-5), inclusive: false);
                
                if (nextOccurrence.HasValue)
                {
                    var nextLocal = nextOccurrence.Value.ToLocalTime();
                    var timeDiff = Math.Abs((pakistanNow - nextLocal).TotalMinutes);
                    return timeDiff <= 5;
                }
                
                return false;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error parsing cron expression: {Cron}", cronExpression);
                return false;
            }
        }

        private async Task ProcessRuleBasedOnCategory(NotificationRule rule)
        {
            switch (rule.Category?.ToUpper())
            {
                case "FEE":
                    await ProcessFeeRuleAsync(rule);
                    break;
                case "ATTENDANCE":
                    await ProcessAttendanceRuleAsync(rule);
                    break;
                case "EXAM":
                    await ProcessExamRuleAsync(rule);
                    break;
                case "HOMEWORK":
                case "ANNOUNCEMENT":
                case "GENERAL":
                case "ALERT":
                    await ProcessGenericRuleAsync(rule);
                    break;
                default:
                    _logger.LogWarning("⚠️ Unknown category '{Category}' for rule {RuleID}", rule.Category, rule.RuleID);
                    break;
            }
        }

        /// <summary>
        /// Gets target users for a rule based on target type and filter
        /// Supports multiple sections with their own class IDs, specific students, and direct student selection
        /// </summary>
        private async Task<List<int>> GetTargetUsersForRuleAsync(NotificationRule rule)
        {
            var users = new List<int>();

            try
            {
                if (string.IsNullOrEmpty(rule.TargetType))
                {
                    return users;
                }

                switch (rule.TargetType.ToUpper())
                {
                    case "ALL":
                        users = await _context.tbl_User
                            .Where(u => u.Status == true)
                            .Select(u => u.ID)
                            .ToListAsync();
                        break;

                    case "PARENTS":
                        users = await _context.tbl_User
                            .Where(u => u.Status == true && u.RoleID == 4)
                            .Select(u => u.ID)
                            .ToListAsync();
                        break;

                    case "TEACHERS":
                        users = await _context.tbl_User
                            .Where(u => u.Status == true && u.RoleID == 3)
                            .Select(u => u.ID)
                            .ToListAsync();
                        break;

                    case "STUDENTS":
                        users = await _context.tbl_User
                            .Where(u => u.Status == true && u.RoleID == 4)
                            .Select(u => u.ID)
                            .ToListAsync();
                        break;

                    case "SECTION":
                        var sections = ParseSectionsFromTargetFilter(rule);
                        
                        if (sections.Any())
                        {
                            var allParentIds = new HashSet<int>();
                            
                            _logger.LogInformation("📋 Processing {Count} sections for rule {RuleID}", sections.Count, rule.RuleID);
                            
                            foreach (var section in sections)
                            {
                                _logger.LogInformation("🎯 Processing Section ID: {SectionID} (Class ID: {ClassID}) with {StudentCount} specific students", 
                                    section.SectionID, section.ClassID, section.StudentIDs.Count);
                                
                                var studentQuery = _context.tbl_StudentRegistration
                                    .Where(s => s.Section_ID == section.SectionID 
                                        && s.Class_ID == section.ClassID
                                        && (s.cur_Status == null || s.cur_Status == "A" || s.cur_Status == "Active" || s.cur_Status == "Admission"));
                                
                                if (section.StudentIDs.Any())
                                {
                                    studentQuery = studentQuery.Where(s => section.StudentIDs.Contains(s.Reg_Id));
                                    _logger.LogInformation("   📍 Filtering to {Count} specific students: {StudentIDs}", 
                                        section.StudentIDs.Count, string.Join(",", section.StudentIDs));
                                }
                                else
                                {
                                    _logger.LogInformation("   📍 No specific student filter - will include ALL students in this section");
                                }
                                
                                var familyCodes = await studentQuery
                                    .Where(s => s.Family_Code.HasValue)
                                    .Select(s => s.Family_Code.Value)
                                    .Distinct()
                                    .ToListAsync();
                                
                                _logger.LogInformation("   👨‍👩‍👧 Found {FamilyCount} family codes for Section {SectionID}", 
                                    familyCodes.Count, section.SectionID);
                                
                                if (familyCodes.Any())
                                {
                                    var parentIds = await _context.tbl_User
                                        .Where(u => familyCodes.Contains(u.FamilyID ?? 0) && u.Status == true)
                                        .Select(u => u.ID)
                                        .ToListAsync();
                                    
                                    foreach (var pid in parentIds)
                                    {
                                        allParentIds.Add(pid);
                                    }
                                    
                                    _logger.LogInformation("   👨‍👩‍👧 Found {ParentCount} parents for Section {SectionID}", 
                                        parentIds.Count, section.SectionID);
                                }
                                else
                                {
                                    _logger.LogWarning("   ⚠️ No family codes found for Section {SectionID}", section.SectionID);
                                }
                            }
                            
                            users = allParentIds.ToList();
                            _logger.LogInformation("📊 Total unique parents from all sections: {TotalParents}", users.Count);
                        }
                        else
                        {
                            _logger.LogWarning("⚠️ No sections parsed from target filter for rule {RuleID}", rule.RuleID);
                        }
                        break;

                    case "CLASS":
                        var classIds = ParseClassFilter(rule);
                        if (classIds.Any())
                        {
                            var familyCodes = await _context.tbl_StudentRegistration
                                .Where(s => classIds.Contains(s.Class_ID ?? 0) 
                                    && (s.cur_Status == null || s.cur_Status == "A" || s.cur_Status == "Active" || s.cur_Status == "Admission")
                                    && s.Family_Code.HasValue)
                                .Select(s => s.Family_Code.Value)
                                .Distinct()
                                .ToListAsync();
                            
                            if (familyCodes.Any())
                            {
                                users = await _context.tbl_User
                                    .Where(u => familyCodes.Contains(u.FamilyID ?? 0) && u.Status == true)
                                    .Select(u => u.ID)
                                    .ToListAsync();
                            }
                        }
                        break;

                    case "SPECIFIC_USERS":
                        var userIds = ParseSpecificUsers(rule);
                        if (userIds.Any())
                        {
                            users = await _context.tbl_User
                                .Where(u => userIds.Contains(u.ID) && u.Status == true)
                                .Select(u => u.ID)
                                .ToListAsync();
                        }
                        break;

                    // NEW: Handle direct student selection - send to parents of selected students
                    case "SPECIFIC_STUDENTS_DIRECT":
                        var specificStudentIds = ParseSpecificStudents(rule);
                        if (specificStudentIds.Any())
                        {
                            _logger.LogInformation("🎯 Processing SPECIFIC_STUDENTS_DIRECT with {Count} students: {StudentIds}", 
                                specificStudentIds.Count, string.Join(",", specificStudentIds));
                            
                            // Get family codes for these specific students
                            var familyCodes = await _context.tbl_StudentRegistration
                                .Where(s => specificStudentIds.Contains(s.Reg_Id) 
                                    && (s.cur_Status == null || s.cur_Status == "A" || s.cur_Status == "Active" || s.cur_Status == "Admission")
                                    && s.Family_Code.HasValue)
                                .Select(s => s.Family_Code.Value)
                                .Distinct()
                                .ToListAsync();
                            
                            _logger.LogInformation("👨‍👩‍👧 Found {FamilyCount} family codes for selected students", familyCodes.Count);
                            
                            if (familyCodes.Any())
                            {
                                // Get parent IDs from family codes
                                var parentIds = await _context.tbl_User
                                    .Where(u => familyCodes.Contains(u.FamilyID ?? 0) && u.Status == true)
                                    .Select(u => u.ID)
                                    .ToListAsync();
                                
                                users = parentIds;
                                _logger.LogInformation("👥 Found {ParentCount} parents for the selected students", parentIds.Count);
                            }
                            else
                            {
                                _logger.LogWarning("⚠️ No family codes found for selected students: {StudentIds}", 
                                    string.Join(",", specificStudentIds));
                            }
                        }
                        break;
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error in GetTargetUsersForRuleAsync for rule {RuleID}", rule.RuleID);
            }

            return users;
        }

        // ==================== PARSING HELPER METHODS ====================

        private class SectionTarget
        {
            public int SectionID { get; set; }
            public int ClassID { get; set; }
            public List<int> StudentIDs { get; set; } = new List<int>();
        }

        private List<SectionTarget> ParseSectionsFromTargetFilter(NotificationRule rule)
        {
            var sections = new List<SectionTarget>();
            
            try
            {
                if (!string.IsNullOrEmpty(rule.TargetFilter) && rule.TargetFilter != "{}")
                {
                    var filter = JObject.Parse(rule.TargetFilter);
                    
                    if (filter["Sections"] is JArray sectionsArray)
                    {
                        foreach (var section in sectionsArray)
                        {
                            var sectionId = section["SectionID"]?.Value<int>();
                            var classId = section["ClassID"]?.Value<int>();
                            var studentIds = section["StudentID"]?.Value<JArray>()
                                ?.Select(x => x.Value<int>()).ToList() ?? new List<int>();
                            
                            if (sectionId.HasValue && classId.HasValue)
                            {
                                sections.Add(new SectionTarget
                                {
                                    SectionID = sectionId.Value,
                                    ClassID = classId.Value,
                                    StudentIDs = studentIds
                                });
                                
                                _logger.LogDebug("📚 Parsed section: SectionID={SectionID}, ClassID={ClassID}, Students={StudentCount}", 
                                    sectionId.Value, classId.Value, studentIds.Count);
                            }
                        }
                    }
                    else if (filter["SectionID"] != null)
                    {
                        var sectionId = filter["SectionID"]?.Value<int>();
                        var studentIds = filter["StudentID"]?.Value<JArray>()
                            ?.Select(x => x.Value<int>()).ToList() ?? new List<int>();
                        
                        if (sectionId.HasValue)
                        {
                            var section = _context.tbl_AddSection
                                .FirstOrDefault(s => s.ID == sectionId.Value);
                            
                            sections.Add(new SectionTarget
                            {
                                SectionID = sectionId.Value,
                                ClassID = section?.Class_ID ?? 0,
                                StudentIDs = studentIds
                            });
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error parsing sections from target filter for rule {RuleID}", rule.RuleID);
            }
            
            return sections;
        }

        private List<int> ParseClassFilter(NotificationRule rule)
        {
            var classIds = new List<int>();
            
            try
            {
                if (!string.IsNullOrEmpty(rule.TargetFilter) && rule.TargetFilter != "{}")
                {
                    var filter = JObject.Parse(rule.TargetFilter);
                    
                    if (filter["ClassID"] is JArray classArray)
                    {
                        classIds = classArray.Select(x => x.Value<int>()).ToList();
                    }
                    else if (filter["ClassID"] != null)
                    {
                        classIds.Add(filter["ClassID"].Value<int>());
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "⚠️ Error parsing class filter for rule {RuleID}", rule.RuleID);
            }
            
            return classIds;
        }

        private List<int> ParseSpecificUsers(NotificationRule rule)

            {

        /// <summary>
        /// Parses specific student IDs from target filter for SPECIFIC_STUDENTS_DIRECT
        /// </summary>
        private List<int> ParseSpecificStudents(NotificationRule rule)
        {
            var studentIds = new List<int>();
            
            try
            {
                if (!string.IsNullOrEmpty(rule.TargetFilter) && rule.TargetFilter != "{}")
                {
                    var filter = JObject.Parse(rule.TargetFilter);
                    
                    if (filter["Students"] is JArray studentArray)
                    {
                        studentIds = studentArray.Select(x => x.Value<int>()).ToList();
                        _logger.LogDebug("📚 Parsed {Count} specific students from Students array: {StudentIds}", 
                            studentIds.Count, string.Join(",", studentIds));
                    }
                    else if (filter["StudentID"] is JArray studentIdArray)
                    {
                        studentIds = studentIdArray.Select(x => x.Value<int>()).ToList();
                    }
                    else if (filter["StudentID"] != null)
                    {
                        studentIds.Add(filter["StudentID"].Value<int>());
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "⚠️ Error parsing specific students filter for rule {RuleID}", rule.RuleID);
            }
            
            return studentIds;
        }

        private async Task ProcessFeeRuleAsync(NotificationRule rule)
        {
            try
            {
                _logger.LogInformation("💰 Processing fee rule {RuleID} - {RuleName}", rule.RuleID, rule.RuleName);

                var ruleSettings = ParseRuleSettings(rule);
                
                var intervalDays = Convert.ToInt32(ruleSettings.GetValueOrDefault("intervalDays", 3));
                var minAmount = Convert.ToInt32(ruleSettings.GetValueOrDefault("minAmount", 0));
                var maxDaysOverdue = Convert.ToInt32(ruleSettings.GetValueOrDefault("maxDaysOverdue", 90));
                var minDaysOverdue = Convert.ToInt32(ruleSettings.GetValueOrDefault("minDaysOverdue", 0));
                var excludePartial = Convert.ToBoolean(ruleSettings.GetValueOrDefault("excludePartialPayments", false));
                
                var classFilter = ruleSettings.ContainsKey("classes") ? 
                    ruleSettings["classes"] as List<int> : null;

                if (rule.LastRunTime.HasValue)
                {
                    var daysSinceLastRun = (GetPakistanTime() - rule.LastRunTime.Value).TotalDays;
                    if (daysSinceLastRun < intervalDays)
                    {
                        _logger.LogInformation("⏸️ Fee rule {RuleID} ran {Days} days ago, interval is {Interval} days", 
                            rule.RuleID, (int)daysSinceLastRun, intervalDays);
                        return;
                    }
                }

                var feeDueStudents = await GetFeeDueStudentsAsync(rule.RuleID, intervalDays, minAmount, 
                    maxDaysOverdue, minDaysOverdue, excludePartial, classFilter);

                _logger.LogInformation("📋 Found {Count} fee due students for rule {RuleID}", 
                    feeDueStudents.Count, rule.RuleID);

                if (feeDueStudents.Count == 0)
                {
                    _logger.LogInformation("📭 No fee due students found for rule {RuleID}", rule.RuleID);
                    await UpdateRuleStatisticsAsync(rule);
                    return;
                }

                Guid? batchId = null;
                if (rule.IsBatch)
                {
                    batchId = await CreateBatchNotificationAsync(rule, feeDueStudents.Count);
                }

                int notificationCount = 0;
                foreach (var student in feeDueStudents)
                {
                    try
                    {
                        var notificationCreated = await CreateFeeNotificationForStudentAsync(rule, student, intervalDays, batchId);
                        if (notificationCreated)
                            notificationCount++;
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(ex, "❌ Error creating fee notification for student {StudentID}", 
                            student.StudentID);
                    }
                }

                if (batchId.HasValue && rule.IsBatch)
                {
                    await UpdateBatchNotificationAsync(batchId.Value, notificationCount);
                }

                await UpdateRuleStatisticsAsync(rule);
                
                _logger.LogInformation("🎉 Fee rule {RuleID} processed successfully. Created {Count} notifications", 
                    rule.RuleID, notificationCount);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error in ProcessFeeRuleAsync for rule {RuleID}", rule.RuleID);
            }
        }

        private async Task<Guid> CreateBatchNotificationAsync(NotificationRule rule, int totalRecipients)
        {
            var batchId = Guid.NewGuid();
            
            var batchNotification = new BatchNotification
            {
                BatchID = batchId,
                RuleID = rule.RuleID,
                Title = rule.TitleTemplate,
                Message = rule.MessageTemplate,
                TargetType = rule.TargetType ?? "ALL",
                TargetFilter = rule.TargetFilter,
                TotalRecipients = totalRecipients,
                Status = "PROCESSING",
                CreatedTime = GetPakistanTime()
            };

            _context.BatchNotifications.Add(batchNotification);
            await _context.SaveChangesAsync();

            _logger.LogInformation("📦 Created batch notification {BatchID} for rule {RuleID} with {Count} recipients", 
                batchId, rule.RuleID, totalRecipients);

            return batchId;
        }

        private async Task UpdateBatchNotificationAsync(Guid batchId, int sentCount)
        {
            var batchNotification = await _context.BatchNotifications
                .FirstOrDefaultAsync(b => b.BatchID == batchId);

        private async Task<List<FeeDueStudent>> GetFeeDueStudentsAsync(int ruleId, int intervalDays, 
            int minAmount, int maxDaysOverdue, int minDaysOverdue, bool excludePartial, List<int>? classFilter)
        {
            var students = new List<FeeDueStudent>();
            
            try
            {
                var sql = @"
                    SELECT DISTINCT 
                        fp.ID as FeePostingID,
                        fp.Reg_ID as StudentID,
                        s.Frist_Name as StudentName,
                        s.RollNo as RollNumber,
                        s.Class_ID as ClassID,
                        s.Section_ID as SectionID,
                        u.ID as ParentID,
                        u.Name as ParentName,
                        u.Mobile_No as ParentMobile,
                        fp.Amount as DueAmount,
                        fp.Fine as LateFee,
                        fp.FeeMonth,
                        fp.FeeYear,
                        fp.IssueDate,
                        fp.DueDate,
                        DATEDIFF(day, fp.DueDate, GETDATE()) as DaysOverdue,
                        fp.ChallanNo,
                        c.Class_Name as ClassName,
                        sec.Section as SectionName,
                        d.FCMToken,
                        ISNULL(fp.Amount, 0) - ISNULL((
                            SELECT SUM(frd.Received) 
                            FROM tbl_FeeReceiveDetail frd
                            INNER JOIN tbl_FeeReceiveMaster frm ON frd.Receipt_ID = frm.Receipt_ID
                            WHERE frd.Challan_No = fp.ChallanNo 
                                AND frm.RegID = fp.Reg_ID
                        ), 0) as RemainingAmount,
                        CASE WHEN EXISTS (
                            SELECT 1 FROM tbl_FeeReceiveDetail frd
                            INNER JOIN tbl_FeeReceiveMaster frm ON frd.Receipt_ID = frm.Receipt_ID
                            WHERE frd.Challan_No = fp.ChallanNo 
                                AND frm.RegID = fp.Reg_ID
                                AND frd.Received > 0
                        ) THEN 1 ELSE 0 END as HasPartialPayment
                    FROM tbl_FeePosting fp
                    INNER JOIN tbl_StudentRegistration s ON fp.Reg_ID = s.Reg_Id
                    INNER JOIN tbl_User u ON s.Family_Code = u.FamilyID
                    INNER JOIN tbl_AddClass c ON s.Class_ID = c.Class_ID
                    LEFT JOIN tbl_AddSection sec ON s.Section_ID = sec.ID
                    LEFT JOIN UserDevices d ON u.ID = d.UserID AND d.IsActive = 1
                    WHERE fp.Type = 'POSTED'
                        AND fp.DueDate <= CAST(GETDATE() AS DATE)
                        AND u.Status = 1
                        AND (s.cur_Status IS NULL OR s.cur_Status IN ('Admission', 'A', 'Active', 'Addmission'))
                        AND fp.Amount >= @minAmount
                        AND DATEDIFF(day, fp.DueDate, GETDATE()) BETWEEN @minDaysOverdue AND @maxDaysOverdue";

                if (classFilter != null && classFilter.Any())
                {
                    sql += $" AND s.Class_ID IN ({string.Join(",", classFilter)})";
                }

                if (excludePartial)
                {
                    sql += " AND NOT EXISTS (" +
                           "    SELECT 1 FROM tbl_FeeReceiveDetail frd" +
                           "    INNER JOIN tbl_FeeReceiveMaster frm ON frd.Receipt_ID = frm.Receipt_ID" +
                           "    WHERE frd.Challan_No = fp.ChallanNo" +
                           "        AND frm.RegID = fp.Reg_ID" +
                           "        AND frd.Received > 0" +
                           ")";
                }

                sql += @"
                    HAVING ISNULL(fp.Amount, 0) > ISNULL((
                        SELECT SUM(frd.Received) 
                        FROM tbl_FeeReceiveDetail frd
                        INNER JOIN tbl_FeeReceiveMaster frm ON frd.Receipt_ID = frm.Receipt_ID
                        WHERE frd.Challan_No = fp.ChallanNo 
                            AND frm.RegID = fp.Reg_ID
                    ), 0)
                    AND NOT EXISTS (
                        SELECT 1 FROM NotificationQueue nq
                        WHERE nq.RuleID = @ruleId
                            AND nq.UserID = u.ID
                            AND nq.CreatedTime >= DATEADD(day, -@intervalDays, CAST(GETDATE() AS DATE))
                            AND nq.Status IN ('SENT', 'READ', 'PENDING', 'PROCESSING')
                    )
                    ORDER BY fp.DueDate ASC, DaysOverdue DESC";

                var parameters = new[]
                {
                    new SqlParameter("@ruleId", ruleId),
                    new SqlParameter("@intervalDays", intervalDays),
                    new SqlParameter("@minAmount", minAmount),
                    new SqlParameter("@maxDaysOverdue", maxDaysOverdue),
                    new SqlParameter("@minDaysOverdue", minDaysOverdue)
                };

                students = await _context.Database.SqlQueryRaw<FeeDueStudent>(sql, parameters).ToListAsync();
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error getting fee due students");
            }

            return students;
        }

        private async Task ProcessAttendanceRuleAsync(NotificationRule rule)
        {
            try
            {
                _logger.LogInformation("📊 Checking attendance rule {RuleID} - {RuleName}", rule.RuleID, rule.RuleName);

                var pakistanNow = GetPakistanTime();
                var today = pakistanNow.Date;
                
                _logger.LogInformation("📅 Looking for absent students on: {Today}", today.ToString("yyyy-MM-dd"));
                
                if (rule.TimingType == "IMMEDIATE" && rule.LastRunTime.HasValue)
                {
                    var minutesSinceLastRun = (pakistanNow - rule.LastRunTime.Value).TotalMinutes;
                    if (minutesSinceLastRun < 5)
                    {
                        _logger.LogInformation("⏸️ Attendance rule {RuleID} ran {Minutes} minutes ago, skipping to avoid duplicates", 
                            rule.RuleID, (int)minutesSinceLastRun);
                        return;
                    }
                }
                
                var sql = @"
                    SELECT DISTINCT 
                        s.Reg_Id as StudentID,
                        u.ID as ParentID,
                        s.Frist_Name as StudentName,
                        s.RollNo as RollNumber,
                        m.AttDate as AbsentDate,
                        a.Status as StatusValue,
                        a.Remarks as StudentRemarks
                    FROM tbl_StudentRegistration s
                    INNER JOIN tbl_User u ON s.Family_Code = u.FamilyID
                    INNER JOIN tbl_StudentAttDetail a ON s.Reg_Id = a.RegID
                    INNER JOIN tbl_StudentAttMaster m ON a.MasterId = m.ID
                    WHERE (
                        UPPER(a.Status) = 'A' OR 
                        UPPER(a.Status) = 'ABSENT' OR
                        a.Status LIKE '%Absent%'
                    )
                    AND CAST(m.AttDate AS DATE) = @todayDate
                    AND u.Status = 1
                    AND (s.cur_Status IS NULL OR s.cur_Status IN ('Admission', 'A'))";
                
                var absentStudents = await _context.Database.SqlQueryRaw<AttendanceRecord>(
                    sql,
                    new SqlParameter("@todayDate", today)
                ).ToListAsync();

                _logger.LogInformation("👥 Found {Count} absent students for rule {RuleID} on {Date}", 
                    absentStudents?.Count ?? 0, rule.RuleID, today.ToString("yyyy-MM-dd"));

                if (absentStudents == null || absentStudents.Count == 0)
                {
                    _logger.LogInformation("📭 No absent students found for rule {RuleID} today", rule.RuleID);
                    await UpdateRuleStatisticsAsync(rule);
                    return;
                }

                var existingParentIds = new HashSet<int>();
                if (rule.TimingType == "IMMEDIATE")
                {
                    existingParentIds = new HashSet<int>(await _context.NotificationQueue
                        .Where(n => n.RuleID == rule.RuleID && n.CreatedTime.Date == today)
                        .Select(n => n.UserID)
                        .Distinct()
                        .ToListAsync());
                    
                    _logger.LogInformation("📋 Found {Count} parents already notified today", existingParentIds.Count);
                }

                Guid? batchId = null;
                if (rule.IsBatch)
                {
                    var parents = absentStudents.Select(s => s.ParentID).Distinct().Count();
                    batchId = await CreateBatchNotificationAsync(rule, parents);
                }

                var studentsByParent = absentStudents
                    .GroupBy(s => s.ParentID)
                    .ToList();

                int notificationCount = 0;
                foreach (var parentGroup in studentsByParent)
                {
                    var parentId = parentGroup.Key;
                    
                    if (existingParentIds.Contains(parentId))
                    {
                        _logger.LogInformation("⏸️ Skipping parent {ParentID} - already notified today", parentId);
                        continue;
                    }
                    
                    var parentStudents = parentGroup.ToList();
                    
                    try
                    {
                        if (parentStudents.Count > 0)
                        {
                            await CreateAttendanceNotificationForParentAsync(rule, parentId, parentStudents, today, batchId);
                            notificationCount++;
                            
                            _logger.LogInformation("✅ Created attendance notification for parent {ParentID} with {Count} absent students", 
                                parentId, parentStudents.Count);
                        }
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(ex, "❌ Error processing attendance for parent {ParentID}", parentId);
                    }
                }

                if (batchId.HasValue && rule.IsBatch)
                {
                    await UpdateBatchNotificationAsync(batchId.Value, notificationCount);
                }

                await UpdateRuleStatisticsAsync(rule);
                
                _logger.LogInformation("🎉 Attendance rule {RuleID} processed successfully. Created {Count} notifications", 
                    rule.RuleID, notificationCount);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error in ProcessAttendanceRuleAsync for rule {RuleID}", rule.RuleID);
            }
        }

        private Dictionary<string, object> ParseRuleSettings(NotificationRule rule)
        {
            var settings = new Dictionary<string, object>();
            
            try
            {
                if (!string.IsNullOrEmpty(rule.TargetFilter) && rule.TargetFilter != "{}")
                {
                    var filter = JObject.Parse(rule.TargetFilter);
                    
                    foreach (var property in filter.Properties())
                    {
                        if (property.Name == "Sections")
                        {
                            settings[property.Name] = property.Value;
                        }
                        else if (property.Value.Type == JTokenType.Integer)
                        {
                            settings[property.Name] = property.Value.Value<int>();
                        }
                        else if (property.Value.Type == JTokenType.Boolean)
                        {
                            settings[property.Name] = property.Value.Value<bool>();
                        }
                        else if (property.Value.Type == JTokenType.Array)
                        {
                            var list = property.Value.ToObject<List<int>>();
                            settings[property.Name] = list ?? new List<int>();
                        }
                        else if (property.Value.Type == JTokenType.String)
                        {
                            settings[property.Name] = property.Value.Value<string>() ?? string.Empty;
                        }
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "⚠️ Error parsing rule settings for rule {RuleID}", rule.RuleID);
            }

            if (!settings.ContainsKey("intervalDays"))
                settings["intervalDays"] = 3;
            if (!settings.ContainsKey("minAmount"))
                settings["minAmount"] = 0;
            if (!settings.ContainsKey("maxDaysOverdue"))
                settings["maxDaysOverdue"] = 90;
            if (!settings.ContainsKey("minDaysOverdue"))
                settings["minDaysOverdue"] = 0;
            if (!settings.ContainsKey("excludePartialPayments"))
                settings["excludePartialPayments"] = false;

            return settings;
        }

        private DateTime? CalculateNextRunTime(NotificationRule rule)
        {
            if (rule.RuleType == "MANUAL" || rule.IsManualTrigger)
                return null;

            var pakistanNow = GetPakistanTime();
            
            if (rule.RuleType == "CRON" && !string.IsNullOrEmpty(rule.CronExpression))
            {
                try
                {
                    var utcNow = pakistanNow.ToUniversalTime();
                    var parts = rule.CronExpression.Trim().Split(' ', StringSplitOptions.RemoveEmptyEntries);
                    
                    if (parts.Length != 5 && parts.Length != 6)
                    {
                        _logger.LogWarning("⚠️ Invalid cron expression format for rule {RuleID}: {CronExpression}", 
                            rule.RuleID, rule.CronExpression);
                        return null;
                    }
                    
                    var format = parts.Length == 6 ? CronFormat.IncludeSeconds : CronFormat.Standard;
                    var expression = CronExpression.Parse(rule.CronExpression, format);
                    
                    var nextUtc = expression.GetNextOccurrence(utcNow, inclusive: false);
                    if (nextUtc.HasValue)
                        return nextUtc.Value.ToLocalTime();
                }
                catch (Exception ex)
                {
                    _logger.LogWarning(ex, "⚠️ Error calculating next run time for cron expression: {CronExpression}", 
                        rule.CronExpression);
                }
            }
            else if (rule.TimingType == "DAYS_OFFSET" && rule.DaysOffset.HasValue)
            {
                return pakistanNow.AddDays(rule.DaysOffset.Value);
            }
            else if (rule.TimingType == "FIXED_TIME" && rule.FixedTime.HasValue)
            {
                var nextRun = pakistanNow.Date.Add(rule.FixedTime.Value);
                if (nextRun <= pakistanNow)
                {
                    nextRun = nextRun.AddDays(1);
                }
                return nextRun;
            }

            return null;
        }

        private string ParseTemplate(string template, Dictionary<string, object> data)
        {
            if (string.IsNullOrEmpty(template))
                return string.Empty;

            var result = template;
            foreach (var kvp in data)
            {
                result = result.Replace($"{{{kvp.Key}}}", kvp.Value?.ToString() ?? "");
            }

            return result;
        }

            {

                    return new PushNotificationResult { Success = false, Message = "Expo token is required" };

                var expoMessage = new
                {
                    to = model.ExpoToken,
                    title = model.Title,
                    body = model.Message,
                    data = model.Data,
                    sound = "default",
                    priority = "high",
                    channelId = "default"
                };

                var messages = new[] { expoMessage };
                var content = new StringContent(JsonConvert.SerializeObject(messages), Encoding.UTF8, "application/json");
                
                _logger.LogInformation("Sending Expo push notification to {Token}", model.ExpoToken);

                var response = await _httpClient.PostAsync(ExpoPushUrl, content);
                var responseString = await response.Content.ReadAsStringAsync();

                if (response.IsSuccessStatusCode)
                {
                    var result = JsonConvert.DeserializeObject<ExpoPushResponse>(responseString);
                    
                    if (result?.Data?.FirstOrDefault()?.Status == "ok")
                    {
                        return new PushNotificationResult 
                        { 
                            Success = true, 
                            Message = "Notification sent successfully",
                            ReceiptId = result.Data.FirstOrDefault()?.Id
                        };
                    }
                    else
                    {
                        return new PushNotificationResult 
                        { 
                            Success = false, 
                            Message = result?.Data?.FirstOrDefault()?.Message ?? "Failed to send notification" 
                        };
                    }
                }

                return new PushNotificationResult 
                { 
                    Success = false, 
                    Message = $"HTTP request failed: {response.StatusCode}" 
                };
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error sending Expo push notification");
                return new PushNotificationResult { Success = false, Message = ex.Message };

using System.Text;
using System.Text.RegularExpressions;

namespace school_management_Backend.Services
{
    public static class HtmlToTextConverter
    {
        public static string ConvertHtmlToPlainText(string html, int maxLength = 0)
        {
            if (string.IsNullOrEmpty(html))
                return string.Empty;

            var result = Regex.Replace(html, @"<script[^>]*>.*?</script>", "", RegexOptions.IgnoreCase | RegexOptions.Singleline);
            result = Regex.Replace(result, @"<style[^>]*>.*?</style>", "", RegexOptions.IgnoreCase | RegexOptions.Singleline);
            result = Regex.Replace(result, @"</?(div|p|h[1-6]|li|tr|section|article)[^>]*>", "\n", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"<br\s*/?>", "\n", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"</?(b|strong)[^>]*>", "", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"</?(i|em)[^>]*>", "", RegexOptions.IgnoreCase);
            result = Regex.Replace(result, @"<[^>]+>", "", RegexOptions.IgnoreCase);
            result = System.Net.WebUtility.HtmlDecode(result);
            result = Regex.Replace(result, @"\n\s*\n", "\n\n");
            result = result.Trim();
            
            if (maxLength > 0 && result.Length > maxLength)
                result = result.Substring(0, maxLength) + "...";
            
            return result;
        }

        public static string ExtractPreview(string html, int maxLength = 100)
        {
            var plainText = ConvertHtmlToPlainText(html);
            if (plainText.Length > maxLength)
                plainText = plainText.Substring(0, maxLength) + "...";
            return plainText;
        }
    }
}

            var device = await _context.UserDevices

        {

            if (model.UserID.HasValue)
            {
                return await _firebaseService.SendToUserAsync(model.UserID.Value, model.Title, model.Message, model.Data);
            }
            else
            {
                return CreateFailedResult("No device token found for sending notification");
            }
        }

        var sendModel = new SendPushNotificationDTO
        {
            ExpoToken = model.ExpoToken,
            Title = isHtml ? HtmlToTextConverter.ConvertHtmlToPlainText(model.Title, 100) : model.Title,
            Message = isHtml ? HtmlToTextConverter.ConvertHtmlToPlainText(model.Message, 200) : model.Message,
            Data = model.Data,
            Sound = model.Sound,
            Priority = model.Priority,
            UserID = model.UserID
        };

        var expoResult = await SendExpoNotificationAsync(sendModel);
        
        if (expoResult.Success)
        {
            return expoResult;
        }
        
        _logger.LogInformation("Expo failed, trying Firebase...");
        
        if (model.UserID.HasValue)
        {
            var firebaseResult = await _firebaseService.SendToUserAsync(model.UserID.Value, model.Title, model.Message, model.Data);
            
            if (firebaseResult.Success)
            {
                return firebaseResult;
            }
        }
        
        return CreateFailedResult("Both Expo and Firebase failed to send notification");
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error sending hybrid push notification");
        return CreateFailedResult(ex.Message);
    }
}

// Services/HybridPushNotificationService.cs - Fix SendToUserAsync
public async Task<PushNotificationResult> SendToUserAsync(int userId, string title, string message, object data = null)
{
    try
    {
        var device = await _context.UserDevices
            .FirstOrDefaultAsync(d => d.UserID == userId && d.IsActive);
            
        if (device == null)
        {
            return CreateFailedResult("No active device found for user");
        }

        bool isHtml = (title.Contains("<") && title.Contains(">")) || 
                     (message.Contains("<") && message.Contains(">"));

        // Extract notification ID from data
        int? notificationId = null;
        if (data != null)
        {
            var dataDict = data as Dictionary<string, object>;
            if (dataDict != null && dataDict.ContainsKey("notificationId"))
            {
                var idValue = dataDict["notificationId"]?.ToString();
                if (!string.IsNullOrEmpty(idValue) && int.TryParse(idValue, out int id))
                {
                    notificationId = id;
                    _logger.LogInformation("Extracted notificationId from data in SendToUserAsync: {NotificationId}", notificationId);
                }
            }
        }

        // Try FCM first for HTML content
        if (isHtml && !string.IsNullOrEmpty(device.FCMToken))
        {
            var titlePlain = HtmlToTextConverter.ConvertHtmlToPlainText(title, 100);
            var messagePlain = HtmlToTextConverter.ConvertHtmlToPlainText(message, 200);

            var dataMessage = new SendFirebaseDataMessageDTO
            {
                FCMToken = device.FCMToken,
                Title = title,
                Message = message,
                TitleHtml = title,
                MessageHtml = message,
                TitlePlain = titlePlain,
                MessagePlain = messagePlain,
                Data = data,
                Priority = "high",
                NotificationType = "html_notification",
                NotificationId = notificationId
            };
            
            var firebaseResult = await _firebaseService.SendDataMessageAsync(dataMessage);
            
            if (firebaseResult.Success)
            {
                return firebaseResult;
            }
        }
        
        // Try Expo for plain text or as fallback
        if (!string.IsNullOrEmpty(device.ExpoToken))
        {
            var plainTitle = isHtml ? HtmlToTextConverter.ConvertHtmlToPlainText(title, 100) : title;
            var plainMessage = isHtml ? HtmlToTextConverter.ConvertHtmlToPlainText(message, 200) : message;
            
            var expoResult = await SendExpoNotificationAsync(new SendPushNotificationDTO
            {
                ExpoToken = device.ExpoToken,
                Title = plainTitle,
                Message = plainMessage,
                Data = data
            });
            
            if (expoResult.Success)
            {
                return expoResult;
            }
        }
        
        // Try FCM as last resort for plain text
        if (!string.IsNullOrEmpty(device.FCMToken))
        {
            var plainTitle = isHtml ? HtmlToTextConverter.ConvertHtmlToPlainText(title, 100) : title;
            var plainMessage = isHtml ? HtmlToTextConverter.ConvertHtmlToPlainText(message, 200) : message;
            
            var firebaseResult = await _firebaseService.SendToUserAsync(userId, plainTitle, plainMessage, data);
            
            if (firebaseResult.Success)
            {
                return firebaseResult;
            }
        }
        
        return CreateFailedResult("No valid device tokens found for sending notification");
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error sending to user {UserId}", userId);
        return CreateFailedResult(ex.Message);
    }
}

        private async Task<PushNotificationResult> SendExpoNotificationAsync(SendPushNotificationDTO model)
        {
            try
            {
                var httpClient = _httpClientFactory.CreateClient("expo");
                
                var expoMessage = new
                {
                    to = model.ExpoToken,
                    title = model.Title,
                    body = model.Message,
                    data = model.Data,
                    sound = model.Sound,
                    priority = model.Priority
                };

                var json = JsonSerializer.Serialize(expoMessage);
                var content = new StringContent(json, Encoding.UTF8, "application/json");

                var response = await httpClient.PostAsync("send", content);
                var responseString = await response.Content.ReadAsStringAsync();

                if (response.IsSuccessStatusCode)
                {
                    using var doc = JsonDocument.Parse(responseString);
                    var root = doc.RootElement;
                    
                    var receiptId = root.GetProperty("data").GetProperty("id").GetString();
                    
                    return new PushNotificationResult
                    {
                        Success = true,
                        Message = "Expo notification sent successfully",
                        ReceiptId = receiptId,
                        NotificationId = 0,
                        MessageId = receiptId,
                        ShouldDeactivateDevice = false
                    };
                }
                else
                {
                    _logger.LogWarning("Expo API error: {StatusCode}, Response: {Response}", 
                        response.StatusCode, responseString);
                    
                    bool shouldDeactivate = responseString.Contains("DeviceNotRegistered") || 
                                           responseString.Contains("InvalidToken") ||
                                           responseString.Contains("NotRegistered");
                    
                    return new PushNotificationResult
                    {
                        Success = false,
                        Message = $"Expo error: {responseString}",
                        ShouldDeactivateDevice = shouldDeactivate
                    };
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error sending Expo notification");
                return new PushNotificationResult
                {
                    Success = false,
                    Message = $"Expo error: {ex.Message}",
                    ShouldDeactivateDevice = false
                };
            }
        }

        private PushNotificationResult CreateFailedResult(string message, bool shouldDeactivate = false)
        {
            return new PushNotificationResult
            {
                Success = false,
                Message = message,
                ShouldDeactivateDevice = shouldDeactivate
            };
        }
    }

        public async Task<List<NotificationQueue>> ExecuteRuleAsync(NotificationRule rule)
        {

            {

                    rule.RuleID, notifications.Count);

        public async Task ExecuteRuleManuallyAsync(int ruleId)
        {
            try
            {
                _logger.LogInformation("🚀 MANUAL EXECUTION triggered for rule ID: {RuleId}", ruleId);
                
                var rule = await _context.NotificationRules
                    .FirstOrDefaultAsync(r => r.RuleID == ruleId);
                
                if (rule == null)
                {
                    _logger.LogWarning("⚠️ Rule {RuleId} not found for manual execution", ruleId);
                    return;
                }
                
                if (!rule.IsActive)
                {
                    _logger.LogWarning("⏸️ Rule {RuleId} is inactive, cannot execute manually", ruleId);
                    return;
                }
                
                _logger.LogInformation("⚡ Manually executing rule: {RuleName} (ID: {RuleID}, Type: {RuleType}, Category: {Category}, Manual: {IsManual})", 
                    rule.RuleName, rule.RuleID, rule.RuleType, rule.Category, rule.IsManualTrigger);
                
                var targetUsers = await GetTargetUsersAsync(rule);
                
                _logger.LogInformation("👥 Found {Count} target users for manual rule {RuleID}", targetUsers.Count, rule.RuleID);
                
                if (targetUsers.Count == 0)
                {
                    _logger.LogInformation("📭 Manual rule {RuleID} executed with 0 target users", rule.RuleID);
                    await UpdateRuleStatisticsForManualExecutionAsync(rule);
                    return;
                }
                
                List<NotificationQueue> notifications;
                
                switch (rule.Category?.ToUpper())
                {
                    case "FEE":
                        notifications = await ProcessFeeNotificationsAsync(rule, targetUsers);
                        break;
                    case "ATTENDANCE":
                        notifications = await ProcessAttendanceNotificationsAsync(rule, targetUsers);
                        break;
                    case "EXAM":
                        notifications = await ProcessExamNotificationsAsync(rule, targetUsers);
                        break;
                    case "HOMEWORK":
                        notifications = await ProcessHomeworkNotificationsAsync(rule, targetUsers);
                        break;
                    case "ANNOUNCEMENT":
                        notifications = await ProcessAnnouncementNotificationsAsync(rule, targetUsers);
                        break;
                    case "GENERAL":
                        notifications = await ProcessGeneralNotificationsAsync(rule, targetUsers);
                        break;
                    case "ALERT":
                        notifications = await ProcessAlertNotificationsAsync(rule, targetUsers);

        // ==================== TIMING VALIDATION ====================

        public async Task<bool> ShouldRuleRunNowAsync(NotificationRule rule)
        {
            try
            {
                var pakistanNow = GetPakistanTime();
                
                if (rule.RuleType == "MANUAL")
                    return false;
                
                if (rule.RuleType == "EVENT")
                {
                    if (rule.IsManualTrigger)
                    {
                        _logger.LogDebug("⏸️ Rule {RuleID} is manual trigger, skipping automatic execution", rule.RuleID);
                        return false;
                    }
                    
                    return true;
                }
                
                if (rule.RuleType == "CRON" || rule.RuleType == "SCHEDULED")
                {
                    return await CheckScheduleTimingAsync(rule, pakistanNow);
                }
                
                return true;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error checking if rule should run now");
                return false;
            }
        }

        private async Task<bool> CheckScheduleTimingAsync(NotificationRule rule, DateTime pakistanNow)
        {
            switch (rule.TimingType?.ToUpper())
            {
                case "IMMEDIATE":
                    return true;
                    
                case "FIXED_TIME":
                    if (!rule.FixedTime.HasValue) return false;
                    var scheduledTime = pakistanNow.Date.Add(rule.FixedTime.Value);
                    var timeDifference = Math.Abs((pakistanNow - scheduledTime).TotalMinutes);
                    return timeDifference <= 15;
                    
                case "DAYS_OFFSET":
                    if (!rule.DaysOffset.HasValue || !rule.LastRunTime.HasValue) return false;
                    var daysSinceLastRun = (pakistanNow.Date - rule.LastRunTime.Value.Date).Days;
                    return daysSinceLastRun >= rule.DaysOffset.Value;
                    
                case "CRON":
                    if (string.IsNullOrEmpty(rule.CronExpression)) return false;
                    return await CheckCronExpressionAsync(rule.CronExpression, pakistanNow);
                    
                case "DAILY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 1;
                    
                case "WEEKLY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 7;
                    
                case "BIWEEKLY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 14;
                    
                case "MONTHLY":
                    if (!rule.LastRunTime.HasValue) return true;
                    return (pakistanNow.Date - rule.LastRunTime.Value.Date).Days >= 30;
                    
                case "WEEKDAY":
                    return pakistanNow.DayOfWeek != DayOfWeek.Saturday && pakistanNow.DayOfWeek != DayOfWeek.Sunday;
                    
                case "WEEKEND":
                    return pakistanNow.DayOfWeek == DayOfWeek.Saturday || pakistanNow.DayOfWeek == DayOfWeek.Sunday;
                    
                default:
                    return true;
            }
        }

        private async Task<bool> CheckCronExpressionAsync(string cronExpression, DateTime pakistanNow)
        {
            try
            {
                var utcNow = pakistanNow.ToUniversalTime();
                var parts = cronExpression.Trim().Split(' ', StringSplitOptions.RemoveEmptyEntries);
                
                if (parts.Length != 5 && parts.Length != 6)
                {
                    _logger.LogError("❌ Invalid cron expression format: {Cron}", cronExpression);
                    return false;
                }
                
                var format = parts.Length == 6 ? CronFormat.IncludeSeconds : CronFormat.Standard;
                var parsedExpression = CronExpression.Parse(cronExpression, format);
                
                var nextOccurrence = parsedExpression.GetNextOccurrence(utcNow.AddMinutes(-5), inclusive: false);
                
                if (nextOccurrence.HasValue)
                {
                    var nextLocal = nextOccurrence.Value.ToLocalTime();
                    var timeDiff = Math.Abs((pakistanNow - nextLocal).TotalMinutes);
                    return timeDiff <= 5;
                }
                
                return false;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error parsing cron expression: {Cron}", cronExpression);
                return false;
            }
        }

        private async Task<List<NotificationQueue>> ProcessFeeNotificationsAsync(NotificationRule rule, List<dynamic> targetUsers)
        {
            var notifications = new List<NotificationQueue>();

            {
                var ruleSettings = ParseRuleSettings(rule);

            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error processing fee notifications");
            }
            
            return notifications;
        }

private async Task<NotificationQueue> CreateFeeNotificationAsync(
    NotificationRule rule, 
    dynamic userData, 
    Dictionary<string, object> ruleSettings,
    Guid? batchId = null)
{
    try
    {
        var userId = GetUserId(userData);
        if (userId == 0)
        {
            _logger.LogWarning("⚠️ Invalid UserID for notification rule {RuleID}", rule.RuleID);
            return null;
        }

        int concreteUserId = userId;
        var pakistanNow = GetPakistanTime();
        var today = pakistanNow.Date;

        var device = await _context.UserDevices
            .FirstOrDefaultAsync(d => d.UserID == concreteUserId && d.IsActive && !string.IsNullOrEmpty(d.FCMToken));
        
        string? fcmToken = device?.FCMToken;

        if (string.IsNullOrEmpty(fcmToken))
        {
            _logger.LogWarning("⚠️ No FCM token found for user {UserID} in rule {RuleID}", 
                concreteUserId, rule.RuleID);
            return null;
        }

        string userType = GetUserType(userData) ?? "USER";

        string notificationType = "FEE_REMINDER";
        string statusText = "Fee pending";
        
        if (userData.DueDate is DateOnly dueDateObj)
        {
            var dueDate = dueDateObj.ToDateTime(TimeOnly.MinValue);
            var daysUntilDue = (dueDate - today).Days;
            var daysOverdue = (today - dueDate).Days;
            
            if (daysUntilDue > 0)
            {
                notificationType = $"DUE_IN_{daysUntilDue}_DAYS";
                statusText = $"Due in {daysUntilDue} days";
            }
            else if (daysOverdue >= 0)
            {
                notificationType = $"OVERDUE_{daysOverdue}_DAYS";
                statusText = $"Overdue by {daysOverdue} days";
            }
        }

        var dataDict = ConvertToDictionary(userData);
        dataDict["NotificationType"] = notificationType;
        dataDict["StatusText"] = statusText;
        dataDict["CurrentDate"] = today.ToString("dd/MM/yyyy");
        dataDict["CurrentTime"] = pakistanNow.ToString("HH:mm");
        
        var titleHtml = await ResolveTemplate(rule.TitleTemplate ?? "Fee Payment Reminder", dataDict);
        var messageHtml = await ResolveTemplate(rule.MessageTemplate ?? "{StudentName} has pending fee", dataDict);
        
        var titlePlain = HtmlToTextConverter.ConvertHtmlToPlainText(titleHtml, 100);
        var messagePlain = HtmlToTextConverter.ConvertHtmlToPlainText(messageHtml, 200);
        
        if (string.IsNullOrEmpty(titlePlain)) titlePlain = "Fee Payment Reminder";
        if (string.IsNullOrEmpty(messagePlain)) messagePlain = "You have a pending fee payment";
        
        string dataPayload;
        if (!string.IsNullOrEmpty(rule.DataTemplate))
        {
            dataPayload = await ResolveTemplate(rule.DataTemplate, dataDict);
        }
        else
        {
            var payloadData = new Dictionary<string, object>
            {
                ["action"] = "view_fee_details",
                ["studentId"] = userData.StudentID,
                ["challanNo"] = userData.ChallanNo,
                ["feePostingId"] = userData.FeePostingID,
                ["amount"] = userData.RemainingAmount,
                ["dueDate"] = userData.DueDate?.ToString("yyyy-MM-dd"),
                ["daysOverdue"] = userData.DaysOverdue,
                ["notificationType"] = notificationType,
                ["category"] = rule.Category,
                ["titleHtml"] = titleHtml,
                ["messageHtml"] = messageHtml,
                ["isHtml"] = true
            };
            
            if (batchId.HasValue)
                payloadData["batchId"] = batchId.Value;
            
            dataPayload = JsonConvert.SerializeObject(payloadData);
        }

        var notification = new NotificationQueue
        {
            RuleID = rule.RuleID,
            UserID = concreteUserId,
            UserType = userType,
            DeviceToken = fcmToken,
            Title = titlePlain,
            Message = messagePlain,
            DataPayload = dataPayload,
            Status = "PENDING",
            Priority = rule.Priority,
            ScheduledTime = pakistanNow,
            ExpiryTime = pakistanNow.AddDays(7),
            CreatedTime = pakistanNow,
            AttemptCount = 0,
            LastAttemptTime = null,
            ErrorMessage = null
        };

        _context.NotificationQueue.Add(notification);
        await _context.SaveChangesAsync();
        
        return notification;
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "❌ Error creating fee notification");
        return null;
    }
}

        private async Task<List<NotificationQueue>> ProcessAttendanceNotificationsAsync(NotificationRule rule, List<dynamic> targetUsers)
        {
            var notifications = new List<NotificationQueue>();
            
            try
            {
                var ruleSettings = ParseRuleSettings(rule);
                
                var parents = targetUsers
                    .GroupBy(u => (int)u.UserID)
                    .Select(g => new
                    {
                        ParentID = g.Key,
                        Students = g.ToList(),
                        DeviceToken = g.First().DeviceToken
                    })
                    .ToList();

                Guid? batchId = null;
                if (rule.IsBatch)
                {
                    batchId = await CreateBatchNotificationAsync(rule, parents.Count);
                }

                foreach (var parent in parents)
                {
                    try
                    {
                        var notification = await CreateAttendanceNotificationAsync(rule, parent.ParentID, parent.Students, ruleSettings, batchId);
                        if (notification != null)
                            notifications.Add(notification);
                    }
                    catch (Exception ex)
                    {
                        _logger.LogError(ex, "❌ Error creating attendance notification for parent {ParentID}", parent.ParentID);
                    }
                }

                if (batchId.HasValue)
                {
                    await UpdateBatchNotificationAsync(batchId.Value, notifications.Count);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error processing attendance notifications");
            }
            
            return notifications;
        }

        private async Task<Guid> CreateBatchNotificationAsync(NotificationRule rule, int totalRecipients)
        {
            try
            {
                var batchId = Guid.NewGuid();

                {
                    BatchID = batchId,
                    RuleID = rule.RuleID,
                    Title = rule.TitleTemplate,
                    Message = rule.MessageTemplate,
                    TargetType = rule.TargetType ?? "ALL",
                    TargetFilter = rule.TargetFilter,
                    TotalRecipients = totalRecipients,
                    Status = "PROCESSING",
                    CreatedTime = GetPakistanTime()
                };

                _context.BatchNotifications.Add(batchNotification);
                await _context.SaveChangesAsync();

                _logger.LogInformation("📦 Created batch notification {BatchID} for rule {RuleID} with {Count} recipients", 
                    batchId, rule.RuleID, totalRecipients);

                return batchId;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error creating batch notification");
                return Guid.Empty;
            }
        }

        private async Task UpdateBatchNotificationAsync(Guid batchId, int sentCount)
        {
            try
            {
                var batchNotification = await _context.BatchNotifications
                    .FirstOrDefaultAsync(b => b.BatchID == batchId);

                if (batchNotification != null)
                {
                    batchNotification.SentCount = sentCount;
                    batchNotification.FailedCount = batchNotification.TotalRecipients - sentCount;
                    batchNotification.Status = sentCount > 0 ? "COMPLETED" : "FAILED";
                    batchNotification.CompletedTime = GetPakistanTime();

                    await _context.SaveChangesAsync();

                    _logger.LogInformation("📊 Updated batch notification {BatchID}: Sent={Sent}, Failed={Failed}", 
                        batchId, sentCount, batchNotification.FailedCount);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error updating batch notification");
            }
        }

        private Dictionary<string, object> ParseRuleSettings(NotificationRule rule)
        {
            var settings = new Dictionary<string, object>();
            
            try
            {
                if (!string.IsNullOrEmpty(rule.TargetFilter) && rule.TargetFilter != "{}")
                {
                    settings = JsonConvert.DeserializeObject<Dictionary<string, object>>(rule.TargetFilter) ?? new Dictionary<string, object>();
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "⚠️ Error parsing rule settings for rule {RuleID}", rule.RuleID);
            }

            return settings;
        }

        public async Task<DateTime?> CalculateNextRunTime(NotificationRule rule)
        {
            if (!rule.IsActive || string.IsNullOrEmpty(rule.TimingType))
                return null;

            if (rule.RuleType == "MANUAL" || rule.IsManualTrigger)
                return null;

            var pakistanNow = GetPakistanTime();

            switch (rule.TimingType.ToUpper())
            {
                case "IMMEDIATE":
                    return pakistanNow.AddMinutes(5);
                    
                case "FIXED_TIME":
                    if (rule.FixedTime.HasValue)
                    {
                        var nextRun = pakistanNow.Date.Add(rule.FixedTime.Value);
                        if (nextRun <= pakistanNow)
                            nextRun = nextRun.AddDays(1);
                        return nextRun;
                    }
                    break;
                    
                case "DAYS_OFFSET":
                    if (rule.DaysOffset.HasValue)
                        return pakistanNow.AddDays(rule.DaysOffset.Value);
                    break;
                    
                case "CRON":
                    if (!string.IsNullOrEmpty(rule.CronExpression))
                    {
                        try
                        {
                            var utcNow = pakistanNow.ToUniversalTime();
                            var parts = rule.CronExpression.Trim().Split(' ', StringSplitOptions.RemoveEmptyEntries);
                            
                            if (parts.Length != 5 && parts.Length != 6)
                            {
                                return null;
                            }
                            
                            var format = parts.Length == 6 ? CronFormat.IncludeSeconds : CronFormat.Standard;
                            var schedule = CronExpression.Parse(rule.CronExpression, format);
                            
                            var nextUtc = schedule.GetNextOccurrence(utcNow, inclusive: false);
                            if (nextUtc.HasValue)
                                return nextUtc.Value.ToLocalTime();
                        }
                        catch (Exception ex)
                        {
                            _logger.LogError(ex, "❌ Error parsing cron expression for rule {RuleID}: {Expression}", 
                                rule.RuleID, rule.CronExpression);
                        }
                    }
                    break;
                    
                case "DAILY":
                    return pakistanNow.AddDays(1);
                    
                case "WEEKLY":
                    return pakistanNow.AddDays(7);
                    
                case "BIWEEKLY":
                    return pakistanNow.AddDays(14);
                    
                case "MONTHLY":
                    return pakistanNow.AddMonths(1);
            }

            return null;
        }

        public async Task ScheduleRule(NotificationRule rule)
        {
            try
            {
                if (rule.RuleType == "CRON" && !string.IsNullOrEmpty(rule.CronExpression) && !rule.IsManualTrigger)
                {
                    RecurringJob.RemoveIfExists($"notification-rule-{rule.RuleID}");
                    
                    RecurringJob.AddOrUpdate<INotificationRuleProcessor>(

                        rule.CronExpression,
                        TimeZoneInfo.Local);
                    
                    _logger.LogInformation("📅 Scheduled cron job for rule {RuleID} with expression: {Expression}", 
                        rule.RuleID, rule.CronExpression);
                }
                
                await Task.CompletedTask;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error scheduling rule {RuleID}", rule.RuleID);
            }
        }

        public async Task<string> ResolveTemplate(string template, dynamic data)
        {
            if (string.IsNullOrEmpty(template))
                return string.Empty;

            var result = template;
            
            try
            {
                var dataDict = ConvertToDictionary(data);

                foreach (var prop in dataDict)
                {
                    var placeholder = "{" + prop.Key + "}";
                    var value = prop.Value?.ToString() ?? string.Empty;
                    
                    if (result.Contains(placeholder))
                    {
                        result = result.Replace(placeholder, value);
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "⚠️ Error resolving template: {Template}", template);
            }

            return result;
        }

        private Dictionary<string, object> ConvertToDictionary(dynamic data)
        {
            var dict = new Dictionary<string, object>();
            
            if (data is ExpandoObject expando)
            {
                var expandoDict = expando as IDictionary<string, object>;
                foreach (var kvp in expandoDict)
                {
                    dict[kvp.Key] = kvp.Value;
                }
            }
            else if (data is IDictionary<string, object> dataDict)
            {
                dict = new Dictionary<string, object>(dataDict);
            }
            
            return dict;
        }

public async Task ProcessNotificationAsync(NotificationQueue notification)
{
    try
    {
        _logger.LogInformation("Processing notification {QueueID} for user {UserID}", notification.QueueID, notification.UserID);

        notification.Status = "PROCESSING";
        notification.LastAttemptTime = DateTime.Now;
        notification.AttemptCount++;
        
        await _context.SaveChangesAsync();
        
        string fcmToken = notification.DeviceToken;
        
        if (string.IsNullOrEmpty(fcmToken))
        {
            var device = await _context.UserDevices
                .FirstOrDefaultAsync(d => d.UserID == notification.UserID && d.IsActive && !string.IsNullOrEmpty(d.FCMToken));
                
            if (device == null || string.IsNullOrEmpty(device.FCMToken))
            {
                notification.Status = "FAILED";
                notification.ErrorMessage = "No active FCM token found";
                await _context.SaveChangesAsync();
                
                await LogNotificationActionAsync(notification.QueueID, notification.UserID, "FAILED", "No active FCM token found");
                _logger.LogWarning("No FCM token for user {UserID}", notification.UserID);
                return;
            }
            
            fcmToken = device.FCMToken;
            notification.DeviceToken = fcmToken;
            await _context.SaveChangesAsync();
        }
        
        // FIX: Create a proper Dictionary instead of JObject
        var dataDict = new Dictionary<string, object>();
        
        if (!string.IsNullOrEmpty(notification.DataPayload))
        {
            try
            {
                var existingData = JsonConvert.DeserializeObject<Dictionary<string, object>>(notification.DataPayload);
                if (existingData != null)
                {
                    foreach (var kvp in existingData)
                    {
                        dataDict[kvp.Key] = kvp.Value;
                    }
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "Failed to parse data payload for notification {QueueID}", notification.QueueID);
            }
        }
        
        // Add notification ID to data - AS STRING, NOT ARRAY
        dataDict["notificationId"] = notification.QueueID.ToString();
        dataDict["queueID"] = notification.QueueID.ToString();
        dataDict["type"] = "notification";
        dataDict["timestamp"] = DateTime.Now.ToString("o");
        
        // Create a custom DTO that includes the notification ID
        var sendDto = new SendPushNotificationDTO
        {
            UserID = notification.UserID,
            Title = notification.Title,
            Message = notification.Message,
            Data = dataDict,  // Use the dictionary instead of JObject
            Sound = "default",
            Priority = "high"
        };
        
        var pushResult = await _pushService.SendPushNotificationAsync(sendDto);
        
        if (pushResult.Success)
        {
            notification.Status = "SENT";
            notification.ProcessedTime = DateTime.Now;
            
            await LogNotificationActionAsync(notification.QueueID, notification.UserID, "SENT", pushResult.Message);
            _logger.LogInformation("Notification {QueueID} sent successfully", notification.QueueID);
        }
        else
        {
            notification.Status = "FAILED";
            notification.ErrorMessage = pushResult.Message;
            
            if (pushResult.ShouldDeactivateDevice)
            {
                var device = await _context.UserDevices
                    .FirstOrDefaultAsync(d => d.UserID == notification.UserID && d.FCMToken == fcmToken);
                if (device != null)
                {
                    device.IsActive = false;
                    _logger.LogWarning("Deactivated device for user {UserID} due to invalid token", notification.UserID);
                }
            }
            
            if (notification.AttemptCount >= 3)
            {
                await ArchiveFailedNotificationAsync(notification);
                _logger.LogInformation("Moved notification {QueueID} to history after {Attempts} failed attempts", 
                    notification.QueueID, notification.AttemptCount);
            }
            
            await _context.SaveChangesAsync();
        }
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Error processing notification {QueueID}", notification.QueueID);
        
        notification.Status = "FAILED";
        notification.ErrorMessage = ex.Message;
        await _context.SaveChangesAsync();
    }

        public async Task<NotificationQueue> CreateNotificationAsync(int ruleId, int userId, Dictionary<string, object> data)
        {
            var rule = await _context.NotificationRules.FindAsync(ruleId);

            // Resolve template with data
            var title = ResolveTemplate(rule.TitleTemplate, data);
            var message = ResolveTemplate(rule.MessageTemplate, data);

            var notification = new NotificationQueue
            {

                UserID = userId,

                DataPayload = dataPayload,
                Status = "PENDING",
                Priority = rule.Priority,
                ScheduledTime = CalculateScheduledTime(rule),
                CreatedTime = DateTime.Now,
                ExpiryTime = DateTime.Now.AddDays(7),
                DeviceToken = null // Will be populated when sending

        private DateTime CalculateScheduledTime(NotificationRule rule)
        {
            var now = DateTime.Now;

            switch (rule.TimingType)
            {
                case "FIXED_TIME":
                    if (rule.FixedTime.HasValue)
                    {
                        var scheduled = DateTime.Today.Add(rule.FixedTime.Value);
                        if (scheduled <= now)
                            scheduled = scheduled.AddDays(1);
                        return scheduled;
                    }
                    break;

                case "DAYS_OFFSET":
                    if (rule.DaysOffset.HasValue)
                        return now.AddDays(rule.DaysOffset.Value);
                    break;

                case "IMMEDIATE":
                    return now;
            }

            return now;
        }

                Category = rule?.Category ?? "GENERAL",
                SentTime = notification.CreatedTime,
                ArchivedDate = DateTime.Now,
                // You might want to add error information here
                // ErrorDetails = notification.ErrorMessage
            };

            _context.NotificationHistory.Add(history);
            _context.NotificationQueue.Remove(notification);
            
            _logger.LogDebug("Archived failed notification {QueueID}", notification.QueueID);
        }

        private void InitializeFirebase()
        {
            try
            {
                if (FirebaseApp.DefaultInstance == null)
                {
                    var serviceAccountKeyPath = _configuration["Firebase:ServiceAccountKeyPath"];
                    
                    if (!string.IsNullOrEmpty(serviceAccountKeyPath) && File.Exists(serviceAccountKeyPath))
                    {
                        FirebaseApp.Create(new AppOptions()
                        {
                            Credential = GoogleCredential.FromFile(serviceAccountKeyPath),
                            ProjectId = _configuration["Firebase:ProjectId"]
                        });
                        _firebaseInitialized = true;
                        _logger.LogInformation("✅ Firebase Admin SDK initialized successfully");
                    }
                    else
                    {
                        _logger.LogWarning("⚠️ Firebase service account key not found at: {Path}", serviceAccountKeyPath);
                    }
                }
                else
                {
                    _firebaseInitialized = true;
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error initializing Firebase");
                _firebaseInitialized = false;
            }
        }

        public async Task<PushNotificationResult> SendNotificationAsync(SendFirebaseNotificationDTO model)
        {
            try
            {
                if (!_firebaseInitialized)
                    return CreateFailedResult("Firebase not initialized");

                if (string.IsNullOrEmpty(model.FCMToken))
                    return CreateFailedResult("FCM token is required");

                var message = new Message()
                {
                    Token = model.FCMToken,
                    Notification = new Notification()
                    {
                        Title = model.Title,
                        Body = model.Message
                    },
                    Data = ConvertToDictionary(model.Data),
                    Android = new AndroidConfig()
                    {
                        Priority = model.Priority == "high" ? Priority.High : Priority.Normal,
                        Notification = new AndroidNotification()
                        {
                            Sound = "default",
                            ChannelId = "default",
                            Icon = "ic_notification"
                        }
                    },
                    Apns = new ApnsConfig()
                    {
                        Aps = new Aps()
                        {
                            Alert = new ApsAlert()
                            {
                                Title = model.Title,
                                Body = model.Message
                            },
                            Sound = "default",
                            Badge = 1
                        }
                    }
                };

                string messageId = await FirebaseMessaging.DefaultInstance.SendAsync(message);
                _logger.LogInformation("✅ Firebase notification sent: {MessageId}", messageId);

                return CreateSuccessResult(messageId);
            }
            catch (FirebaseMessagingException ex)
            {
                _logger.LogError(ex, "❌ Firebase messaging error");
                bool shouldDeactivate = ex.MessagingErrorCode == MessagingErrorCode.Unregistered ||
                                       ex.MessagingErrorCode == MessagingErrorCode.InvalidArgument;
                return CreateFailedResult(ex.Message, shouldDeactivate);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error sending Firebase notification");
                return CreateFailedResult(ex.Message);
            }
        }

// Services/PureFirebaseNotificationService.cs - Update SendDataMessageAsync
public async Task<PushNotificationResult> SendDataMessageAsync(SendFirebaseDataMessageDTO model)
{
    try
    {
        if (!_firebaseInitialized)
            return CreateFailedResult("Firebase not initialized");

        if (string.IsNullOrEmpty(model.FCMToken))
            return CreateFailedResult("FCM token is required");

        var dataDict = new Dictionary<string, string>();
        
        string titlePlain = model.TitlePlain;
        string messagePlain = model.MessagePlain;
        
        if (string.IsNullOrEmpty(titlePlain) && !string.IsNullOrEmpty(model.Title))
        {
            titlePlain = HtmlToTextConverter.ConvertHtmlToPlainText(model.Title, 100);
        }
        
        if (string.IsNullOrEmpty(messagePlain) && !string.IsNullOrEmpty(model.Message))
        {
            messagePlain = HtmlToTextConverter.ConvertHtmlToPlainText(model.Message, 200);
        }
        
        if (string.IsNullOrEmpty(titlePlain)) titlePlain = "School Notification";
        if (string.IsNullOrEmpty(messagePlain)) messagePlain = "You have a new notification";
        
        _logger.LogInformation("📤 Sending to FCM - Title: '{Title}', Message: '{Message}'", titlePlain, messagePlain);
        
        dataDict["title"] = titlePlain;
        dataDict["body"] = messagePlain;
        dataDict["click_action"] = "FLUTTER_NOTIFICATION_CLICK";
        
        if (!string.IsNullOrEmpty(model.TitleHtml))
            dataDict["title_html"] = model.TitleHtml;
        if (!string.IsNullOrEmpty(model.MessageHtml))
            dataDict["message_html"] = model.MessageHtml;
        
        dataDict["is_html"] = "true";
        dataDict["type"] = model.NotificationType ?? "notification";
        dataDict["timestamp"] = DateTime.UtcNow.ToString("o");
        
        // ADD NOTIFICATION ID TO PAYLOAD
        if (model.NotificationId.HasValue)
        {
            dataDict["notificationId"] = model.NotificationId.Value.ToString();
            dataDict["queueID"] = model.NotificationId.Value.ToString();
        }
        
        if (model.Data != null)
        {
            var customData = ConvertToDictionary(model.Data);
            foreach (var kvp in customData)
            {
                if (!dataDict.ContainsKey(kvp.Key))
                    dataDict[kvp.Key] = kvp.Value;
            }
        }

        var message = new Message()
        {
            Token = model.FCMToken,
            Data = dataDict,
            Android = new AndroidConfig()
            {
                Priority = model.Priority == "high" ? Priority.High : Priority.Normal,
                Notification = new AndroidNotification()
                {
                    Title = titlePlain,
                    Body = messagePlain,
                    Sound = "default",
                    ChannelId = "html_notifications",
                    ClickAction = "FLUTTER_NOTIFICATION_CLICK"
                }
            },
            Apns = new ApnsConfig()
            {
                Aps = new Aps()
                {
                    Alert = new ApsAlert()
                    {
                        Title = titlePlain,
                        Body = messagePlain
                    },
                    Sound = "default",
                    Badge = 1,
                    ContentAvailable = true
                },
                Headers = new Dictionary<string, string>()
                {
                    { "apns-priority", "10" }
                }
            }
        };

        string messageId = await FirebaseMessaging.DefaultInstance.SendAsync(message);
        _logger.LogInformation("✅ Firebase notification sent with ID: {NotificationId}", model.NotificationId);

        return CreateSuccessResult(messageId);
    }
    catch (FirebaseMessagingException ex)
    {
        _logger.LogError(ex, "❌ Firebase messaging error");
        bool shouldDeactivate = ex.MessagingErrorCode == MessagingErrorCode.Unregistered ||
                               ex.MessagingErrorCode == MessagingErrorCode.InvalidArgument;
        return CreateFailedResult(ex.Message, shouldDeactivate);
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "❌ Error sending Firebase data message");
        return CreateFailedResult(ex.Message);
    }
}

        public async Task<PushNotificationResult> SendToUserAsync(int userId, string title, string message, object data = null)
        {
            try
            {
                var device = await _context.UserDevices
                    .FirstOrDefaultAsync(d => d.UserID == userId && d.IsActive && !string.IsNullOrEmpty(d.FCMToken));
                    
                if (device == null)
                    return CreateFailedResult("No active device with FCM token found");

                // Check if this is HTML content
                bool isHtml = (title.Contains("<") && title.Contains(">")) || 
                             (message.Contains("<") && message.Contains(">"));
                
                if (isHtml)
                {
                    var titlePlain = HtmlToTextConverter.ConvertHtmlToPlainText(title, 100);
                    var messagePlain = HtmlToTextConverter.ConvertHtmlToPlainText(message, 200);
                    
                    var dataMessage = new SendFirebaseDataMessageDTO
                    {
                        FCMToken = device.FCMToken,
                        Title = title,
                        Message = message,
                        TitleHtml = title,
                        MessageHtml = message,
                        TitlePlain = titlePlain,
                        MessagePlain = messagePlain,
                        Data = data,
                        Priority = "high",
                        NotificationType = "html_notification"
                    };
                    return await SendDataMessageAsync(dataMessage);
                }
                else
                {
                    var notification = new SendFirebaseNotificationDTO
                    {
                        FCMToken = device.FCMToken,
                        Title = title,
                        Message = message,
                        Data = data,
                        Priority = "high"
                    };
                    return await SendNotificationAsync(notification);
                }
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error sending to user {UserId}", userId);
                return CreateFailedResult(ex.Message);
            }
        }

        public async Task<BatchNotificationResult> SendBatchAsync(List<SendFirebaseNotificationDTO> models)
        {
            var result = new BatchNotificationResult();
            
            try
            {
                if (!_firebaseInitialized)
                    throw new InvalidOperationException("Firebase not initialized");

                var messages = models
                    .Where(m => !string.IsNullOrEmpty(m.FCMToken))
                    .Select(CreateFirebaseMessage)
                    .ToList();

                if (!messages.Any())
                {
                    _logger.LogWarning("No valid FCM tokens in batch");
                    return result;
                }

                var batchResponse = await FirebaseMessaging.DefaultInstance.SendEachAsync(messages);

                result.SuccessCount = batchResponse.SuccessCount;
                result.FailureCount = batchResponse.FailureCount;
                
                _logger.LogInformation("✅ Firebase batch sent: {Success}/{Total}",
                    batchResponse.SuccessCount, messages.Count);

                return result;
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "❌ Error sending Firebase batch");
                throw;
            }
        }

        private Dictionary<string, string> ConvertToDictionary(object data)
        {
            var dict = new Dictionary<string, string>();
            if (data == null) return dict;

            try
            {
                var json = JsonSerializer.Serialize(data);
                using var doc = JsonDocument.Parse(json);
                
                foreach (var prop in doc.RootElement.EnumerateObject())
                {
                    dict[prop.Name] = prop.Value.ToString();
                }
            }
            catch (Exception ex)
            {
                _logger.LogWarning(ex, "Error converting data to dictionary");
            }

            return dict;
        }

        private PushNotificationResult CreateSuccessResult(string messageId = null)
        {
            return new PushNotificationResult
            {
                Success = true,
                Message = "Firebase notification sent successfully",
                MessageId = messageId,
                ShouldDeactivateDevice = false
            };
        }

        private PushNotificationResult CreateFailedResult(string error, bool shouldDeactivate = false)
        {
            return new PushNotificationResult
            {
                Success = false,
                Message = error,
                ShouldDeactivateDevice = shouldDeactivate
            };
