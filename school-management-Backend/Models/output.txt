        // NEW: Flag to mark manual trigger rules

        // Event Configuration
        [StringLength(50)]
        public string? SourceTable { get; set; }
        
        [StringLength(50)]
        public string? TriggerField { get; set; }
        
        [StringLength(100)]
        public string? TriggerValue { get; set; }
        
        // Timing Configuration
        [StringLength(20)]
        public string TimingType { get; set; } = "IMMEDIATE"; // 'IMMEDIATE', 'FIXED_TIME', 'DAYS_OFFSET', 'CRON'
        
        public int? DaysOffset { get; set; }
        
        public TimeSpan? FixedTime { get; set; }
        
        [StringLength(50)]
        public string? CronExpression { get; set; }
        
        // Content Templates
        [Required]
        [StringLength(200)]
        public string TitleTemplate { get; set; } = string.Empty;
        
        [Required]
        [StringLength(500)]
        public string MessageTemplate { get; set; } = string.Empty;
        
        public string? DataTemplate { get; set; }
        
        // Target Configuration
        [Required]
        [StringLength(20)]
        public string TargetType { get; set; } = "ALL"; // 'ALL', 'CLASS', 'SECTION', 'PARENT', 'STUDENT', 'TEACHER'
        
        public string? TargetFilter { get; set; }
        
        // Delivery Options
        public bool IsBatch { get; set; } = false;
        
        [Range(1, 3)]

        // Status & Tracking
        public bool IsActive { get; set; } = true;
        
        public DateTime? LastRunTime { get; set; }
        
        public DateTime? NextRunTime { get; set; }
        
        public int TotalRuns { get; set; } = 0;
        
        // Audit
        public int CreatedBy { get; set; }

        public int? ModifiedBy { get; set; }

    }

using System.ComponentModel.DataAnnotations;
using System.ComponentModel.DataAnnotations.Schema;

namespace school_management_Backend.Models
{
    [Table("UserDevices")]
    public class UserDevice
    {
        [Key]
        public int DeviceID { get; set; }
        
        public int UserID { get; set; }
        
        [Required]
        [StringLength(20)]
        public string UserType { get; set; }
        
        // Store both Expo and FCM tokens
        [StringLength(500)]
        public string? ExpoToken { get; set; }
        
        [StringLength(500)]
        public string FCMToken { get; set; }
        
        // Device Info
        [Required]
        [StringLength(10)]
        public string Platform { get; set; } // 'ANDROID', 'IOS', 'WEB'
        
        [StringLength(100)]
        public string DeviceModel { get; set; }
        
        [StringLength(20)]
        public string AppVersion { get; set; }
        
        [StringLength(20)]
        public string OSVersion { get; set; }
        
        // Status
        public bool IsActive { get; set; } = true;
        
        public DateTime LastActive { get; set; } = DateTime.Now;
        
        public DateTime CreatedDate { get; set; } = DateTime.Now;
        
        // Helper property to get the best token
        [NotMapped]
        public string PreferredToken => !string.IsNullOrEmpty(ExpoToken) ? ExpoToken : FCMToken;
    }
}
